7// FAPL-scoped parallel filter pipeline — second CRTP descendant of
8// pipeline_t<Derived> alongside basic_pipeline_t (synchronous).
9//
10// pool_pipeline_t owns a strong reference to an external worker_pool_t
11// (resolved from the file's FAPL) and submits compress closures to it;
12// H5Dwrite_chunk runs on the calling thread.
13//
14// The pipeline reuses pipeline_t<>::write/read (and via that
15// split_to_chunk_write/read) for chunk decomposition, so consumer sites
16// — pt_t, h5::write, h5::read — never see the pool dispatch directly.
17// They just construct a pool_pipeline_t when their file's FAPL has one.

30struct pool_pipeline_t : public pipeline_t<pool_pipeline_t> {
31 pool_pipeline_t() = default; // variant requires default-constructible
32 // when this is the chosen alternative; pool_ stays null
33 // and dispatch fast-fails (no pool installed).
34
35 pool_pipeline_t(std::shared_ptr<worker_pool_t> pool, unsigned cap)
36 : pool_(std::move(pool)), cap_(cap) {}
37
38 ~pool_pipeline_t() {
39 // Drain any remaining in-flight work in submission order before the
40 // pipeline (and its shared_ptr to the pool) goes away. Blocks the
41 // destructor on each front future.
42 while (!in_flight_.empty()) drain_in_flight(/*blocking=*/true);
43 while (!read_in_flight_.empty()) drain_in_flight_read(nullptr, /*blocking=*/true);
44 }
45
46 pool_pipeline_t(const pool_pipeline_t&) = delete;
47 pool_pipeline_t& operator=(const pool_pipeline_t&) = delete;
48 pool_pipeline_t(pool_pipeline_t&&) = delete;
49 pool_pipeline_t& operator=(pool_pipeline_t&&) = delete;
50
51 // CRTP entry points called by pipeline_t<>::write / pipeline_t<>::read
52 // via split_to_chunk_write / split_to_chunk_read.
53 void write_chunk_impl(const hsize_t* offset, std::size_t nbytes, const void* src);
54 void read_chunk_impl (const hsize_t* offset, std::size_t nbytes, void* dst);
55
56 // Parallel read override — shadows pipeline_t<pool_pipeline_t>::read.
57 // Rank-1 chunked datasets are decompressed across the worker pool;
58 // rank>1 falls back to the synchronous base-class path.
59 void read(const h5::ds_t& ds, const h5::offset_t& offset,
60 const h5::stride_t& stride, const h5::block_t& block,
61 const h5::count_t& count, const h5::dxpl_t& dxpl, void* ptr);
62
63 // Public quiesce — block until every in-flight future has been
64 // drained. Called by pt_t::flush so users get the expected
65 // "data on disk after flush returns" semantics for the pool path.
66 void drain() {
67 while (!in_flight_.empty()) drain_in_flight(/*blocking=*/true);
68 while (!read_in_flight_.empty()) drain_in_flight_read(nullptr, /*blocking=*/true);
69 }
70
71private:
72 // Worker-produced compressed chunk, ready for H5Dwrite_chunk on the
73 // calling thread. unique_ptr<std::byte[]> keeps it move-only and
74 // aligned to operator new[]'s default alignment.
75 struct result_t {
76 std::unique_ptr<std::byte[]> data;
77 std::size_t nbytes{0};
78 std::uint32_t mask{0};
79 std::array<hsize_t, H5CPP_MAX_RANK> offset{};
80 };
81
82 void drain_in_flight(bool blocking);
83
84 // Result of a worker decompression task for the parallel read path.
85 struct read_result_t {
86 std::unique_ptr<std::byte[]> data;
87 std::size_t copy_size{0};
88 std::size_t dst_offset{0};
89 };
90
91 void drain_in_flight_read(char* dst, bool blocking);
92
93 std::shared_ptr<worker_pool_t> pool_;
94 unsigned cap_{0};
95 std::deque<std::future<result_t>> in_flight_;
96 std::deque<std::future<read_result_t>> read_in_flight_;
97 char* read_dst_ = nullptr;
98};

102inline void pool_pipeline_t::drain_in_flight(bool blocking) {
103 using namespace std::chrono_literals;
104 if (blocking) {
105 if (in_flight_.empty()) return;
106 auto r = in_flight_.front().get(); // blocks
107 H5Dwrite_chunk(static_cast<::hid_t>(this->ds), static_cast<::hid_t>(this->dxpl),
108 r.mask, r.offset.data(), r.nbytes, r.data.get());
109 in_flight_.pop_front();
110 return;
111 }
112 while (!in_flight_.empty() &&
113 in_flight_.front().wait_for(0s) == std::future_status::ready) {
114 auto r = in_flight_.front().get();
115 H5Dwrite_chunk(static_cast<::hid_t>(this->ds), static_cast<::hid_t>(this->dxpl),
116 r.mask, r.offset.data(), r.nbytes, r.data.get());
117 in_flight_.pop_front();
118 }
119}

121inline void pool_pipeline_t::write_chunk_impl(const hsize_t* offset_in,
122 std::size_t nbytes,
123 const void* src) {
124 if (!pool_) {
125 // No pool was wired in (default-constructed alternative). Fall
126 // through to the synchronous filter chain identical to
127 // basic_pipeline_t::write_chunk_impl. This branch shouldn't be
128 // reached in normal use; it exists to keep the variant safe.
129 if (tail == 0) {
130 H5Dwrite_chunk(static_cast<::hid_t>(ds), static_cast<::hid_t>(dxpl),
131 0, offset_in, nbytes, src);
132 return;
133 }
134 // Skip-aware ping-pong (see basic_pipeline_t::write_chunk_impl, #287): a
135 // filter returning 0 keeps the data + length, sets its mask bit.
136 if (src != chunk0) std::memcpy(chunk0, src, nbytes);
137 void* bufs[2] = { chunk0, chunk1 };
138 int cur = 0;
139 std::size_t length = nbytes;
140 std::uint32_t mask = 0;
141 for (hsize_t j = 0; j < tail; ++j) {
142 int nxt = cur ^ 1;
143 std::size_t n = filter[j](bufs[nxt], bufs[cur], length, flags[j], cd_size[j], cd_values[j]);
144 if (n == 0) mask |= (1u << j);
145 else { cur = nxt; length = n; }
146 }
147 H5Dwrite_chunk(static_cast<::hid_t>(ds), static_cast<::hid_t>(dxpl),
148 mask, offset_in, length, bufs[cur]);
149 return;
150 }

152 // Snapshot filter chain into a POD captured by value in the closure.
153 struct filter_chain_t {
154 filter::call_t filter[H5CPP_MAX_FILTER];
155 unsigned flags[H5CPP_MAX_FILTER];
156 std::size_t cd_size[H5CPP_MAX_FILTER];
157 unsigned cd_values[H5CPP_MAX_FILTER][H5CPP_MAX_FILTER_PARAM];
158 hsize_t tail;
159 };
160 filter_chain_t fc{};
161 fc.tail = this->tail;
162 std::memcpy(fc.filter, this->filter, sizeof(fc.filter));
163 std::memcpy(fc.flags, this->flags, sizeof(fc.flags));
164 std::memcpy(fc.cd_size, this->cd_size, sizeof(fc.cd_size));
165 std::memcpy(fc.cd_values, this->cd_values, sizeof(fc.cd_values));
166
167 // Worker-owned input buffer.
168 auto raw = std::make_unique<std::byte[]>(nbytes);
169 std::memcpy(raw.get(), src, nbytes);
170
171 std::array<hsize_t, H5CPP_MAX_RANK> off{};
172 std::copy(offset_in, offset_in + this->rank, off.begin());
173
174 auto fut = pool_->submit(
175 [raw = std::move(raw), nbytes, off, fc]() mutable -> result_t {
176 result_t out;
177 out.offset = off;
178
179 if (fc.tail == 0) {
180 out.data = std::move(raw);
181 out.nbytes = nbytes;
182 out.mask = 0;
183 return out;
184 }
185
186 const std::size_t scratch = filter::filter_scratch_bound(nbytes);
187 auto wbuf0 = std::make_unique<std::byte[]>(scratch);
188 auto wbuf1 = std::make_unique<std::byte[]>(scratch);
189
190 std::uint32_t mask = 0;
191 // A filter returning 0 (e.g. deflate on incompressible data) is SKIPPED:
192 // its mask bit is set and the data passes through UNCHANGED, so the
193 // running buffer + length must stay put rather than collapse to a
194 // 0-length chunk (which reads back as garbage). Mirrors
195 // basic_pipeline_t::write_chunk_impl. #287.
196 std::memcpy(wbuf0.get(), raw.get(), nbytes);
197 std::byte* bufs[2] = { wbuf0.get(), wbuf1.get() };
198 int cur = 0;
199 std::size_t length = nbytes;
200 for (hsize_t j = 0; j < fc.tail; ++j) {
201 int nxt = cur ^ 1;
202 std::size_t n = fc.filter[j](bufs[nxt], bufs[cur], length,
203 fc.flags[j], fc.cd_size[j], fc.cd_values[j]);
204 if (n == 0) mask |= (1u << j);
205 else { cur = nxt; length = n; }
206 }
207
208 out.data = std::make_unique<std::byte[]>(length);
209 std::memcpy(out.data.get(), bufs[cur], length);
210 out.nbytes = length;
211 out.mask = mask;
212 return out;
213 });
214
215 in_flight_.push_back(std::move(fut));
216
217 // Opportunistic drain.
218 drain_in_flight(/*blocking=*/false);
219
220 // Bounded back-pressure: block on the front future when the deque
221 // reaches the cap. Producer memory ≤ cap × chunk_size per pipeline.
222 while (in_flight_.size() >= cap_)
223 drain_in_flight(/*blocking=*/true);
224}

226// ─── read path ───────────────────────────────────────────────────────────────
227//
228// Phase 1.3.3 kept read synchronous. Phase 1.5 adds parallel decompression
229// for rank-1 chunked datasets by running the reverse filter chain on the
230// worker pool while the calling thread performs sequential H5Dread_chunk I/O.
231
232inline void pool_pipeline_t::drain_in_flight_read(char* dst, bool blocking) {
233 using namespace std::chrono_literals;
234 char* base = dst ? dst : read_dst_;
235 if (!base) return; // no destination available (shouldn't happen outside read())
236
237 if (blocking) {
238 if (read_in_flight_.empty()) return;
239 auto r = read_in_flight_.front().get();
240 std::memcpy(base + r.dst_offset, r.data.get(), r.copy_size);
241 read_in_flight_.pop_front();
242 return;
243 }
244 while (!read_in_flight_.empty() &&
245 read_in_flight_.front().wait_for(0s) == std::future_status::ready) {
246 auto r = read_in_flight_.front().get();
247 std::memcpy(base + r.dst_offset, r.data.get(), r.copy_size);
248 read_in_flight_.pop_front();
249 }
250}

252inline void pool_pipeline_t::read(const h5::ds_t& ds, const h5::offset_t& offset,
253 const h5::stride_t& stride, const h5::block_t& block,
254 const h5::count_t& count, const h5::dxpl_t& dxpl, void* ptr) {
255 // Fallback to synchronous base-class path when no pool is present or when
256 // the dataset is not rank-1. Rank>1 parallel decompression is future work.
257 if (!pool_ || rank != 1) {
258 pipeline_t<pool_pipeline_t>::read(ds, offset, stride, block, count, dxpl, ptr);
259 return;
260 }
261
262 this->dxpl = dxpl;
263 this->ds = ds;
264
265 h5::offset_t offset_;
266 h5::count_t count_;
267 for (hsize_t i = 0; i < rank; ++i) {
268 offset_[i] = offset[i];
269 count_[i] = count[i] * block[i];
270 }
271
272 // Rank-1 chunk decomposition (mirrors split_to_chunk_read).
273 hsize_t n = count_[0]; // total elements to read
274 hsize_t b = B[0]; // elements per chunk
275 hsize_t ry = n % b; // trailing partial chunk
276
277 // Snapshot filter chain into a POD closure capture (same pattern as write).
278 struct filter_chain_t {
279 filter::call_t filter[H5CPP_MAX_FILTER];
280 unsigned flags[H5CPP_MAX_FILTER];
281 std::size_t cd_size[H5CPP_MAX_FILTER];
282 unsigned cd_values[H5CPP_MAX_FILTER][H5CPP_MAX_FILTER_PARAM];
283 hsize_t tail;
284 };
285 filter_chain_t fc{};
286 fc.tail = this->tail;
287 std::memcpy(fc.filter, this->filter, sizeof(fc.filter));
288 std::memcpy(fc.flags, this->flags, sizeof(fc.flags));
289 std::memcpy(fc.cd_size, this->cd_size, sizeof(fc.cd_size));
290 std::memcpy(fc.cd_values, this->cd_values, sizeof(fc.cd_values));
291
292 const std::size_t scratch = filter::filter_scratch_bound(block_size);
293 char* dst = static_cast<char*>(ptr);
294 read_dst_ = dst;
295 const std::size_t block_sz = this->block_size;
296
297 for (hsize_t j = 0; j < n; j += b) {
298 hsize_t chunk_offset = j + offset_[0];
299 hsize_t copy_size = (ry != 0 && j == n - ry) ? ry * element_size : b * element_size;
300 hsize_t dst_offset = j * element_size;
301
302 // Main thread performs HDF5 I/O (H5Dread_chunk is not thread-safe).
303 auto compressed = std::make_unique<std::byte[]>(scratch);
304 std::uint32_t filter_mask = 0;
305 hsize_t C[1] = {chunk_offset};
306 std::size_t compressed_size = block_sz; // pre-2.0 default (deflate stream self-terminates)
307#if H5_VERSION_GE(2,0,0)
308 compressed_size = scratch;
309 H5Dread_chunk2(static_cast<::hid_t>(ds), static_cast<::hid_t>(dxpl),
310 C, &filter_mask, compressed.get(), &compressed_size); // -> actual compressed bytes
311#else
312 H5Dread_chunk(static_cast<::hid_t>(ds), static_cast<::hid_t>(dxpl),
313 C, &filter_mask, compressed.get());
314#endif
315
316 // Submit decompression to the worker pool.
317 auto fut = pool_->submit([
318 compressed = std::move(compressed), block_sz, copy_size,
319 dst_offset, scratch, fc, compressed_size, filter_mask
320 ]() mutable -> read_result_t {
321 read_result_t result;
322 result.copy_size = copy_size;
323 result.dst_offset = dst_offset;
324
325 if (fc.tail == 0) {
326 result.data = std::make_unique<std::byte[]>(copy_size);
327 std::memcpy(result.data.get(), compressed.get(), copy_size);
328 return result;
329 }
330
331 auto work_buf = std::make_unique<std::byte[]>(scratch);
332 void* src = compressed.get();
333 void* dst_buf = work_buf.get();
334 std::size_t length = compressed_size; // compressed input length for the first reverse filter
335 (void)block_sz;
336
337 for (hsize_t fi = fc.tail; fi > 0; --fi) {
338 const hsize_t idx = fi - 1;
339 if (filter_mask & (1u << idx)) // chunk stored without this filter — pass through
340 std::memcpy(dst_buf, src, length);
341 else
342 length = fc.filter[idx](dst_buf, src, length,
343 fc.flags[idx] | H5Z_FLAG_REVERSE,
344 fc.cd_size[idx], fc.cd_values[idx]);
345 std::swap(src, dst_buf);
346 }
347
348 result.data = std::make_unique<std::byte[]>(copy_size);
349 std::memcpy(result.data.get(), src, copy_size);
350 return result;
351 });
352
353 read_in_flight_.push_back(std::move(fut));
354
355 // Opportunistic drain — if front future is ready, copy and free buffer.
356 drain_in_flight_read(dst, /*blocking=*/false);
357
358 // Bounded back-pressure: block when the deque reaches the cap.
359 while (read_in_flight_.size() >= cap_)
360 drain_in_flight_read(dst, /*blocking=*/true);
361 }
362
363 // Drain any remaining decompression tasks.
364 while (!read_in_flight_.empty())
365 drain_in_flight_read(dst, /*blocking=*/true);
366
367 read_dst_ = nullptr;
368}

370inline void pool_pipeline_t::read_chunk_impl(const hsize_t* offset_in,
371 std::size_t nbytes,
372 void* /*dst*/) {
373 std::size_t length = nbytes;
374 std::uint32_t filter_mask;
375
376 if (tail == 0) {
377#if H5_VERSION_GE(2,0,0)
378 std::size_t buf_size = nbytes;
379 H5Dread_chunk2(static_cast<::hid_t>(ds), dxpl, offset_in, &filter_mask, chunk0, &buf_size);
380#else
381 H5Dread_chunk(static_cast<::hid_t>(ds), dxpl, offset_in, &filter_mask, chunk0);
382#endif
383 return;
384 }
385
386 void* read_target = (tail % 2 == 1) ? chunk1 : chunk0;
387#if H5_VERSION_GE(2,0,0)
388 std::size_t buf_size = filter::filter_scratch_bound(nbytes); // buffer capacity (expanding filters); see H5Zpipeline_basic.hpp
389 H5Dread_chunk2(static_cast<::hid_t>(ds), dxpl, offset_in, &filter_mask, read_target, &buf_size);
390 length = buf_size; // OUT: stored chunk bytes — the reverse-filter input size
391#else
392 H5Dread_chunk(static_cast<::hid_t>(ds), dxpl, offset_in, &filter_mask, read_target);
393#endif
394
395 void* src = read_target;
396 void* dst = (read_target == chunk0) ? static_cast<void*>(chunk1)
397 : static_cast<void*>(chunk0);
398 for (hsize_t j = tail; j > 0; --j) {
399 const hsize_t fi = j - 1;
400 if (filter_mask & (1u << fi)) // chunk stored without this filter — pass through
401 std::memcpy(dst, src, length);
402 else
403 length = filter[fi](dst, src, length,
404 flags[fi] | H5Z_FLAG_REVERSE,
405 cd_size[fi], cd_values[fi]);
406 std::swap(src, dst);
407 }
408}
