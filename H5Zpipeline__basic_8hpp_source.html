8inline void h5::impl::basic_pipeline_t::write_chunk_impl( const hsize_t* offset, size_t nbytes, const void* data ){
9 size_t length = nbytes; // filter may change this, think of compression
10 uint32_t mask = 0x0; // filter mask = 0x0 all filters applied
11
12 // Determine if all filters in the chain are in-place safe.
13 // Shuffle, fletcher32, scaleoffset, nbit, and add are safe.
14 // Compression filters (gzip, szip, lz4, zstd) require separate buffers.
15 bool all_inplace = true;
16 for (hsize_t j = 0; j < tail; ++j) {
17 if (filter_id[j] != H5Z_FILTER_SHUFFLE &&
18 filter_id[j] != H5Z_FILTER_FLETCHER32 &&
19 filter_id[j] != H5Z_FILTER_SCALEOFFSET &&
20 filter_id[j] != H5Z_FILTER_NBIT) {
21 all_inplace = false;
22 break;
23 }
24 }
25
26 if (tail == 0) {
27 // no filters, ( if blocking ) -> data == chunk0 otherwise directly from container
28 H5Dwrite_chunk(static_cast<::hid_t>(ds), dxpl, 0x0, offset, nbytes, data);
29 return;
30 }
31
32 if (all_inplace) {
33 // In-place path: copy data once, then apply all filters on chunk0.
34 std::memcpy(chunk0, data, nbytes);
35 void* buf = chunk0;
36 for (hsize_t j = 0; j < tail; ++j) {
37 length = filter[j](buf, buf, length, flags[j], cd_size[j], cd_values[j]);
38 if (!length)
39 mask |= 1u << j;
40 }
41 H5Dwrite_chunk(static_cast<::hid_t>(ds), dxpl, mask, offset, length, buf);
42 return;
43 }
44
45 // Ping-pong path for filters that require separate buffers. A filter that
46 // returns 0 (e.g. deflate on incompressible data that won't shrink, common
47 // for tiny chunks) is SKIPPED: its mask bit is set and the data passes through
48 // UNCHANGED. The running buffer + length must therefore stay put — NOT collapse
49 // to a 0-length chunk (which writes an empty chunk that reads back as garbage).
50 // HDF5 honours the per-chunk mask on read and skips that filter's inverse. #287.
51 if (data != chunk0) std::memcpy(chunk0, data, nbytes);
52 void* buf[2] = { chunk0, chunk1 };
53 int cur = 0;
54 length = nbytes;
55 for (hsize_t j = 0; j < tail; ++j) {
56 int nxt = cur ^ 1;
57 size_t n = filter[j](buf[nxt], buf[cur], length, flags[j], cd_size[j], cd_values[j]);
58 if (n == 0)
59 mask |= 1u << j; // skip — buffer + length unchanged
60 else { cur = nxt; length = n; }
61 }
62 // direct write available from > 1.10.4
63 H5Dwrite_chunk(static_cast<::hid_t>(ds), dxpl, mask, offset, length, buf[cur]);
64}

67inline void h5::impl::basic_pipeline_t::read_chunk_impl( const hsize_t* offset, size_t nbytes, void* data){
68 (void)data;
69 size_t length = nbytes;
70 uint32_t filter_mask;
71
72 if (tail == 0) {
73 // No filters: read decompressed chunk directly into chunk0
74#if H5_VERSION_GE(2,0,0)
75 size_t buf_size = nbytes;
76 H5Dread_chunk2(static_cast<::hid_t>(ds), dxpl, offset, &filter_mask, chunk0, &buf_size);
77#else
78 H5Dread_chunk(static_cast<::hid_t>(ds), dxpl, offset, &filter_mask, chunk0);
79#endif
80 return;
81 }
82
83 // Parity-based buffer selection: N ping-pong swaps must leave the result in chunk0.
84 // If tail is odd we start reading into chunk1; if even, into chunk0.
85 // After the loop 'src' always points to chunk0 (the decompressed result).
86 void* read_target = (tail % 2 == 1) ? chunk1 : chunk0;
87#if H5_VERSION_GE(2,0,0)
88 // buf_size is IN (buffer capacity) / OUT (stored chunk bytes). Pass the real
89 // scratch-bound capacity, NOT nbytes: an EXPANDING filter (e.g. fletcher32 adds a
90 // 4-byte checksum) stores a chunk LARGER than the uncompressed nbytes, and 2.x
91 // H5Dread_chunk2 fails if the declared capacity is smaller than the stored chunk.
92 size_t buf_size = filter::filter_scratch_bound(nbytes);
93 H5Dread_chunk2(static_cast<::hid_t>(ds), dxpl, offset, &filter_mask, read_target, &buf_size);
94 // H5Dread_chunk2 returns the COMPRESSED byte count in buf_size; the reverse
95 // filter must start from that, not the uncompressed `nbytes`. Pre-2.0 H5Dread_chunk
96 // gave no size and the deflate stream self-terminated, so `nbytes` happened to
97 // work; HDF5 2.x is strict and decodes garbage / fails unless we use buf_size.
98 length = buf_size;
99#else
100 H5Dread_chunk(static_cast<::hid_t>(ds), dxpl, offset, &filter_mask, read_target);
101#endif
102
103 void* src = read_target;
104 void* dst = (read_target == chunk0) ? static_cast<void*>(chunk1) : static_cast<void*>(chunk0);
105
106 // Apply filters in reverse order (highest index first) with H5Z_FLAG_REVERSE.
107 // HONOUR filter_mask: a set bit means HDF5 stored the chunk WITHOUT that filter
108 // (e.g. deflate skipped a chunk where compression didn't pay — HDF5 2.x does this
109 // more readily than 1.x). Reversing a filter that was never applied corrupts the
110 // data, so for a masked filter we pass the bytes through unchanged (still swapping
111 // to keep the ping-pong parity that lands the result in chunk0).
112 for (hsize_t j = tail; j > 0; --j) {
113 const hsize_t fi = j - 1;
114 if (filter_mask & (1u << fi))
115 std::memcpy(dst, src, length);
116 else
117 length = filter[fi](dst, src, length,
118 flags[fi] | H5Z_FLAG_REVERSE, cd_size[fi], cd_values[fi]);
119 void* tmp = src; src = dst; dst = tmp;
120 }
121 // src now points to chunk0, which holds the decompressed chunk data
122}
