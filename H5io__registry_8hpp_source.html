22// close_global — close a conversion-off descriptor under the global HDF5 lock.
23//
24// Every HDF5 close on a conversion-off handle runs while holding the one global
25// HDF5 lock (capi_lock), so it never races another thread's C-API call. The
26// H5Iis_valid check is done INSIDE the lock — never an unlocked C-API call.
27// Defined here because capi_lock / capi_close_t come from H5Iall.hpp, which
28// includes this header after defining them.
29inline void close_global(::hid_t handle, capi_close_t capi_close) {
30 if (handle <= 0) return;
31 capi_lock _lk; // serialize with every other HDF5 C-API call
32 if (!H5Iis_valid(handle)) return;
33 capi_close(handle); // the actual close — under the lock
34}
35#endif
