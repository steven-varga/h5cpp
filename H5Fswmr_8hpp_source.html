41// ── filesystem-class detection ───────────────────────────────────────────────
42// SWMR relies on POSIX page-cache coherence between writer and readers on one
43// host; network / layered / parallel filesystems break it. Policy: WARN at open,
44// never refuse. Opt out with H5CPP_SWMR_NO_FS_CHECK=1.

64 enum class fs_verdict { local_ok, network_bad, layered_caution, parallel_caution, unknown };
65 struct fs_info { fs_verdict verdict = fs_verdict::unknown; const char* name = "unknown"; };
66
67 inline fs_info classify( unsigned long f_type ) {
68 switch( f_type ) {
69 case h5_EXT4_MAGIC: return { fs_verdict::local_ok, "ext4" };
70 case h5_XFS_MAGIC: return { fs_verdict::local_ok, "xfs" };
71 case h5_BTRFS_MAGIC: return { fs_verdict::local_ok, "btrfs" };
72 case h5_TMPFS_MAGIC: return { fs_verdict::local_ok, "tmpfs" };
73 case h5_F2FS_MAGIC: return { fs_verdict::local_ok, "f2fs" };
74 case h5_NFS_MAGIC: return { fs_verdict::network_bad, "nfs" };
75 case h5_SMB_MAGIC: return { fs_verdict::network_bad, "smb" };
76 case h5_CIFS_MAGIC: return { fs_verdict::network_bad, "cifs" };
77 case h5_OVERLAYFS_MAGIC: return { fs_verdict::layered_caution, "overlayfs" };
78 case h5_LUSTRE_MAGIC: return { fs_verdict::parallel_caution, "lustre" };
79 case h5_GPFS_MAGIC: return { fs_verdict::parallel_caution, "gpfs" };
80 case h5_BEEGFS_MAGIC: return { fs_verdict::parallel_caution, "beegfs" };
81 case h5_FUSE_MAGIC: return { fs_verdict::unknown, "fuse" };
82 default: return { fs_verdict::unknown, "unknown" };
83 }
84 }

86 // Classify the filesystem backing `path`; warn (once, to cerr) when SWMR
87 // coherence is at risk. Never throws, never refuses. Returns the verdict.
88 inline fs_verdict check_filesystem( const std::string& path ) noexcept {
89 if( const char* off = std::getenv("H5CPP_SWMR_NO_FS_CHECK");
90 off != nullptr && off[0] != '\0' && off[0] != '0' )
91 return fs_verdict::unknown;
92
93 struct statfs sfs;
94 if( ::statfs( path.c_str(), &sfs ) != 0 )
95 return fs_verdict::unknown;
96
97 const fs_info info = classify( static_cast<unsigned long>( sfs.f_type ) );
98 switch( info.verdict ) {
99 case fs_verdict::network_bad:
100 std::cerr << "h5cpp SWMR WARNING: '" << path << "' is on " << info.name
101 << " — network filesystems break reader cache coherence; "
102 "SWMR visibility is NOT guaranteed (issue #267).\n";
103 break;
104 case fs_verdict::layered_caution:
105 std::cerr << "h5cpp SWMR WARNING: '" << path << "' is on " << info.name
106 << " — layered/overlay filesystems may hide writer flushes "
107 "from readers in other namespaces (issue #267).\n";
108 break;
109 case fs_verdict::parallel_caution:
110 std::cerr << "h5cpp SWMR WARNING: '" << path << "' is on " << info.name
111 << " — parallel filesystems require correct mount options for "
112 "SWMR coherence; not certified (issue #267).\n";
113 break;
114 case fs_verdict::unknown:
115 case fs_verdict::local_ok:
116 break;
117 }
118 return info.verdict;
119 }
120}

124 // h5::create(path, h5::swmr_write, h5::latest_version) — creates a NORMAL
125 // latest-format RDWR file. SWMR can't be enabled at create (no datasets yet);
126 // call h5::start_swmr_write(fd) once the datasets exist. Requires latest_version.
127 template<class... args_t>
128 inline h5::fd_t create( const std::string& path, h5::swmr_write_t, args_t&&... args ) {
129 using tlibver = typename arg::tpos<const h5::libver_bounds&, const args_t&...>;
130 static_assert( tlibver::present,
131 "h5::create(..., h5::swmr_write, ...) requires h5::latest_version — "
132 "SWMR needs (LATEST,LATEST) library-version bounds." );
133 const auto slash = path.find_last_of('/');
134 const std::string dir = (slash == std::string::npos) ? std::string(".")
135 : path.substr(0, slash + 1);
136 h5::impl::swmr::check_filesystem( dir );
137 const h5::libver_bounds& libver = arg::get( h5::latest_version, args... );
138 h5::fapl_t fapl = static_cast<h5::fapl_t>( libver );
139 return h5::create( path, H5F_ACC_TRUNC, h5::default_fcpl, fapl );
140 }

142 // h5::flush(ds) — writer-side SWMR metadata flush (H5Dflush). After it returns,
143 // the appended data is visible to readers that subsequently h5::refresh(ds).
144 inline void flush( const h5::ds_t& ds ) {
145 H5CPP_CHECK_NZ( H5Dflush( static_cast<::hid_t>( ds ) ),
146 h5::error::io::dataset::write,
147 "h5::flush: H5Dflush failed — is the file opened SWMR-write?" );
148 }
149
150 // h5::refresh(ds) — reader-side metadata reload (H5Drefresh) so the reader sees
151 // the writer's latest flushed state. Poll cadence is application policy.
152 inline void refresh( const h5::ds_t& ds ) {
153 H5CPP_CHECK_NZ( H5Drefresh( static_cast<::hid_t>( ds ) ),
154 h5::error::io::dataset::read,
155 "h5::refresh: H5Drefresh failed — is the file opened SWMR-read?" );
156 }
157
158 // h5::start_swmr_write(fd) — transition an open RDWR (latest-bounds) file into
159 // SWMR-write once its datasets exist. Idempotent: a no-op when the file is
160 // already SWMR-write (e.g. opened that way on a warm restart).
161 inline void start_swmr_write( const h5::fd_t& fd ) {
162 unsigned intent = 0;
163 if( H5Fget_intent( static_cast<::hid_t>( fd ), &intent ) >= 0
164 && (intent & H5F_ACC_SWMR_WRITE) )
165 return;
166 H5CPP_CHECK_NZ( H5Fstart_swmr_write( static_cast<::hid_t>( fd ) ),
167 h5::error::io::file::write,
168 "h5::start_swmr_write: H5Fstart_swmr_write failed — file must be "
169 "opened RDWR with latest_version bounds and all observed datasets "
170 "must already exist." );
171 }
172}
