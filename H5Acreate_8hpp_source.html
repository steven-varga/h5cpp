102 h5::acpl_t default_acpl{ H5Pcreate(H5P_ATTRIBUTE_CREATE) };
103 const h5::acpl_t& acpl = arg::get(default_acpl, args...);
104
105 H5CPP_CHECK_PROP( acpl, h5::error::property_list::misc, "invalid attribute create property" );
106
107 // and dimensions
108 h5::current_dims_t current_dims_default{0}; // if no current dims_present
109 // this mutable value will be referenced
110 const h5::current_dims_t& current_dims = arg::get(current_dims_default, args...);
111 // no partial IO or chunks
112 h5::sp_t space = h5::create_simple( current_dims );
113 using element_t = typename meta::decay<T>::type;
114 h5::dt_t<element_t> type;
115 ::hid_t id = H5I_UNINIT;
116 H5CPP_CHECK_NZ( (id = H5Acreate2( static_cast<::hid_t>( parent ), path.c_str(),
117 static_cast<::hid_t>(type), static_cast<::hid_t>( space ), static_cast<::hid_t>( acpl ), H5P_DEFAULT )),
118 h5::error::io::attribute::create, "couldn't create attribute");
119 return h5::at_t{id};
120 } catch( const std::runtime_error& err ) {
121 throw h5::error::io::attribute::create( err.what() );
122 }
