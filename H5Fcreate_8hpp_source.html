51 inline h5::fd_t create( const std::string& path, unsigned flags,
52 const h5::fcpl_t& fcpl=h5::default_fcpl, const h5::fapl_t& fapl=h5::default_fapl ) {
53 H5CPP_CHECK_PROP( fcpl, h5::error::io::file::create, "invalid file control property list" );
54 H5CPP_CHECK_PROP( fapl, h5::error::io::file::create, "invalid file access property list" );
55
56#if defined(H5F_ACC_SWMR_WRITE)
57 // Raw-CAPI SWMR request: the caller passed H5F_ACC_SWMR_WRITE in the flags.
58 // H5Fcreate cannot enable SWMR at create time — no datasets exist yet, so
59 // SWMR is activated later via h5::start_swmr_write(fd); passing the flag
60 // straight to H5Fcreate is the silent mode-confusion bug. So drop the
61 // open-only bits, force (LATEST,LATEST) bounds that SWMR requires, and
62 // create a normal latest-format file.
63 if( flags & H5F_ACC_SWMR_WRITE ){
64 unsigned cflags = flags & ~( H5F_ACC_SWMR_WRITE | H5F_ACC_RDWR );
65 if( !(cflags & (H5F_ACC_TRUNC | H5F_ACC_EXCL)) ) cflags |= H5F_ACC_TRUNC;
66 return h5::create( path, cflags, fcpl, static_cast<h5::fapl_t>( h5::latest_version ) );
67 }
68#endif
69
70 // MT: the file create + fileno derivation + registry attach run under the
71 // process-global HDF5 lock — under Threadsafety-OFF HDF5, only one thread
72 // may be inside the C-API at a time (its global free-lists/id-tables corrupt
73 // otherwise). on_collector takes that lock for the whole op. No-op
74 // pass-through in a classic build.
75 return h5::impl::on_collector([&]() -> h5::fd_t {
76 hid_t fd;
77 H5CPP_CHECK_NZ(
78 (fd = H5Fcreate(path.data(), flags, static_cast<hid_t>( fcpl ), static_cast<hid_t>( fapl ) )),
79 h5::error::io::file::create, h5::error::msg::create_file);
80 return fd_t{fd};
81 });
82 }
