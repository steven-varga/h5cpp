11// Sparse matrix / vector meta-traits.
12//
13// The h5cpp public sparse API lives in H5Dsparse.hpp; this header only declares
14// the trait surface that per-library mappers (H5Marma, H5Meigen, ...) specialize.
15//
16// On-disk layout is canonical Compressed Sparse Column (CSC):
17// group/
18// data : 1-D, dtype=T, non-zeros, length nnz
19// indices : 1-D, uint32, row indices, length nnz
20// indptr : 1-D, uint32, column pointers, length n_cols+1
21// shape : 1-D, uint64, length=2, [n_rows, n_cols]
22// @format = "csc"
23// @axis = "column"
24//
25// Index width is fixed uint32 on disk (10x Genomics / Loompy convention) with
26// an overflow guard at write time: if any of {nnz, n_rows, n_cols} exceeds
27// 2^32-1 the write throws. The trait writes converted uint32 buffers directly
28// into caller-supplied storage so per-library precision is preserved up to
29// the file boundary.
30//
31// Sparse vectors (Arma SpRow / SpCol, Eigen SparseVector) are promoted to
32// 1xN / Nx1 CSC matrices so files round-trip with scipy.sparse.csc_matrix
33// and 10x-style consumers unchanged.
34
35namespace h5::meta {
36
37 // Primary template: T is not a sparse linalg type.
38 // Library mappers specialize this to std::true_type.
39 template <class T, class = void>
40 struct is_sparse : std::false_type {};
41
42 template <class T>
43 inline constexpr bool is_sparse_v = is_sparse<T>::value;
44
45 // sparse_traits<Sparse>: per-library accessor contract.
46 //
47 // Specializations must provide:
48 // using value_type = T; // element scalar type
49 // static std::size_t nnz (const Sparse&);
50 // static std::size_t rows (const Sparse&);
51 // static std::size_t cols (const Sparse&);
52 // static const value_type* values(const Sparse&); // non-zero values, length nnz
53 //
54 // // fill caller-provided uint32 buffers with the CSC indices.
55 // // outer length = cols()+1, inner length = nnz().
56 // static void to_u32_outer(const Sparse&, std::uint32_t* dst);
57 // static void to_u32_inner(const Sparse&, std::uint32_t* dst);
58 //
59 // // construct from disk-decoded uint32 CSC buffers + values.
60 // static Sparse construct(
61 // std::size_t rows, std::size_t cols, std::size_t nnz,
62 // const std::uint32_t* outer, // length cols+1
63 // const std::uint32_t* inner, // length nnz
64 // const value_type* data); // length nnz
65 //
66 // Precondition for write: the source matrix must be in compressed/synced
67 // CSC form (Arma: post-sync(); Eigen: makeCompressed() && !RowMajor).
68 template <class T>
69 struct sparse_traits; // intentionally undefined; specializations live with each library mapper.
