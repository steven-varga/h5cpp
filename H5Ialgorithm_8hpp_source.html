22 // Callback for H5Literate (single level, used by h5::ls).
23 inline static herr_t iterate_callback( ::hid_t gid, const char *name, const H5L_info_t *info, void *op_data){
24 (void)gid; (void)info;
25 // this must not throw error, CAPI has to clean up
26 try {
27 std::vector<std::string> *data = static_cast<std::vector<std::string>* >(op_data);
28 data->push_back( std::string(name) );
29 } catch ( ... ){
30 return -1;
31 }
32 return 0;
33 }

35 // Callback for H5Lvisit (recursive, used by h5::dfs / h5::bfs). HDF5 hands
36 // the relative path from the visit root in `name`; we keep the path itself,
37 // not just the leaf — that's what makes the result globally addressable.
38 inline static herr_t visit_callback( ::hid_t /*gid*/, const char* name,
39 const H5L_info_t* /*info*/, void* op_data ) {
40 try {
41 auto* data = static_cast<std::vector<std::string>*>(op_data);
42 data->emplace_back(name);
43 } catch ( ... ) {
44 return -1;
45 }
46 return 0;
47 }
48}

53 // Single-level listing — immediate children of `directory` (datasets,
54 // sub-groups, and links). For recursive traversal use h5::dfs / h5::bfs.
55 inline std::vector<std::string> ls(const h5::fd_t& fd, const std::string& directory ){
56 hid_t group_id;
57 H5CPP_CHECK_NZ(
58 (group_id = H5Gopen( static_cast<hid_t>(fd), directory.c_str(), H5P_DEFAULT )), std::runtime_error, h5::error::msg::open_group );
59 std::vector<std::string> files;
60 H5CPP_CHECK_NZ( H5Literate( group_id, H5_INDEX_NAME, H5_ITER_INC, 0, &impl::iterate_callback, &files ), std::runtime_error, h5::error::msg::list_directory );
61 H5CPP_CHECK_NZ( H5Gclose(group_id), std::runtime_error, h5::error::msg::close_group);
62 return files;
63 }

65 // Recursive listing — every object path under `directory` in
66 // name-sorted depth-first order. Paths are relative to `directory`
67 // (e.g. starting from "/sensors" you get "imu", "imu/left",
68 // "imu/left/raw", "lidar", …). HDF5's H5Lvisit does the work.
69 inline std::vector<std::string> dfs(const h5::fd_t& fd, const std::string& directory) {
70 hid_t group_id;
71 H5CPP_CHECK_NZ(
72 (group_id = H5Gopen(static_cast<hid_t>(fd), directory.c_str(), H5P_DEFAULT)),
73 std::runtime_error, h5::error::msg::open_group);
74 std::vector<std::string> paths;
75 H5CPP_CHECK_NZ(
76 H5Lvisit(group_id, H5_INDEX_NAME, H5_ITER_INC, &impl::visit_callback, &paths),
77 std::runtime_error, h5::error::msg::list_directory);
78 H5CPP_CHECK_NZ(H5Gclose(group_id),
79 std::runtime_error, h5::error::msg::close_group);
80 return paths;
81 }

83 // Recursive listing in breadth-first order — all paths at depth 1 before
84 // any path at depth 2, etc. Implementation: collect via H5Lvisit (DFS),
85 // then stable-sort by slash count. Same set as dfs(), just shuffled into
86 // BFS order — useful when you want to truncate output by depth or process
87 // top-level subgroups first.
88 inline std::vector<std::string> bfs(const h5::fd_t& fd, const std::string& directory) {
89 auto paths = dfs(fd, directory);
90 std::stable_sort(paths.begin(), paths.end(),
91 [](const std::string& a, const std::string& b) {
92 return std::count(a.begin(), a.end(), '/')
93 < std::count(b.begin(), b.end(), '/');
94 });
95 return paths;
96 }

100 // True if `path` resolves to an object or link under `loc`. Cheap wrapper
101 // around H5Lexists; returns false on missing path (not throws).
102 template <class Loc>
103 inline std::enable_if_t<impl::is_valid_loc<Loc>::value, bool>
104 exists(const Loc& loc, const std::string& path) {
105 return H5Lexists(static_cast<hid_t>(loc), path.c_str(), H5P_DEFAULT) > 0;
106 }

110 // Soft link (POSIX symlink semantics): stores `target_path` as a string;
111 // resolved lazily on open. Dangling links are allowed at creation.
112 template <class Loc>
113 inline std::enable_if_t<impl::is_valid_loc<Loc>::value, void>
114 link_soft(const std::string& target_path,
115 const Loc& link_loc, const std::string& link_name,
116 const h5::lcpl_t& lcpl = h5::default_lcpl) {
117 H5CPP_CHECK_NZ(
118 H5Lcreate_soft(target_path.c_str(),
119 static_cast<hid_t>(link_loc), link_name.c_str(),
120 static_cast<hid_t>(lcpl), H5P_DEFAULT),
121 std::runtime_error, "couldn't create soft link");
122 }
