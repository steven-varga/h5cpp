30 // pt_t::pipeline selects between two pipeline implementations:
31 //
32 // basic_pipeline_t — synchronous filter chain on the calling
33 // thread; default when the file's FAPL has
34 // no h5::threads{N} pool installed.
35 // pool_pipeline_t — FAPL-scoped shared worker pool, async-
36 // pipelined dispatch with back-pressure.
37 // Selected when init() resolves a pool
38 // from the file's FAPL.
39 //
40 // Both are indirect-owned through unique_ptr so the variant
41 // remains move-assignable regardless of the underlying pipeline's
42 // move semantics.
43 using pt_pipeline_t = std::variant<
44 std::unique_ptr<impl::basic_pipeline_t>,
45 std::unique_ptr<impl::pool_pipeline_t>
46 >;
47}

60 pt_t& operator=( h5::pt_t&& pt ){
61 // prevent self assign
62 if (this == &pt) return *this;
63 if(H5Iis_valid(static_cast<hid_t>(this->ds))){ // flush and close dataset
64 this->flush();

66 }
67
68 this->ds = std::move(pt.ds);

70 this->pipeline = std::move(pt.pipeline);
71

73 this->element_size = pt.element_size;
74 this->N = pt.N; this->n = pt.n; this->rank = pt.rank;
75 this->ptr = pt.ptr; this->fill_value = pt.fill_value;
76 this->swmr_write_ = pt.swmr_write_;
77

79 pt.N=0; pt.n=0; pt.rank=0;
80 for(hsize_t i=0; i<rank; i++){
81 this->offset[i] = pt.offset[i];
82 this->current_dims[i] = pt.current_dims[i];
83 this->chunk_dims[i] = pt.chunk_dims[i];
84 this->count[i] = pt.count[i];
85 }

110 // Resolve the variant to a reference to the live pipeline_t<Derived> CRTP
111 // instance, regardless of which alternative is active. Both alternatives
112 // expose the same set of fields (chunk0, block_size, ds, dxpl, ...) inherited
113 // from pipeline_t<Derived>, so the visiting lambda can use the result
114 // duck-typed across alternatives.
115 template <typename F>
116 decltype(auto) visit_pipeline(F&& f) {
117 return std::visit([&](auto& p) -> decltype(auto) {
118 return std::forward<F>(f)(*p); // both alternatives are unique_ptr
119 }, pipeline);
120 }
121
122 impl::pt_pipeline_t pipeline;
123 h5::dxpl_t dxpl;
124 h5::ds_t ds;
125 h5::dt_t<void> dt;
126
127 hsize_t offset[H5CPP_MAX_RANK], current_dims[H5CPP_MAX_RANK],
128 chunk_dims[H5CPP_MAX_RANK], count[H5CPP_MAX_RANK];
129 size_t block_size,element_size,N,n,rank;
130 void *ptr, *fill_value;
131
132 // SWMR-write detection (issue #267). When a packet table owns its
133 // dataset the user holds no h5::ds_t to flush, so flush(pt) must issue
134 // the SWMR metadata flush itself. Resolved once in init() from the
135 // file's access intent; gates a single H5Dflush in flush().
136 bool swmr_write_{false};
137
138 // Phase 1.3.3 — chunk dispatch is uniform across all variant
139 // alternatives via visit_pipeline + write_chunk. pool_pipeline_t
140 // holds the pool reference, in-flight deque, and back-pressure
141 // logic internally; pt_t no longer needs per-instance pool fields.

174inline
175void h5::pt_t::init( const h5::ds_t& handle ){
176 try {
177 h5::impl::on_collector([&]{ // open + registry resolve + space/type probing under the global lock (MT)
178 // Re-open with zero HDF5 chunk cache: the default DAPL allocates ~1MB per H5Dopen2
179 // call which accumulates in malloc arenas when pt_t is used in loops, even though
180 // the memory is logically freed on close.
181 hid_t raw = static_cast<hid_t>(handle);
182 hid_t fid = H5Iget_file_id(raw);
183 ssize_t nlen = H5Iget_name(raw, nullptr, 0);
184 std::vector<char> dname(static_cast<size_t>(nlen) + 1);
185 H5Iget_name(raw, dname.data(), dname.size());
186 hid_t dapl = H5Pcreate(H5P_DATASET_ACCESS);
187 H5Pset_chunk_cache(dapl, 0, 0, H5D_CHUNK_CACHE_W0_DEFAULT);
188 ds = h5::ds_t{H5Dopen2(fid, dname.data(), dapl)};
189 H5Pclose(dapl);
190
191 // threads{N} lives on the dataset's DAPL — it survives the
192 // H5Dget_access_plist round-trip (unlike a FAPL property), so read it
193 // from the original handle (the re-open above used a fresh zero-cache
194 // DAPL) and fan this dataset out across the global pool.
195 {
196 hid_t orig_dapl = H5Dget_access_plist(static_cast<hid_t>(handle));
197 const unsigned n = (orig_dapl >= 0)
198 ? impl::resolve_dataset_threads(orig_dapl) : 0u;
199 if (n > 0) {
200 const unsigned cap = impl::resolve_dataset_backpressure(orig_dapl, n);
201 pipeline.emplace<std::unique_ptr<impl::pool_pipeline_t>>(
202 std::make_unique<impl::pool_pipeline_t>(impl::global_pool_ptr(), cap));
203 }
204 if (orig_dapl >= 0) H5Pclose(orig_dapl);
205 }
206
207#if defined(H5F_ACC_SWMR_WRITE) && defined(H5F_ACC_SWMR_READ)
208 // Issue #267: detect SWMR-write once so flush(pt) can make appends
209 // visible without the caller holding a separate ds_t. Non-SWMR packet
210 // tables leave the flag false and pay only a branch in flush().
211 {
212 unsigned intent = 0;
213 if( H5Fget_intent(fid, &intent) >= 0 )
214 swmr_write_ = (intent & H5F_ACC_SWMR_WRITE) != 0;
215 }
216#endif
217 H5Fclose(fid);
218 dt = h5::dt_t<void>{H5Dget_type(static_cast<hid_t>(ds))};
219 h5::sp_t file_space = h5::get_space( handle );
220 rank = h5::get_simple_extent_dims( file_space, current_dims, nullptr );
221
222 h5::dcpl_t dcpl = h5::get_dcpl( ds );
223 h5::dt_t<void*> type = h5::get_type<void*>( ds );
224 hsize_t size = h5::get_size( type );
225 this->fill_value = h5::get_fill_value(dcpl, type, size);
226 visit_pipeline([&](auto& p) {
227 p.set_cache(dcpl, size);
228 this->ptr = p.chunk0;
229 this->block_size = p.block_size;
230 this->element_size = p.element_size;
231 this->N = p.n;
232 p.ds = ds; p.dxpl = dxpl;
233 });
234 h5::get_chunk_dims( dcpl, chunk_dims );
235 // NB: `this->current_dims` is REQUIRED — unqualified `current_dims[ax]` at a
236 // statement start is parsed by MSVC as a declaration ('current_dims' resolves
237 // to the h5::current_dims property-tag TYPE, not the member), giving
238 // C2371/C3694. The explicit member access forces an expression. #287.
239 for(hsize_t ax = 1; ax < rank; ++ax)
240 this->current_dims[ax] = chunk_dims[ax];
241 }); // on_collector
242 } catch ( ... ){
243 throw h5::error::io::packet_table::misc( H5CPP_ERROR_MSG("CTOR: unable to create handle from dataset..."));
244 }
245}

259inline void h5::pt_t::append( const std::string& ref ) {
260 static_cast<const char**>( ptr )[n++] = ref.data();
261 if( n != N ) return;
262
263 *offset = *current_dims;
264 *current_dims += *chunk_dims;
265 h5::impl::on_collector([&]{ // serialize the variable-length flush under the global lock (MT)
266 h5::set_extent(ds, current_dims);
267
268 hsize_t block = 1, count = n;
269 h5::sp_t mem_space{H5Screate_simple(static_cast<int>(rank), &count, nullptr )};
270 h5::sp_t file_space{H5Dget_space( static_cast<::hid_t>(ds) )};
271 h5::select_all( mem_space );
272 H5Sselect_hyperslab( static_cast<hid_t>(file_space), H5S_SELECT_SET, offset, nullptr, &block, &count);
273
274 H5Dwrite( static_cast<hid_t>( ds ),
275 dt, static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), ptr);
276 });
277 n = 0;
278}

319template<class T> inline std::enable_if_t< !h5::meta::is_scalar<T>::value,
320void> h5::pt_t::append( const T& ref ) try {
321 // Iterator-only containers (list, forward_list, deque, set, …) have no .data().
322 // Delegate element-by-element to the scalar append overload.
323 using access = h5::meta::access_traits_t<T>;
324 if constexpr (access::kind == h5::meta::access_t::iterators) {
325 for (const auto& elem : ref)
326 this->append(elem);
327 } else {
328
329 auto dims = meta::size( ref );
330
331 *offset = *current_dims;
332 *current_dims += 1;
333 auto ptr_ = meta::data( ref );
334 auto dims_ = meta::size( ref );
335
336 h5::impl::on_collector([&]{ // serialize the HDF5 chunk flush under the global lock (MT)
337 h5::set_extent(ds, current_dims);
338 switch( dims_.size() ){
339 case 1: // vector
340 if( dims[0] * element_size == block_size )
341 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, (void*) ptr_ ); });
342 else throw h5::error::io::packet_table::write(
343 H5CPP_ERROR_MSG("dimension mismatch: "
344 + std::to_string( dims[0] * element_size) + " != " + std::to_string(block_size) ));
345 break;
346 case 2: //matrix
347 if( dims[0] * dims[1] * element_size == block_size )
348 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, (void*) ptr_ ); });
349 else throw h5::error::io::packet_table::write(
350 H5CPP_ERROR_MSG("dimension mismatch: "
351 + std::to_string( dims[0] * dims[1] * element_size) + " != " + std::to_string(block_size) ));
352 break;
353 case 3: // cube
354 if( dims[0] * dims[1] * dims[2] * element_size == block_size )
355 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, (void*) ptr_ ); });
356 else throw h5::error::io::packet_table::write(
357 H5CPP_ERROR_MSG("dimension mismatch: "
358 + std::to_string( dims[0] * dims[1] * dims[2] * element_size) + " != " + std::to_string(block_size) ));
359 ;break;
360 default:
361 throw h5::error::io::packet_table::misc( H5CPP_ERROR_MSG("objects with rank > 2 are not supported... "));
362 }
363 }); // on_collector
364 } // end else (non-iterator path)
365} catch( const std::runtime_error& err ){
366 throw h5::error::io::dataset::append( err.what() );
367}

369inline
370void h5::pt_t::flush(){
371 // Whole flush — the trailing partial-chunk write AND the pool drain (which
372 // issues H5Dwrite_chunk for in-flight compressed chunks) — runs under the
373 // global HDF5 lock when built MT; no-op pass-through in a classic build.
374 h5::impl::on_collector([&]{
375 if( n != 0 ) {
376 *offset = *current_dims;
377 *current_dims += *chunk_dims;
378 h5::set_extent(ds, current_dims);
379
380 if( H5Tis_variable_str(this->dt)) {
381 hsize_t block = 1, count = n;
382 h5::sp_t mem_space{H5Screate_simple(static_cast<int>(rank), &count, nullptr )};
383 h5::sp_t file_space{H5Dget_space( static_cast<::hid_t>(ds) )};
384 h5::select_all( mem_space );
385 H5Sselect_hyperslab( static_cast<hid_t>(file_space), H5S_SELECT_SET, offset, nullptr, &block, &count);
386
387 H5Dwrite( static_cast<hid_t>( ds ),
388 dt, static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), ptr);
389 } else {
390 // the remainder of last chunk must be set to fill_value; arbitrary type size supported
391 for(hsize_t i=0; i<(N-n); i++)
392 for(size_t j=0; j < element_size; j++)
393 static_cast<char*>( ptr )[(n + i) * element_size + j] = static_cast<char*>( fill_value )[ j ];
394 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, ptr); });
395 }
396 n = 0;
397 }
398 // Pool path: drain in-flight chunks so flush() honors the "data
399 // on disk after this returns" contract. basic_pipeline_t writes
400 // inline; the visit is a no-op for that alternative.
401 std::visit([](auto& p) {
402 using T = std::decay_t<decltype(*p)>;
403 if constexpr (std::is_same_v<T, impl::pool_pipeline_t>)
404 p->drain();
405 }, pipeline);
406
407#if defined(H5F_ACC_SWMR_WRITE) && defined(H5F_ACC_SWMR_READ)
408 // Issue #267: when the file is open SWMR-write, make every append since the
409 // last flush visible to readers now. Deliberately outside the n!=0 guard —
410 // append() auto-writes full chunks, which still need the metadata flush.
411 // flush(pt) is thus the complete writer-side SWMR call; no separate
412 // h5::flush(ds) is required (and the packet-table user has no ds_t anyway).
413 //
414 // swmr_write_ latches ON once true: in the create-then-transition idiom the
415 // packet table is constructed BEFORE h5::start_swmr_write(fd), so init() saw
416 // a non-SWMR file. We re-probe intent each flush WHILE still false (SWMR can
417 // be turned on between flushes), and stop the moment it latches true. flush()
418 // is not the per-element hot path (append() is), and H5Fget_intent does no
419 // I/O, so a non-SWMR packet table pays only a cheap intent probe per flush.
420 //
421 // Runs inside on_collector's lock scope (with the drain above): H5Iget_file_id/
422 // H5Fget_intent/H5Dflush are CAPI calls that must hold the global HDF5 lock too.
423 if( !swmr_write_ && h5::is_valid(ds) ){
424 hid_t fid = H5Iget_file_id(static_cast<hid_t>(ds));
425 if( fid >= 0 ){
426 unsigned intent = 0;
427 if( H5Fget_intent(fid, &intent) >= 0 )
428 swmr_write_ = (intent & H5F_ACC_SWMR_WRITE) != 0;
429 H5Fclose(fid);
430 }
431 }
432 if( swmr_write_ && h5::is_valid(ds) )
433 H5Dflush(static_cast<hid_t>(ds));
434#endif
435 });
436}
