28 // pt_t::pipeline selects between two pipeline implementations:
29 //
30 // basic_pipeline_t — synchronous filter chain on the calling
31 // thread; default when the file's FAPL has
32 // no h5::threads{N} pool installed.
33 // pool_pipeline_t — FAPL-scoped shared worker pool, async-
34 // pipelined dispatch with back-pressure.
35 // Selected when init() resolves a pool
36 // from the file's FAPL.
37 //
38 // Both are indirect-owned through unique_ptr so the variant
39 // remains move-assignable regardless of the underlying pipeline's
40 // move semantics.
41 using pt_pipeline_t = std::variant<
42 std::unique_ptr<impl::basic_pipeline_t>,
43 std::unique_ptr<impl::pool_pipeline_t>
44 >;
45}

59 // prevent self assign
60 if (this == &pt) return *this;
61 if(H5Iis_valid(this->ds)){ // flush and close dataset
62 this->flush();
63 free(this->fill_value);
64 }
65
66 this->ds = std::move(pt.ds);
67 this->dxpl = std::move(pt.dxpl);
68 this->pipeline = std::move(pt.pipeline);
69
70 this->block_size = pt.block_size;
71 this->element_size = pt.element_size;
72 this->N = pt.N; this->n = pt.n; this->rank = pt.rank;
73 this->ptr = pt.ptr; this->fill_value = pt.fill_value;
74
75 pt.ptr = nullptr; pt.fill_value = nullptr;
76 pt.N=0; pt.n=0; pt.rank=0;
77 for(hsize_t i=0; i<rank; i++){
78 this->offset[i] = pt.offset[i];
79 this->current_dims[i] = pt.current_dims[i];
80 this->chunk_dims[i] = pt.chunk_dims[i];
81 this->count[i] = pt.count[i];
82 }
83 return *this;

106 // Resolve the variant to a reference to the live pipeline_t<Derived> CRTP
107 // instance, regardless of which alternative is active. Both alternatives
108 // expose the same set of fields (chunk0, block_size, ds, dxpl, ...) inherited
109 // from pipeline_t<Derived>, so the visiting lambda can use the result
110 // duck-typed across alternatives.
111 template <typename F>
112 decltype(auto) visit_pipeline(F&& f) {
113 return std::visit([&](auto& p) -> decltype(auto) {
114 return std::forward<F>(f)(*p); // both alternatives are unique_ptr
115 }, pipeline);
116 }
117
118 impl::pt_pipeline_t pipeline;
119 h5::dxpl_t dxpl;
120 h5::ds_t ds;
121 h5::dt_t<void> dt;
122
123 hsize_t offset[H5CPP_MAX_RANK], current_dims[H5CPP_MAX_RANK],
124 chunk_dims[H5CPP_MAX_RANK], count[H5CPP_MAX_RANK];
125 size_t block_size,element_size,N,n,rank;
126 void *ptr, *fill_value;
127
128 // Phase 1.3.3 — chunk dispatch is uniform across all variant
129 // alternatives via visit_pipeline + write_chunk. pool_pipeline_t
130 // holds the pool reference, in-flight deque, and back-pressure
131 // logic internally; pt_t no longer needs per-instance pool fields.
132 };

164inline
165void h5::pt_t::init( const h5::ds_t& handle ){
166 try {
167 // Re-open with zero HDF5 chunk cache: the default DAPL allocates ~1MB per H5Dopen2
168 // call which accumulates in malloc arenas when pt_t is used in loops, even though
169 // the memory is logically freed on close.
170 hid_t raw = static_cast<hid_t>(handle);
171 hid_t fid = H5Iget_file_id(raw);
172 ssize_t nlen = H5Iget_name(raw, nullptr, 0);
173 std::vector<char> dname(static_cast<size_t>(nlen) + 1);
174 H5Iget_name(raw, dname.data(), dname.size());
175 hid_t dapl = H5Pcreate(H5P_DATASET_ACCESS);
176 H5Pset_chunk_cache(dapl, 0, 0, H5D_CHUNK_CACHE_W0_DEFAULT);
177 ds = h5::ds_t{H5Dopen2(fid, dname.data(), dapl)};
178 H5Pclose(dapl);
179
180 // Phase 1.3.3 — resolve the file's FAPL pool while we still hold
181 // a live fid. When the FAPL has h5::threads{N} installed, swap
182 // the variant from basic_pipeline_t (default) to pool_pipeline_t
183 // constructed with the pool + back-pressure cap. When no pool
184 // is present, the default basic_pipeline_t stays — synchronous
185 // behavior, identical to pre-Phase-I.
186 hid_t fapl = H5Fget_access_plist(fid);
187 if (auto pool = impl::resolve_worker_pool(fapl)) {
188 const unsigned cap = impl::resolve_backpressure(fapl, pool->worker_count());
189 pipeline.emplace<std::unique_ptr<impl::pool_pipeline_t>>(
190 std::make_unique<impl::pool_pipeline_t>(std::move(pool), cap));
191 }
192 H5Pclose(fapl);
193
194 H5Fclose(fid);
195 dt = h5::dt_t<void>{H5Dget_type(static_cast<hid_t>(ds))};
196 h5::sp_t file_space = h5::get_space( handle );
197 rank = h5::get_simple_extent_dims( file_space, current_dims, nullptr );
198
199 h5::dcpl_t dcpl = h5::get_dcpl( ds );
200 h5::dt_t<void*> type = h5::get_type<void*>( ds );
201 hsize_t size = h5::get_size( type );
202 this->fill_value = h5::get_fill_value(dcpl, type, size);
203 visit_pipeline([&](auto& p) {
204 p.set_cache(dcpl, size);
205 this->ptr = p.chunk0;
206 this->block_size = p.block_size;
207 this->element_size = p.element_size;
208 this->N = p.n;
209 p.ds = ds; p.dxpl = dxpl;
210 });
211 h5::get_chunk_dims( dcpl, chunk_dims );
212 for(hsize_t i=1; i<rank; i++)
213 current_dims[i] = chunk_dims[i];
214 } catch ( ... ){
215 throw h5::error::io::packet_table::misc( H5CPP_ERROR_MSG("CTOR: unable to create handle from dataset..."));
216 }
217}

229inline void h5::pt_t::append( const std::string& ref ) {
230 static_cast<const char**>( ptr )[n++] = ref.data();
231 if( n != N ) return;
232
233 *offset = *current_dims;
234 *current_dims += *chunk_dims;
235 h5::set_extent(ds, current_dims);
236
237 hsize_t block = 1, count = n;
238 h5::sp_t mem_space{H5Screate_simple(static_cast<int>(rank), &count, nullptr )};
239 h5::sp_t file_space{H5Dget_space( static_cast<::hid_t>(ds) )};
240 h5::select_all( mem_space );
241 H5Sselect_hyperslab( static_cast<hid_t>(file_space), H5S_SELECT_SET, offset, nullptr, &block, &count);
242
243 H5Dwrite( static_cast<hid_t>( ds ),
244 dt, mem_space, file_space, static_cast<hid_t>(dxpl), ptr);
245 n = 0;
246}

283template<class T> inline std::enable_if_t< !h5::meta::is_scalar<T>::value,
284void> h5::pt_t::append( const T& ref ) try {
285 // Iterator-only containers (list, forward_list, deque, set, …) have no .data().
286 // Delegate element-by-element to the scalar append overload.
287 using access = h5::meta::access_traits_t<T>;
288 if constexpr (access::kind == h5::meta::access_t::iterators) {
289 for (const auto& elem : ref)
290 this->append(elem);
291 } else {
292
293 auto dims = meta::size( ref );
294
295 *offset = *current_dims;
296 *current_dims += 1;
297 h5::set_extent(ds, current_dims);
298 auto ptr_ = meta::data( ref );
299 auto dims_ = meta::size( ref );
300
301 switch( dims_.size() ){
302 case 1: // vector
303 if( dims[0] * element_size == block_size )
304 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, (void*) ptr_ ); });
305 else throw h5::error::io::packet_table::write(
306 H5CPP_ERROR_MSG("dimension mismatch: "
307 + std::to_string( dims[0] * element_size) + " != " + std::to_string(block_size) ));
308 break;
309 case 2: //matrix
310 if( dims[0] * dims[1] * element_size == block_size )
311 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, (void*) ptr_ ); });
312 else throw h5::error::io::packet_table::write(
313 H5CPP_ERROR_MSG("dimension mismatch: "
314 + std::to_string( dims[0] * dims[1] * element_size) + " != " + std::to_string(block_size) ));
315 break;
316 case 3: // cube
317 if( dims[0] * dims[1] * dims[2] * element_size == block_size )
318 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, (void*) ptr_ ); });
319 else throw h5::error::io::packet_table::write(
320 H5CPP_ERROR_MSG("dimension mismatch: "
321 + std::to_string( dims[0] * dims[1] * dims[2] * element_size) + " != " + std::to_string(block_size) ));
322 ;break;
323 default:
324 throw h5::error::io::packet_table::misc( H5CPP_ERROR_MSG("objects with rank > 2 are not supported... "));
325 }
326 } // end else (non-iterator path)
327} catch( const std::runtime_error& err ){
328 throw h5::error::io::dataset::append( err.what() );
329}

331inline
332void h5::pt_t::flush(){
333 if( n != 0 ) {
334 *offset = *current_dims;
335 *current_dims += *chunk_dims;
336 h5::set_extent(ds, current_dims);
337
338 if( H5Tis_variable_str(this->dt)) {
339 hsize_t block = 1, count = n;
340 h5::sp_t mem_space{H5Screate_simple(static_cast<int>(rank), &count, nullptr )};
341 h5::sp_t file_space{H5Dget_space( static_cast<::hid_t>(ds) )};
342 h5::select_all( mem_space );
343 H5Sselect_hyperslab( static_cast<hid_t>(file_space), H5S_SELECT_SET, offset, nullptr, &block, &count);
344
345 H5Dwrite( static_cast<hid_t>( ds ),
346 dt, mem_space, file_space, static_cast<hid_t>(dxpl), ptr);
347 } else {
348 // the remainder of last chunk must be set to fill_value; arbitrary type size supported
349 for(hsize_t i=0; i<(N-n); i++)
350 for(size_t j=0; j < element_size; j++)
351 static_cast<char*>( ptr )[(n + i) * element_size + j] = static_cast<char*>( fill_value )[ j ];
352 visit_pipeline([&](auto& p){ p.write_chunk(offset, block_size, ptr); });
353 }
354 n = 0;
355 }
356 // Pool path: drain in-flight chunks so flush() honors the "data
357 // on disk after this returns" contract. basic_pipeline_t writes
358 // inline; the visit is a no-op for that alternative.
359 std::visit([](auto& p) {
360 using T = std::decay_t<decltype(*p)>;
361 if constexpr (std::is_same_v<T, impl::pool_pipeline_t>)
362 p->drain();
363 }, pipeline);
364}

417inline std::ostream& operator<<(std::ostream &os, const h5::pt_t& pt) {
418 os << std::dec;
419 os <<"packet table:\n"
420 "------------------------------------------\n";
421 if( !H5Iis_valid(pt.ds)) {
422 os << "ds: H5I_UNINIT" <<std::endl;
423 return os;
424 }
425 std::vector<size_t> current_dims(pt.current_dims, pt.current_dims + pt.rank);
426 std::vector<size_t> chunk_dims(pt.chunk_dims, pt.chunk_dims + pt.rank);
427 std::vector<size_t> count(pt.count, pt.count + pt.rank);
428 std::vector<size_t> offset(pt.offset, pt.offset + pt.rank);
429
430 os << "rank: " << pt.rank << " N:" << pt.N <<" n:" << pt.n << "\n";
431 os << "element size: " << pt.element_size << " block size: " << pt.block_size << "\n";
432 os << "current dims: " << current_dims << std::endl;
433 os << "chunk dims: " << chunk_dims << std::endl;
434 os << "offset : " << offset << " count : " << count << std::endl;
435 os << "ds: "<< static_cast<hid_t>( pt.ds ) <<" dxpl: "<< static_cast<hid_t>( pt.dxpl ) << std::endl;
436 os << "fill value: " << std::hex << pt.fill_value << " buffer: " << pt.ptr;
437 os << "\n\n";
438 os << std::dec;
439 return os;
