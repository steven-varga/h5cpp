71 template <class T, class... args_t>
72 inline h5::ds_t write( const h5::ds_t& ds, const T* ptr, args_t&&... args ) try {
73 // element types: pod | [signed|unsigned](int8 | int16 | int32 | int64) | float | double
74 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
75 using toffset = typename arg::tpos<const h5::offset_t&, const args_t&...>;
76 using tstride = typename arg::tpos<const h5::stride_t&, const args_t&...>;
77 using tblock = typename arg::tpos<const h5::block_t&, const args_t&...>;
78 static_assert( tcount::present,"h5::count_t{ ... } must be provided to describe T* memory region" );
79 //static_assert( utils::is_supported<T>, "error: " H5CPP_supported_elementary_types );
80
81 auto tuple = std::forward_as_tuple(args...);
82 h5::count_t count = std::get<tcount::value>( tuple ); //we make a copy of it
83 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...); // gets reference to default value if property not present
84 h5::sp_t file_space{H5Dget_space( static_cast<::hid_t>(ds) )};
85
86 int rank = h5::get_simple_extent_ndims( file_space );
87 hsize_t n_elements = static_cast<hsize_t>(count);
88 hid_t dapl = h5::get_access_plist( ds );
89 herr_t err = 0;
90
91 // The high_throughput pipeline uses H5Dwrite_chunk, which requires a
92 // chunked dataset layout. Guard on H5D_CHUNKED so a DAPL with the flag
93 // applied to a contiguous/compact dataset falls through to standard
94 // H5Dwrite rather than triggering "not a chunked dataset" errors
95 // (which manifest as a SegFault on Windows MSVC, see #242 follow-up).
96 const bool use_pipeline = [&]() {
97 if (!H5Pexist(dapl, H5CPP_DAPL_HIGH_THROUGHPUT)) return false;
98 hid_t dcpl_id = H5Dget_create_plist(static_cast<hid_t>(ds));
99 if (dcpl_id < 0) return false;
100 H5D_layout_t layout = H5Pget_layout(dcpl_id);
101 H5Pclose(dcpl_id);
102 return layout == H5D_CHUNKED;
103 }();
104 if( use_pipeline ){
105 const h5::block_t& block = arg::get( h5::default_block, args...);
106 const h5::offset_t& offset = arg::get( h5::default_offset, args...);
107 const h5::stride_t& stride = arg::get( h5::default_stride, args...);
108
109 // Phase 1.3.3 — if the file's FAPL has h5::threads{N}, route
110 // compress work through the shared pool via a local
111 // pool_pipeline_t. Otherwise use the existing DAPL-stored
112 // basic_pipeline_t pointer for synchronous filter chain.
113 hid_t fid = H5Iget_file_id(static_cast<hid_t>(ds));
114 hid_t fapl = H5Fget_access_plist(fid);
115 auto pool = h5::impl::resolve_worker_pool(fapl);
116 if (pool) {
117 const unsigned cap = h5::impl::resolve_backpressure(
118 fapl, pool->worker_count());
119 h5::impl::pool_pipeline_t pipe(std::move(pool), cap);
120 // set_cache populates the filter chain from the dataset's DCPL.
121 h5::dcpl_t dcpl{H5Dget_create_plist(static_cast<hid_t>(ds))};
122 hid_t type_id = H5Dget_type(static_cast<hid_t>(ds));
123 size_t elem_sz = H5Tget_size(type_id);
124 H5Tclose(type_id);
125 pipe.set_cache(dcpl, elem_sz);
126 pipe.write(ds, offset, stride, block, count, dxpl, ptr);
127 // pipe destructor drains in_flight before pool refcount drop.
128 } else {
129 h5::impl::pipeline_t<impl::basic_pipeline_t>* filters;
130 H5Pget(dapl, H5CPP_DAPL_HIGH_THROUGHPUT, &filters);
131 filters->write(ds, offset, stride, block, count, dxpl, ptr);
132 }
133 H5Pclose(fapl);
134 H5Fclose(fid);
135 } else {
136 h5::sp_t mem_space = h5::create_simple( n_elements );
137 h5::select_all( mem_space );
138 // we want to optimize for best hyper block selection, in order to do that
139 // let's find out what level of control is needed
140 if constexpr (toffset::present || tstride::present || tblock::present){
141 // HYPERBLOCK selection: we either have the argument in `args...` or using default values
142 const h5::block_t& block = arg::get( h5::default_block, args...);
143 const h5::offset_t& offset = arg::get( h5::default_offset, args...);
144 const h5::stride_t& stride = arg::get( h5::default_stride, args...);
145 if constexpr( tblock::present ){ // we have to normalise `count` such that `size[i] = count[i] * block[i]` holds
146 for(int i=0; i < rank; i++) count[i] /= block[i];
147 err = H5Sselect_hyperslab(file_space, H5S_SELECT_SET, *offset, *stride, *count, *block);
148 } else { // we have to convert h5::count_t{..} to h5::block{..} and initiate a single block transfer
149 h5::block_t block_ = static_cast<h5::block_t>(count);
150 block_.rank = rank; // memory space may have different rank, be sure to use the rank of file_space
151 err = H5Sselect_hyperslab(file_space, H5S_SELECT_SET, *offset, *stride, *h5::default_count, *block_);
152 }
153 } else // SELECT_ALL this is the fastest approach, mem_space and file_space must match
154 err = H5Sselect_all(file_space);
155 // throw an exception if eny error
156 H5CPP_CHECK_NZ(err, h5::error::io::dataset::write, h5::error::msg::select_hyperslab);
157 // MSVC partial-ordering bug: the unqualified `::h5::write(ds, mem, file, dxpl, ptr)`
158 // is ambiguous between the inner overload at line 21 and the variadic forwarder at
159 // line 181 (with T=sp_t). Inline the H5Dwrite call here — matches the H5Dread.hpp:73
160 // idiom — so there is no recursive call back into this overload set.
161 using element_t = typename impl::decay<T>::type;
162 h5::meta::resolved_type_t<element_t> type;
163 H5CPP_CHECK_NZ(
164 H5Dwrite( static_cast<hid_t>(ds), static_cast<hid_t>(type),
165 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
166 static_cast<hid_t>(dxpl), ptr),
167 h5::error::io::dataset::write, h5::error::msg::write_dataset);
168 }
169 return ds;
170 } catch ( const std::exception& err ){
171 throw h5::error::io::dataset::write( err.what() );
172 }

228 template <class T, class... args_t,
229 class = std::enable_if_t<!std::is_pointer_v<std::decay_t<T>>>>
230 inline h5::ds_t write(const h5::ds_t& ds, const T& ref, args_t&&... args) try {
231 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
232 using element_t = typename impl::decay<T>::type;
233
234 using traits = h5::meta::access_traits_t<T>;
235 constexpr auto kind = traits::kind;
236
237 if constexpr (kind == h5::meta::access_t::contiguous
238 || kind == h5::meta::access_t::object
239 || kind == h5::meta::access_t::text) {
240 auto ptr = traits::data(ref);
241 if constexpr (tcount::present)
242 ::h5::write(ds, ptr, args...);
243 else {
244 h5::count_t count = traits::size( ref );
245 ::h5::write(ds, ptr, count, args...);
246 }
247 } else if constexpr (kind == h5::meta::access_t::pointers) {
248 // e.g. vector<string>, vector<vector<int>> — elements have .data() but are not flat
249 using element_t = typename impl::decay<typename traits::element_t>::type;
250 std::vector<element_t> elements;
251 const element_t* ptrs = h5::gather(ref, elements);
252 if constexpr (tcount::present)
253 ::h5::write<element_t>(ds, ptrs, args...);
254 else {
255 h5::count_t count = traits::size( ref );
256 ::h5::write<element_t>(ds, ptrs, count, args...);
257 }
258 } else if constexpr (kind == h5::meta::access_t::iterators) {
259 // e.g. list<int>, set<int>, map<K,V> — no direct pointer, copy to staging buffer
260 using element_t = typename impl::decay<typename traits::element_t>::type;
261 auto count = traits::size(ref);
262 size_t n = 1;
263 for (std::size_t i = 0; i < count.size(); ++i) n *= count[i];
264 std::vector<element_t> buffer;
265 buffer.reserve(n);
266 for (const auto& elem : ref)
267 buffer.push_back(elem);
268 ::h5::write(ds, buffer.data(), h5::count_t(count), args...);
269 } else {
270 static_assert(kind != h5::meta::access_t::unsupported,
271 "unsupported type for h5::write");
272 }
273 return ds;
274 } catch ( const std::exception& err ){
275 throw h5::error::io::dataset::write( err.what() );
276 }

324 template <class T, class... args_t>
325 inline h5::ds_t write( const h5::fd_t& fd, const std::string& dataset_path, const T* ptr, args_t&&... args ){
326 using tcount = typename arg::tpos<const h5::count_t&, const args_t&...>;
327 static_assert( tcount::present, "h5::count_t{ ... } must be provided to describe T* memory region" );
328 h5::ds_t ds; // initialized to H5I_UNINIT
329
330 h5::mute(); // find out if we have to create the dataset
331 bool is_dataset_present = H5Lexists(fd, dataset_path.c_str(), H5P_DEFAULT) > 0;
332 h5::unmute(); // <- make sure not to mute error handling longer than needed
333 if (is_dataset_present) {
334 const h5::dapl_t& dapl = arg::get(h5::default_dapl, args...);
335 ds = h5::open(fd, dataset_path, dapl);
336 } else {
337 // dataset doesn't exist, or some error happened, since h5::create doesn't know of the
338 // memory space size as `T& ref` never passed along we have to compute the `h5::current_dims_t{}` upfront
339 using tcurrent_dims = typename arg::tpos<const h5::current_dims_t&, const args_t&...>;
340 if constexpr (tcurrent_dims::present) // user knows what he is doing, specified h5::current_dims{} explicitly
341 ds = h5::create<T>(fd, dataset_path, args...);
342 else {
343 h5::current_dims_t current_dims = static_cast<h5::current_dims_t>(std::get<tcount::value>(std::forward_as_tuple(args...))); // count must be present, we checked
344 ds = h5::create<T>(fd, dataset_path, current_dims, args...); // and use it to create dataset
345 }
346 }
347 // we either have `ds` != H5I_UNINIT or an exception thrown, safe to delegate
348 return ::h5::write(ds, ptr, args...);
349 }

399 inline h5::ds_t write( const h5::fd_t& fd, const std::string& dataset_path, const T& ref, args_t&&... args ){
400 if constexpr (h5::has_scatter<std::decay_t<T>>::value) {
401 // Scatter path: compiler-generated scatter<T> handles open/create + row append.
402 // Call-site properties (chunk, compress, etc.) are ignored here; the generated
403 // specialization embeds them or the dataset was pre-created.
404 return h5::scatter<std::decay_t<T>>(fd, dataset_path, ref);
405 } else {
406 h5::ds_t ds; // initialized to H5I_UNINIT
407 // find out if we have to create the dataset
408 h5::mute();
409 // Returns a negative value when the function fails and may return a negative value if the link does not exist.
410 // - name is not local to the group specified by loc_id or, if loc_id is something other than a group identifier,
411 // name is not local to the root group
412 // - Any element of the relative path or absolute path in name, except the target link, does not exist.
413 bool is_dataset_present = H5Lexists(fd, dataset_path.c_str(), H5P_DEFAULT) > 0;
414 h5::unmute(); // <- make sure not to mute error handling longer than needed
415
416 if (is_dataset_present) {
417 const h5::dapl_t& dapl = arg::get(h5::default_dapl, args...);
418 ds = h5::open(fd, dataset_path, dapl);
419 } else {
420 // dataset doesn't exist, or some error happened, since h5::create doesn't know of the
421 // memory space size as `T& ref` never passed along we have to compute the `h5::current_dims_t{}` upfront
422 using tcurrent_dims = typename arg::tpos<const h5::current_dims_t&, const args_t&...>;
423 using element_t = typename h5::impl::decay<T>::type;
424 if constexpr (tcurrent_dims::present) // user knows what he is doing, specified h5::current_dims{} explicitly
425 ds = h5::create<element_t>(fd, dataset_path, args...);
426 else { // h5::current_dims{..} is explicitly given by `h5::count` and optional h5::offset{}, h5::stride{}, h5::block{}
427 h5::count_t count = impl::size(ref);
428 h5::current_dims_t current_dims = h5::impl::get_current_dims(count, args...); // get correct dimensions
429 ds = h5::create<element_t>(fd, dataset_path, current_dims, args...); // and use it to create dataset
430 }
431 }
432 // we either have `ds` != H5I_UNINIT or an exception thrown, safe to delegate
433 return ::h5::write(ds, ref, args...);
434 }
