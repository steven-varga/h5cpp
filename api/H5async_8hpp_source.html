7// Phase II user-facing factories — h5::async::create / h5::async::open.
8// These are the only call sites where the word "async" appears in user
9// code; every downstream operation (h5::write, h5::read, etc.) deduces
10// async-ness from the FD type via TAD and dispatches through the
11// executor with `if constexpr (is_async_v<FD>)`.
12//
13// Mechanism:
14// 1. Resolve / install a worker_pool_t on a local FAPL clone so the
15// executor has a compression backend (Phase II PR-B will wire that
16// connection through h5::write / h5::read).
17// 2. Construct the executor with that pool.
18// 3. Call H5Fcreate / H5Fopen on the user's FAPL (unchanged).
19// 4. Wrap the resulting ::hid_t plus the executor shared_ptr in the
20// async fd_t. The executor lives as long as any descriptor
21// derived from this fd holds the shared_ptr.
22//
23// We deliberately don't rely on H5Pinsert2 + H5Fget_access_plist for
24// the executor handoff — HDF5 1.10.9 strips user properties from the
25// FAPL retrieved off a file id, which would make the executor
26// unreachable on the read path. Storing it on the wrapper sidesteps
27// that limitation.
28

30#include "H5Pthreads.hpp" // worker_pool_t + h5::threads + resolve_worker_pool
31#include "H5executor.hpp" // executor_t — complete type required at make_shared

41// Build the executor that backs an async fd. Reuses an h5::threads{N}
42// pool from the user's FAPL chain when present, otherwise stands up a
43// default-sized pool on a freshly-created FAPL (we don't try to
44// H5Pcopy the user's FAPL because H5P_DEFAULT — the common case — is
45// not a copyable property list).
46inline std::shared_ptr<h5::impl::executor_t>
47make_executor_for(const h5::fapl_t& fapl) {
48 auto pool = h5::impl::resolve_worker_pool(static_cast<::hid_t>(fapl));
49 if (!pool) {
50 h5::fapl_t scratch{H5Pcreate(H5P_FILE_ACCESS)};
51 h5::impl::fapl_threads_set(static_cast<::hid_t>(scratch), 0);
52 pool = h5::impl::resolve_worker_pool(static_cast<::hid_t>(scratch));
53 }
54 return std::make_shared<h5::impl::executor_t>(std::move(pool));
55}
56
57} // namespace impl_detail
58
59inline h5::async::fd_t create(const std::string& path, unsigned flags,
60 const h5::fcpl_t& fcpl = h5::default_fcpl,
61 const h5::fapl_t& fapl = h5::default_fapl) {
62 H5CPP_CHECK_PROP(fcpl, h5::error::io::file::create, "invalid file control property list");
63 H5CPP_CHECK_PROP(fapl, h5::error::io::file::create, "invalid file access property list");
64
65 auto exec = impl_detail::make_executor_for(fapl);
66
67 hid_t fd;
68 H5CPP_CHECK_NZ(
69 (fd = H5Fcreate(path.data(), flags,
70 static_cast<::hid_t>(fcpl),
71 static_cast<::hid_t>(fapl))),
72 h5::error::io::file::create, h5::error::msg::create_file);
73 return h5::async::fd_t{fd, std::move(exec)};
74}

76inline h5::async::fd_t open(const std::string& path, unsigned flags,
77 const h5::fapl_t& fapl = h5::default_fapl) {
78 H5CPP_CHECK_PROP(fapl, h5::error::io::file::open, "invalid file access property list");
79
80 auto exec = impl_detail::make_executor_for(fapl);
81
82 hid_t fd;
83 H5CPP_CHECK_NZ(
84 (fd = H5Fopen(path.data(), flags, static_cast<::hid_t>(fapl))),
85 h5::error::io::file::open, h5::error::msg::open_file);
86 return h5::async::fd_t{fd, std::move(exec)};
87}
