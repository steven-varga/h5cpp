13 // --- low-level attribute writers (leaf operations used by simple branches) ---
14 // Generic template — handles all element pointer types. Used by both the
15 // initializer_list overload and the iter-staging / object / contiguous
16 // branches of the high-level awrite below.
17 template <class T>
18 inline void awrite( const h5::at_t& attr, const T* ptr ){
19 using element_t = typename meta::decay<T>::type;
20 h5::dt_t<element_t> type;
21 H5CPP_CHECK_NZ( H5Awrite( static_cast<hid_t>(attr), static_cast<hid_t>( type ), ptr ),
22 h5::error::io::attribute::write, "couldn't write attribute.");
23 }
24 // Note: a previously-present non-template `awrite(at_t, const char*)` overload
25 // was removed because it caused overload-resolution conflicts with the template
26 // above when called via explicit template syntax `awrite<char>(attr, const char*)`
27 // (e.g., from the initializer_list overload writing `{'a','b','c'}`). Per C++
28 // overload rules, the non-template wins over an equally-matching template, so
29 // the vlen-string special path was hijacking ordinary `NATIVE_CHAR` writes and
30 // causing HDF5 type-conversion failures. Writing a scalar vlen string to an
31 // attribute is now done by the high-level `kind == text` branch directly.
32
33 namespace detail {
34 // Helper: open existing attribute or create new one with the given type+space.
35 // Used by the special branches (vlen_text / ragged_vlen / key_value / composite)
36 // where the standard h5::create<T> path (which uses dt_t<T>) cannot supply the
37 // non-trivial HDF5 type — the caller builds the type explicitly.
38 inline h5::at_t open_or_create_attr(
39 hid_t parent, const std::string& name, hid_t type, hid_t space, hid_t acpl)
40 {
41 if (H5Aexists(parent, name.c_str()) > 0)
42 return h5::open(parent, name, h5::default_acpl);
43 hid_t id = H5I_UNINIT;
44 H5CPP_CHECK_NZ((id = H5Acreate2(parent, name.c_str(), type, space, acpl, H5P_DEFAULT)),
45 h5::error::io::attribute::create, "couldn't create attribute.");
46 return h5::at_t{id};
47 }
48 }

50 // --- high-level dispatched awrite: kind × storage matrix, mirrors H5Dwrite ---
51 template <class T, class P, class... args_t>
52 inline std::enable_if_t<h5::impl::is_valid_attr<P>::value,
53 h5::at_t> awrite( const P& parent, const std::string& name, const T& ref, const h5::acpl_t& acpl = h5::default_acpl ) try {
54 using traits = h5::meta::access_traits_t<T>;
55 using sr_t = h5::meta::storage_representation_t;
56 using element_t = typename impl::decay<T>::type;
57 constexpr auto kind = traits::kind;
58 constexpr auto storage = h5::meta::storage_representation_v<T>;
59
60 // Stopper: unsupported storage usually means an unregistered POD aggregate, a
61 // deeply-nested container, or std::vector<bool>. Bootstrap-aware (see H5Dwrite).
62#ifndef H5CPP_BUILDING_TYPE_INFO
63 static_assert(storage != sr_t::unsupported,
64 "h5::awrite: storage_representation_v<T> resolved to 'unsupported'. "
65 "Check: unregistered POD aggregate (use H5CPP_REGISTER_STRUCT), "
66 "std::vector<bool>, or container nesting beyond vector<vector<T>>/vector<string>.");
67#endif
68
69 // Stopper: containers of containers must route via VLEN storage.
70 static_assert(
71 !h5::meta::is_stl_like<element_t>::value ||
72 storage == sr_t::ragged_vlen_dataset ||
73 storage == sr_t::vlen_text_dataset ||
74 storage == sr_t::array_dataset ||
75 storage == sr_t::array_element ||
76 storage == sr_t::fls_dataset ||
77 storage == sr_t::fixed_inner_extent_dataset,
78 "h5::awrite: containers of containers are only supported for vector<string> "
79 "(vlen_text_dataset) or vector<vector<T>> (ragged_vlen_dataset).");
80
81 if constexpr (kind == h5::meta::access_t::composite) {
82 // scalar composite (std::tuple<Ts...>): pack into buffer matching
83 // the HDF5 compound type's field offsets, scalar dataspace.
84 std::vector<char> buf(traits::bytes());
85 traits::pack(ref, buf.data());
86 h5::meta::resolved_type_t<T> mem_type;
87 h5::sp_t space{H5Screate(H5S_SCALAR)};
88 h5::at_t attr = detail::open_or_create_attr(
89 static_cast<hid_t>(parent), name,
90 static_cast<hid_t>(mem_type), static_cast<hid_t>(space),
91 static_cast<hid_t>(acpl));
92 H5CPP_CHECK_NZ(
93 H5Awrite(static_cast<hid_t>(attr), static_cast<hid_t>(mem_type), buf.data()),
94 h5::error::io::attribute::write, "couldn't write composite attribute.");
95 return attr;
96 } else if constexpr (kind == h5::meta::access_t::text
97 && storage == sr_t::fixed_length_string) {
98 // char[N] / std::array<char,N> — H5T_C_S1 + H5Tset_size(N), scalar.
99 hid_t fls_type = H5Tcopy(H5T_C_S1);
100 H5Tset_size(fls_type, traits::fixed_length);
101 h5::sp_t space{H5Screate(H5S_SCALAR)};
102 h5::at_t attr = detail::open_or_create_attr(
103 static_cast<hid_t>(parent), name,
104 fls_type, static_cast<hid_t>(space), static_cast<hid_t>(acpl));
105 H5CPP_CHECK_NZ(
106 H5Awrite(static_cast<hid_t>(attr), fls_type, traits::data(ref)),
107 h5::error::io::attribute::write, "couldn't write fixed-length string attribute.");
108 H5Tclose(fls_type);
109 return attr;
110 } else if constexpr (kind == h5::meta::access_t::text) {
111 // Scalar VLEN text (std::string, std::string_view, const char*, char*).
112 h5::dt_t<char*> vlen_str;
113 h5::sp_t space{H5Screate(H5S_SCALAR)};
114 h5::at_t attr = detail::open_or_create_attr(
115 static_cast<hid_t>(parent), name,
116 static_cast<hid_t>(vlen_str), static_cast<hid_t>(space),
117 static_cast<hid_t>(acpl));
118 const char* ptr = traits::data(ref);
119 H5CPP_CHECK_NZ(
120 H5Awrite(static_cast<hid_t>(attr), static_cast<hid_t>(vlen_str), &ptr),
121 h5::error::io::attribute::write, "couldn't write text attribute.");
122 return attr;
123 } else if constexpr (storage == sr_t::array_element) {
124 // std::array<T,N> / T[N] (non-char) — scalar dataspace, H5T_ARRAY[N] element.
125 using element_t_loc = typename traits::element_t;
126 auto dims = traits::size(ref);
127 hsize_t array_dims[H5CPP_MAX_RANK];
128 for (std::size_t i = 0; i < dims.size(); ++i) array_dims[i] = dims[i];
129 h5::meta::resolved_type_t<element_t_loc> base_type;
130 hid_t array_type = H5Tarray_create(static_cast<hid_t>(base_type),
131 static_cast<unsigned>(dims.size()), array_dims);
132 h5::sp_t space{H5Screate(H5S_SCALAR)};
133 h5::at_t attr = detail::open_or_create_attr(
134 static_cast<hid_t>(parent), name,
135 array_type, static_cast<hid_t>(space), static_cast<hid_t>(acpl));
136 H5CPP_CHECK_NZ(
137 H5Awrite(static_cast<hid_t>(attr), array_type, traits::data(ref)),
138 h5::error::io::attribute::write, "couldn't write array_element attribute.");
139 H5Tclose(array_type);
140 return attr;
141 } else if constexpr (storage == sr_t::array_dataset) {
142 // std::vector<std::array<T,N>> (non-char) — rank-1 of H5T_ARRAY[N].
143 using inner_t = std::remove_cv_t<typename std::remove_reference_t<T>::value_type>;
144 using elem_scalar = typename inner_t::value_type;
145 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
146 hsize_t array_dims[1] = { static_cast<hsize_t>(N_inner) };
147 h5::meta::resolved_type_t<elem_scalar> base_type;
148 hid_t array_type = H5Tarray_create(static_cast<hid_t>(base_type), 1, array_dims);
149 hsize_t outer = static_cast<hsize_t>(ref.size());
150 h5::sp_t space{H5Screate_simple(1, &outer, nullptr)};
151 h5::at_t attr = detail::open_or_create_attr(
152 static_cast<hid_t>(parent), name,
153 array_type, static_cast<hid_t>(space), static_cast<hid_t>(acpl));
154 H5CPP_CHECK_NZ(
155 H5Awrite(static_cast<hid_t>(attr), array_type,
156 ref.empty() ? nullptr : ref.front().data()),
157 h5::error::io::attribute::write, "couldn't write array_dataset attribute.");
158 H5Tclose(array_type);
159 return attr;
160 } else if constexpr (storage == sr_t::fls_dataset) {
161 // std::vector<std::array<char,N>> — rank-1 of H5T_C_S1+set_size(N).
162 using inner_t = std::remove_cv_t<typename std::remove_reference_t<T>::value_type>;
163 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
164 hid_t fls_type = H5Tcopy(H5T_C_S1);
165 H5Tset_size(fls_type, N_inner);
166 hsize_t outer = static_cast<hsize_t>(ref.size());
167 h5::sp_t space{H5Screate_simple(1, &outer, nullptr)};
168 h5::at_t attr = detail::open_or_create_attr(
169 static_cast<hid_t>(parent), name,
170 fls_type, static_cast<hid_t>(space), static_cast<hid_t>(acpl));
171 H5CPP_CHECK_NZ(
172 H5Awrite(static_cast<hid_t>(attr), fls_type,
173 ref.empty() ? nullptr : ref.front().data()),
174 h5::error::io::attribute::write, "couldn't write fls_dataset attribute.");
175 H5Tclose(fls_type);
176 return attr;
177 } else if constexpr (kind == h5::meta::access_t::object ||
178 kind == h5::meta::access_t::contiguous) {
179 // arithmetic / pair / complex / std::vector<T> / std::array<T,N> / registered aggregate
180 h5::current_dims_t current_dims = traits::size(ref);
181 using attr_element_t = typename traits::element_t;
182 h5::at_t attr = ( H5Aexists(static_cast<hid_t>(parent), name.c_str() ) > 0 ) ?
183 h5::open(parent, name, h5::default_acpl) :
184 h5::create<attr_element_t>(parent, name, current_dims);
185 h5::awrite(attr, traits::data(ref));
186 return attr;
187 } else if constexpr (kind == h5::meta::access_t::pointers) {
188 // Nested: traits::element_t only referenced inside this block, where
189 // the pointers access_traits_t spec guarantees it exists.
190 if constexpr (storage == sr_t::vlen_text_dataset) {
191 // vector<string>: char* relay + H5T_VARIABLE string type, rank-1 attribute.
192 std::vector<const char*> relay;
193 relay.reserve(ref.size());
194 for (const auto& s : ref) relay.push_back(s.c_str());
195 h5::dt_t<char*> vlen_str;
196 hsize_t n = ref.size();
197 h5::sp_t space{H5Screate_simple(1, &n, nullptr)};
198 h5::at_t attr = detail::open_or_create_attr(
199 static_cast<hid_t>(parent), name,
200 static_cast<hid_t>(vlen_str), static_cast<hid_t>(space),
201 static_cast<hid_t>(acpl));
202 H5CPP_CHECK_NZ(
203 H5Awrite(static_cast<hid_t>(attr), static_cast<hid_t>(vlen_str), relay.data()),
204 h5::error::io::attribute::write, "couldn't write vlen_text attribute.");
205 return attr;
206 } else if constexpr (storage == sr_t::ragged_vlen_dataset) {
207 // vector<vector<T>>: hvl_t relay + H5Tvlen_create(base_type).
208 using inner_t = typename traits::element_t;
209 using elem_t = typename inner_t::value_type;
210 std::vector<hvl_t> relay(ref.size());
211 for (std::size_t i = 0; i < ref.size(); ++i) {
212 relay[i].len = ref[i].size();
213 relay[i].p = const_cast<void*>(static_cast<const void*>(ref[i].data()));
214 }
215 h5::meta::resolved_type_t<elem_t> base_type;
216 hid_t vlen_type = H5Tvlen_create(static_cast<hid_t>(base_type));
217 hsize_t n = ref.size();
218 h5::sp_t space{H5Screate_simple(1, &n, nullptr)};
219 h5::at_t attr = detail::open_or_create_attr(
220 static_cast<hid_t>(parent), name, vlen_type,
221 static_cast<hid_t>(space), static_cast<hid_t>(acpl));
222 H5CPP_CHECK_NZ(
223 H5Awrite(static_cast<hid_t>(attr), vlen_type, relay.data()),
224 h5::error::io::attribute::write, "couldn't write ragged_vlen attribute.");
225 H5Tclose(vlen_type);
226 return attr;
227 } else if constexpr (h5::meta::access_kind_v<typename traits::element_t> == h5::meta::access_t::composite) {
228 // vector<tuple<Ts...>>: pack each element via element traits.
229 using elem_t = typename traits::element_t;
230 using elem_traits = h5::meta::access_traits_t<elem_t>;
231 std::size_t n = ref.size();
232 std::vector<char> buf(n * elem_traits::bytes());
233 for (std::size_t i = 0; i < n; ++i)
234 elem_traits::pack(ref[i], buf.data() + i * elem_traits::bytes());
235 h5::meta::resolved_type_t<elem_t> mem_type;
236 hsize_t hn = n;
237 h5::sp_t space{H5Screate_simple(1, &hn, nullptr)};
238 h5::at_t attr = detail::open_or_create_attr(
239 static_cast<hid_t>(parent), name,
240 static_cast<hid_t>(mem_type), static_cast<hid_t>(space),
241 static_cast<hid_t>(acpl));
242 H5CPP_CHECK_NZ(
243 H5Awrite(static_cast<hid_t>(attr), static_cast<hid_t>(mem_type), buf.data()),
244 h5::error::io::attribute::write, "couldn't write composite-element attribute.");
245 return attr;
246 } else {
247 static_assert(storage == sr_t::vlen_text_dataset ||
248 storage == sr_t::array_dataset ||
249 storage == sr_t::array_element ||
250 storage == sr_t::fls_dataset ||
251 storage == sr_t::fixed_inner_extent_dataset, "h5::awrite: unsupported pointers storage class.");
252 return h5::at_t{H5I_UNINIT};
253 }
254 } else if constexpr (kind == h5::meta::access_t::iterators) {
255 // Nested for the same reason as the pointers block above.
256 if constexpr (storage == sr_t::key_value_dataset) {
257 // map<K,V> and variants: compound HDF5 type with "key" and "value" fields.
258 using element_t = typename traits::element_t;
259 using key_t = std::remove_const_t<typename element_t::first_type>;
260 using value_t = typename element_t::second_type;
261 struct kv_t { key_t key; value_t value; };
262 h5::meta::resolved_type_t<key_t> kt;
263 h5::meta::resolved_type_t<value_t> vt;
264 hid_t compound = H5Tcreate(H5T_COMPOUND, sizeof(kv_t));
265 H5Tinsert(compound, "key", offsetof(kv_t, key), static_cast<hid_t>(kt));
266 H5Tinsert(compound, "value", offsetof(kv_t, value), static_cast<hid_t>(vt));
267 std::vector<kv_t> buffer;
268 buffer.reserve(ref.size());
269 for (const auto& [k, v] : ref)
270 buffer.push_back({static_cast<key_t>(k), v});
271 hsize_t n = ref.size();
272 h5::sp_t space{H5Screate_simple(1, &n, nullptr)};
273 h5::at_t attr = detail::open_or_create_attr(
274 static_cast<hid_t>(parent), name, compound,
275 static_cast<hid_t>(space), static_cast<hid_t>(acpl));
276 H5CPP_CHECK_NZ(
277 H5Awrite(static_cast<hid_t>(attr), compound, buffer.data()),
278 h5::error::io::attribute::write, "couldn't write key_value attribute.");
279 H5Tclose(compound);
280 return attr;
281 } else if constexpr (h5::meta::access_kind_v<typename traits::element_t> == h5::meta::access_t::composite) {
282 // list<tuple>, set<tuple>, deque<tuple>: pack via element traits.
283 using elem_t = typename traits::element_t;
284 using elem_traits = h5::meta::access_traits_t<elem_t>;
285 std::size_t n = traits::size(ref)[0];
286 std::vector<char> buf(n * elem_traits::bytes());
287 std::size_t i = 0;
288 for (const auto& elem : ref) {
289 elem_traits::pack(elem, buf.data() + i * elem_traits::bytes());
290 ++i;
291 }
292 h5::meta::resolved_type_t<elem_t> mem_type;
293 hsize_t hn = n;
294 h5::sp_t space{H5Screate_simple(1, &hn, nullptr)};
295 h5::at_t attr = detail::open_or_create_attr(
296 static_cast<hid_t>(parent), name,
297 static_cast<hid_t>(mem_type), static_cast<hid_t>(space),
298 static_cast<hid_t>(acpl));
299 H5CPP_CHECK_NZ(
300 H5Awrite(static_cast<hid_t>(attr), static_cast<hid_t>(mem_type), buf.data()),
301 h5::error::io::attribute::write, "couldn't write iter-composite attribute.");
302 return attr;
303 } else {
304 // list<T>, set<T>, deque<T> with non-composite, std-layout element_t.
305 using iter_elem_t = typename impl::decay<typename traits::element_t>::type;
306 static_assert(std::is_standard_layout_v<iter_elem_t>,
307 "h5::awrite: iterator-staging path requires standard-layout element_t. "
308 "Wrap non-std-layout types in std::tuple (composite kind) or convert "
309 "to a flat representation before writing.");
310 std::vector<iter_elem_t> buffer;
311 buffer.reserve(traits::size(ref)[0]);
312 for (const auto& elem : ref) buffer.push_back(elem);
313 h5::current_dims_t current_dims = traits::size(ref);
314 h5::at_t attr = ( H5Aexists(static_cast<hid_t>(parent), name.c_str() ) > 0 ) ?
315 h5::open(parent, name, h5::default_acpl) :
316 h5::create<iter_elem_t>(parent, name, current_dims);
317 h5::awrite(attr, buffer.data());
318 return attr;
319 }
320 } else {
321 static_assert(kind != h5::meta::access_t::unsupported, "unsupported type for h5::awrite");
322 return h5::at_t{H5I_UNINIT}; // unreachable; silences "no return" diagnostic
323 }
324 } catch( const std::runtime_error& err ){
325 throw h5::error::io::attribute::write( err.what() );
326 }

328 // std::initializer_list<T> overload preserved verbatim (specialized form)
329 template<class T, class P>
330 inline std::enable_if_t<h5::impl::is_valid_attr<P>::value,
331 h5::at_t> awrite( const P& parent, const std::string& name, const std::initializer_list<T> ref, const h5::acpl_t& acpl = h5::default_acpl ) try {
332 h5::current_dims_t current_dims = impl::size( ref );
333 using element_t = typename impl::decay<std::initializer_list<T>>::type;
334
335 h5::at_t attr = ( H5Aexists(static_cast<hid_t>(parent), name.c_str() ) > 0 ) ?
336 h5::open(parent, name, h5::default_acpl) : h5::create<element_t>(parent, name, current_dims);
337 h5::awrite<element_t>(attr, impl::data( ref ) );
338 return attr;
339 } catch( const std::runtime_error& err ){
340 throw h5::error::io::attribute::write( err.what() );
341 }

344template<> inline
345h5::at_t h5::ds_t::operator[]( const char name[] ){
346 //we don't have the object parameters yet available the only thing to do is
347 //mark it H5I_UNINIT and in the second phase create the attribute
348 h5::at_t attr = ( H5Aexists(static_cast<hid_t>(*this), name ) > 0 ) ?
349 h5::open(static_cast<hid_t>( *this ), name, h5::default_acpl) : h5::at_t{H5I_UNINIT};
350 attr.ds = static_cast<hid_t>(*this);
351 attr.name = std::string(name);
352 return attr;
353}
