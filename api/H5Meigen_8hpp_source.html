253 namespace detail {
254 // SparseMatrix accessor — CSC only. innerIndex = row indices,
255 // outerIndex = column pointers (length cols+1).
256 // Precondition: src.makeCompressed() has been called. h5::write does
257 // not call it implicitly to avoid mutating a `const SparseMatrix&`.
258 template <class T, int O, class I>
259 struct eigen_spmat_accessor {
260 using value_type = T;
261 using sparse_type = ::Eigen::SparseMatrix<T,O,I>;
262 static_assert((O & ::Eigen::RowMajorBit) == 0,
263 "h5cpp: only ColMajor Eigen::SparseMatrix can be written as CSC.");
264 static const value_type* values(const sparse_type& m) { return m.valuePtr(); }
265 static std::size_t nnz (const sparse_type& m) { return static_cast<std::size_t>(m.nonZeros()); }
266 static std::size_t rows(const sparse_type& m) { return static_cast<std::size_t>(m.rows()); }
267 static std::size_t cols(const sparse_type& m) { return static_cast<std::size_t>(m.cols()); }
268
269 static void to_u32_outer(const sparse_type& m, std::uint32_t* dst) {
270 const std::size_t n = static_cast<std::size_t>(m.cols()) + 1;
271 const I* src = m.outerIndexPtr();
272 for (std::size_t i = 0; i < n; ++i)
273 dst[i] = static_cast<std::uint32_t>(src[i]);
274 }
275 static void to_u32_inner(const sparse_type& m, std::uint32_t* dst) {
276 const std::size_t n = static_cast<std::size_t>(m.nonZeros());
277 const I* src = m.innerIndexPtr();
278 for (std::size_t i = 0; i < n; ++i)
279 dst[i] = static_cast<std::uint32_t>(src[i]);
280 }
281
282 static sparse_type construct(
283 std::size_t n_rows, std::size_t n_cols, std::size_t nnz,
284 const std::uint32_t* outer, const std::uint32_t* inner,
285 const value_type* data)
286 {
287 // Stage uint32 buffers into the matrix's native StorageIndex
288 // type, then build via Eigen::Map (read-only) and assign.
289 std::vector<I> outer_buf(n_cols + 1);
290 for (std::size_t i = 0; i <= n_cols; ++i) outer_buf[i] = static_cast<I>(outer[i]);
291 std::vector<I> inner_buf(nnz);
292 for (std::size_t i = 0; i < nnz; ++i) inner_buf[i] = static_cast<I>(inner[i]);
293 // Map<const SparseMatrix> needs const-correct pointers; cast away
294 // const on data — Map only reads, then `mat = map` copies out.
295 ::Eigen::Map<const sparse_type> view(
296 static_cast<typename sparse_type::Index>(n_rows),
297 static_cast<typename sparse_type::Index>(n_cols),
298 static_cast<typename sparse_type::Index>(nnz),
299 outer_buf.data(), inner_buf.data(),
300 const_cast<value_type*>(data));
301 return sparse_type(view);
302 }
303 };

305 // SparseVector accessor — Eigen stores it as a single inner vector,
306 // so outerIndexPtr() is null. On-disk we synthesize a 2-element
307 // column pointer [0, nnz] for an Nx1 CSC matrix.
308 template <class T, int O, class I>
309 struct eigen_spvec_accessor {
310 using value_type = T;
311 using sparse_type = ::Eigen::SparseVector<T,O,I>;
312 static_assert((O & ::Eigen::RowMajorBit) == 0,
313 "h5cpp: only ColumnVector Eigen::SparseVector is supported.");
314 static const value_type* values(const sparse_type& v) { return v.valuePtr(); }
315 static std::size_t nnz (const sparse_type& v) { return static_cast<std::size_t>(v.nonZeros()); }
316 static std::size_t rows(const sparse_type& v) { return static_cast<std::size_t>(v.size()); }
317 static std::size_t cols(const sparse_type& /*v*/) { return 1; }
318
319 static void to_u32_outer(const sparse_type& v, std::uint32_t* dst) {
320 // synthesized [0, nnz] for an Nx1 column.
321 dst[0] = 0;
322 dst[1] = static_cast<std::uint32_t>(v.nonZeros());
323 }
324 static void to_u32_inner(const sparse_type& v, std::uint32_t* dst) {
325 const std::size_t n = static_cast<std::size_t>(v.nonZeros());
326 const I* src = v.innerIndexPtr();
327 for (std::size_t i = 0; i < n; ++i)
328 dst[i] = static_cast<std::uint32_t>(src[i]);
329 }
330
331 static sparse_type construct(
332 std::size_t n_rows, std::size_t n_cols, std::size_t nnz,
333 const std::uint32_t* /*outer*/, const std::uint32_t* inner,
334 const value_type* data)
335 {
336 if (n_cols != 1)
337 throw std::runtime_error(
338 "h5::read<Eigen::SparseVector>: on-disk matrix has more than "
339 "one column; cannot be loaded as a vector.");
340 sparse_type v(static_cast<typename sparse_type::Index>(n_rows));
341 v.reserve(static_cast<typename sparse_type::Index>(nnz));
342 for (std::size_t i = 0; i < nnz; ++i)
343 v.insertBack(static_cast<typename sparse_type::Index>(inner[i])) = data[i];
344 v.finalize();
345 return v;
346 }
347 };
