33 template<class T, class... args_t>
34 inline std::enable_if_t<!std::is_same_v<T,char**>,
35 void> read( const h5::ds_t& ds, T* ptr, args_t&&... args ) try {
36 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
37 static_assert( tcount::present, "h5::count_t{ ... } must be specified" );
38 static_assert( utils::is_supported<T>, "error: " H5CPP_supported_elementary_types );
39
40 auto tuple = std::forward_as_tuple(args...);
41 const h5::count_t& count = std::get<tcount::value>( tuple );
42
43 h5::offset_t default_offset{0,0,0,0,0,0,0}; // must match H5CPP_MAX_RANK=7
44 const h5::offset_t& offset = arg::get( default_offset, args...);
45
46 h5::stride_t default_stride{1,1,1,1,1,1,1};
47 const h5::stride_t& stride = arg::get( default_stride, args...);
48
49 h5::block_t default_block{1,1,1,1,1,1,1};
50 const h5::block_t& block = arg::get( default_block, args...);
51
52 h5::count_t size; // compute actual memory space
53 for(int i=0;i<count.rank;i++) size[i] = count[i] * block[i];
54 size.rank = count.rank;
55
56 const h5::dxpl_t& dxpl = arg::get( h5::default_dxpl, args...);
57 H5CPP_CHECK_PROP( dxpl, h5::error::property_list::misc, "invalid data transfer property" );
58
59 h5::sp_t file_space = h5::get_space(ds);
60 int rank = h5::get_simple_extent_ndims( file_space );
61
62 if( rank != count.rank ) throw h5::error::io::dataset::read( H5CPP_ERROR_MSG( h5::error::msg::rank_mismatch ));
63 using element_t = typename impl::decay<T>::type;
64 h5::meta::resolved_type_t<element_t> mem_type;
65 hid_t dapl = h5::get_access_plist( ds );
66 // See H5Dwrite.hpp for the rationale: pipeline path uses H5Dread_chunk,
67 // which only works on chunked datasets. Guard on H5D_CHUNKED so DAPLs
68 // with the flag applied to contiguous datasets fall through to H5Dread.
69 const bool use_pipeline = [&]() {
70 if (!H5Pexist(dapl, H5CPP_DAPL_HIGH_THROUGHPUT)) return false;
71 hid_t dcpl_id = H5Dget_create_plist(static_cast<hid_t>(ds));
72 if (dcpl_id < 0) return false;
73 H5D_layout_t layout = H5Pget_layout(dcpl_id);
74 H5Pclose(dcpl_id);
75 return layout == H5D_CHUNKED;
76 }();
77 if( use_pipeline ){
78 // Phase 1.3.3 — if the file's FAPL has h5::threads{N}, route
79 // reads through a local pool_pipeline_t. Currently pool_pipeline_t::
80 // read_chunk_impl is synchronous (parallel decompress is Phase 1.5+),
81 // so the FAPL-pool branch is semantically equivalent to the DAPL
82 // path today; the structure is in place for the read-ahead
83 // optimization to land later without changing call sites.
84 hid_t fid = H5Iget_file_id(static_cast<hid_t>(ds));
85 hid_t fapl = H5Fget_access_plist(fid);
86 auto pool = h5::impl::resolve_worker_pool(fapl);
87 if (pool) {
88 const unsigned cap = h5::impl::resolve_backpressure(
89 fapl, pool->worker_count());
90 h5::impl::pool_pipeline_t pipe(std::move(pool), cap);
91 h5::dcpl_t dcpl{H5Dget_create_plist(static_cast<hid_t>(ds))};
92 hid_t type_id = H5Dget_type(static_cast<hid_t>(ds));
93 size_t elem_sz = H5Tget_size(type_id);
94 H5Tclose(type_id);
95 pipe.set_cache(dcpl, elem_sz);
96 pipe.read(ds, offset, stride, block, count, dxpl, ptr);
97 } else {
98 h5::impl::pipeline_t<impl::basic_pipeline_t>* filters;
99 H5Pget(dapl, H5CPP_DAPL_HIGH_THROUGHPUT, &filters);
100 filters->read(ds, offset, stride, block, count, dxpl, ptr);
101 }
102 H5Pclose(fapl);
103 H5Fclose(fid);
104 }else{
105 h5::sp_t mem_space = h5::create_simple( size );
106 h5::select_all( mem_space );
107 h5::select_hyperslab( file_space, offset, stride, count, block);
108
109 H5CPP_CHECK_NZ( H5Dread(
110 static_cast<hid_t>( ds ), static_cast<hid_t>(mem_type), static_cast<hid_t>(mem_space),
111 static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), ptr ), h5::error::io::dataset::read, h5::error::msg::read_dataset);
112 }
113 } catch ( const std::runtime_error& err ){
114 throw h5::error::io::dataset::read( err.what() );
115 }

234 template<class T, class D=typename impl::decay<T>::type, class... args_t>
235 inline std::enable_if_t<!std::is_same_v<D,std::string>,
236 T> read( const h5::ds_t& ds, args_t&&... args ){
237 // if 'count' isn't specified use the one inside the hdf5 file, once it is obtained
238 // collapse dimensions to the rank of the object returned and create this T object
239 // update the content by we're good to go, since stride and offset can be processed in the
240 // update step
241 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
242 using element_type = typename impl::decay<T>::type;
243
244 h5::count_t size;
245 const h5::count_t& count = arg::get(size, args...);
246 h5::block_t default_block{1,1,1,1,1,1,1};
247 const h5::block_t& block = arg::get( default_block, args...);
248
249 if constexpr( !tcount::present ){ // read count ::= current_dim of file space
250 h5::sp_t file_space = h5::get_space( ds );
251 h5::get_simple_extent_dims(file_space, size);
252 } else {
253 for(int i=0;i<count.rank;i++) size[i] = count[i] * block[i];
254 size.rank = count.rank;
255 }
256
257 using traits = h5::meta::access_traits_t<T>;
258 constexpr auto kind = traits::kind;
259 if constexpr (kind == h5::meta::access_t::iterators) {
260 size_t n = 1;
261 for (int i = 0; i < size.rank; ++i) n *= size[i];
262 std::vector<element_type> flat(n);
263 ::h5::read<element_type>(ds, flat.data(), count, args...);
264 if constexpr (std::is_same_v<T, std::forward_list<element_type>>) {
265 T result;
266 result.assign(flat.begin(), flat.end());
267 return result;
268 } else {
269 return T(flat.begin(), flat.end());
270 }
271 } else {
272 T ref = impl::get<T>::ctor( count );
273 element_type *ptr = impl::data( ref );
274 ::h5::read<element_type>(ds, ptr, count, args...);
275 return ref;
276 }
277 }
