7// Phase II FAPL executor property — same shared_ptr-in-slot pattern proven
8// by Phase I's worker pool (H5Pthreads.hpp). An executor_t lives behind
9// a shared_ptr stored in a heap-allocated holder whose address occupies
10// the H5Pinsert2 value slot; copy_cb clones the slot but aliases the same
11// executor (refcount +1), close_cb drops one slot (refcount -1). When
12// the last fd dies, the executor destructor joins the worker thread.
13//
14// h5::async::create / h5::async::open install this property in addition
15// to (and consuming) any h5::threads{N} pool the user chained in. If no
16// h5::threads{N} is present, fapl_async_set auto-installs a default-sized
17// pool so the executor always has a pool reference.
18

28#define H5CPP_FAPL_EXECUTOR "h5cpp_fapl_executor"
29
30// Heap-allocated holder, parallel to worker_pool_slot_t in H5Pthreads.hpp.
31struct executor_slot_t {
32 std::shared_ptr<executor_t> exec;
33};
34
35// Copy: HDF5 memcpy'd the slot pointer into the destination. Allocate a
36// fresh holder whose shared_ptr aliases the same executor (++refcount).
37inline herr_t fapl_exec_copy_cb(const char* /*name*/, size_t /*size*/, void* value) {
38 auto** slot_loc = static_cast<executor_slot_t**>(value);
39 *slot_loc = new executor_slot_t{(*slot_loc)->exec};
40 return 0;
41}
42
43// Close: delete one holder; shared_ptr drops one reference. The
44// executor_t destructor (which joins the worker thread) runs when the
45// last reference is released.
46inline herr_t fapl_exec_close_cb(const char* /*name*/, size_t /*size*/, void* ptr) {
47 delete *static_cast<executor_slot_t**>(ptr);
48 return 0;
49}
50
51// Setter invoked by h5::async::create / h5::async::open. Idempotent —
52// if the FAPL already has the property installed, leaves it alone.
53// Auto-installs a default-sized worker_pool_t (n=0) when no h5::threads{N}
54// was chained into the FAPL, so the executor always has a pool to hand
55// to compression callbacks (Phase II PR-B wires that connection).
56inline herr_t fapl_async_set(::hid_t fapl) {
57 if (H5Pexist(fapl, H5CPP_FAPL_EXECUTOR)) return 0;
58
59 auto pool = resolve_worker_pool(fapl);
60 if (!pool) {
61 // No h5::threads{N} in the chain — install a default-sized pool
62 // so the executor has a compression backend available.
63 fapl_threads_set(fapl, 0); // 0 → hardware_concurrency()
64 pool = resolve_worker_pool(fapl);
65 }
66
67 auto* slot = new executor_slot_t{
68 std::make_shared<executor_t>(std::move(pool))
69 };
70 return H5Pinsert2(fapl, H5CPP_FAPL_EXECUTOR,
71 sizeof(executor_slot_t*), &slot,
72 nullptr, // set
73 nullptr, // get
74 nullptr, // prp_del
75 fapl_exec_copy_cb,
76 nullptr, // compare
77 fapl_exec_close_cb);
78}
79
80// Consumer-site: given a FAPL id, retrieve the executor shared_ptr if one
81// is installed. Returns nullptr when the property is absent (classic-mode
82// FAPL — caller should not have been routed here).
83inline std::shared_ptr<executor_t> resolve_executor(::hid_t fapl_id) noexcept {
84 if (fapl_id < 0 || H5Iis_valid(fapl_id) <= 0) return nullptr;
85 if (!H5Pexist(fapl_id, H5CPP_FAPL_EXECUTOR)) return nullptr;
86 executor_slot_t* slot = nullptr;
87 H5Pget(fapl_id, H5CPP_FAPL_EXECUTOR, &slot);
88 return slot ? slot->exec : nullptr;
89}
