18// Sparse matrix / vector I/O.
19//
20// Public API:
21// h5::gr_t h5::write(parent, "name", spmat); // writes CSC group
22// Sparse h5::read<Sparse>(parent, "name"); // reads it back
23//
24// Parent may be h5::fd_t, h5::gr_t, or hid_t. The dense h5::write / h5::read
25// overloads (in H5Dwrite.hpp / H5Dread.hpp) exclude sparse T via the
26// is_sparse_v guard so these overloads win without ambiguity.

36 // Write a contiguous 1-D dataset of `n` elements of type T under `loc`.
37 // Uses the raw HDF5 C API so this helper works for any location id
38 // (group, file). Returns nothing; throws on HDF5 error.
39 template <class T>
40 inline void write_1d(::hid_t loc, const char* name, const T* ptr, hsize_t n) {
41 h5::meta::resolved_type_t<T> type;
42 hsize_t dims[1] = { n };
43 ::hid_t space = H5Screate_simple(1, dims, nullptr);
44 ::hid_t ds = H5Dcreate2(loc, name, static_cast<::hid_t>(type), space,
45 H5P_DEFAULT, H5P_DEFAULT, H5P_DEFAULT);
46 if (ds < 0) {
47 H5Sclose(space);
48 throw h5::error::io::dataset::create(
49 std::string("h5::write(sparse): failed to create dataset '") + name + "'");
50 }
51 herr_t err = H5Dwrite(ds, static_cast<::hid_t>(type),
52 H5S_ALL, H5S_ALL, H5P_DEFAULT, ptr);
53 H5Dclose(ds);
54 H5Sclose(space);
55 if (err < 0)
56 throw h5::error::io::dataset::write(
57 std::string("h5::write(sparse): H5Dwrite failed for '") + name + "'");
58 }

60 // Read a 1-D dataset of element type T into a std::vector<T>. Throws if
61 // the dataset is not 1-D.
62 template <class T>
63 inline std::vector<T> read_1d(::hid_t loc, const char* name) {
64 ::hid_t ds = H5Dopen2(loc, name, H5P_DEFAULT);
65 if (ds < 0)
66 throw h5::error::io::dataset::read(
67 std::string("h5::read(sparse): dataset '") + name + "' not found");
68 ::hid_t space = H5Dget_space(ds);
69 int rank = H5Sget_simple_extent_ndims(space);
70 if (rank != 1) {
71 H5Sclose(space); H5Dclose(ds);
72 throw h5::error::io::dataset::read(
73 std::string("h5::read(sparse): dataset '") + name + "' is not 1-D");
74 }
75 hsize_t n = 0;
76 H5Sget_simple_extent_dims(space, &n, nullptr);
77 std::vector<T> out(static_cast<std::size_t>(n));
78 h5::meta::resolved_type_t<T> type;
79 herr_t err = H5Dread(ds, static_cast<::hid_t>(type),
80 H5S_ALL, H5S_ALL, H5P_DEFAULT, out.data());
81 H5Sclose(space); H5Dclose(ds);
82 if (err < 0)
83 throw h5::error::io::dataset::read(
84 std::string("h5::read(sparse): H5Dread failed for '") + name + "'");
85 return out;
86 }

88 // Read a scalar VLEN string attribute. We can't reuse h5::aread<std::string>
89 // because it is keyed to h5::ds_t — but the underlying H5A* C API accepts
90 // any location, so this helper is a thin wrapper.
91 inline std::string read_string_attr(::hid_t loc, const char* name) {
92 ::hid_t attr = H5Aopen(loc, name, H5P_DEFAULT);
93 if (attr < 0)
94 throw h5::error::io::attribute::read(
95 std::string("h5::read(sparse): attribute '") + name + "' not found");
96 ::hid_t atype = H5Aget_type(attr);
97 // copy to a known VLEN UTF-8 string type so the read produces a char*
98 ::hid_t mem_type = H5Tcopy(H5T_C_S1);
99 H5Tset_size(mem_type, H5T_VARIABLE);
100 H5Tset_cset(mem_type, H5T_CSET_UTF8);
101 char* raw = nullptr;
102 herr_t err = H5Aread(attr, mem_type, &raw);
103 std::string out = (err >= 0 && raw) ? std::string(raw) : std::string{};
104 // reclaim if HDF5 allocated
105 if (raw) {
106 ::hid_t sp = H5Aget_space(attr);
107#if H5_VERSION_GE(1,12,0)
108 H5Treclaim(mem_type, sp, H5P_DEFAULT, &raw);
109#else
110 H5Dvlen_reclaim(mem_type, sp, H5P_DEFAULT, &raw);
111#endif
112 H5Sclose(sp);
113 }
114 H5Tclose(mem_type);
115 H5Tclose(atype);
116 H5Aclose(attr);
117 if (err < 0)
118 throw h5::error::io::attribute::read(
119 std::string("h5::read(sparse): H5Aread failed for '") + name + "'");
120 return out;
121 }

125 // -------- write -----------------------------------------------------
126 // SFINAE: this overload activates only for sparse T; the dense
127 // h5::write(fd, path, ref) overload excludes is_sparse_v<T>.
128 template <class T, class LOC,
129 class = std::enable_if_t<h5::impl::is_valid_group_parent<LOC>::value
130 && h5::meta::is_sparse_v<T>>>
131 inline h5::gr_t write(const LOC& parent, const std::string& path, const T& src) try {
132 using S = h5::meta::sparse_traits<T>;
133 using value_type = typename S::value_type;
134
135 const std::size_t n_rows = S::rows(src);
136 const std::size_t n_cols = S::cols(src);
137 const std::size_t nnz = S::nnz(src);
138
139 constexpr std::size_t U32_MAX = std::numeric_limits<std::uint32_t>::max();
140 if (n_rows > U32_MAX || n_cols > U32_MAX || nnz > U32_MAX)
141 throw h5::error::io::dataset::write(
142 "h5::write(sparse): on-disk indices are uint32; shape or nnz exceeds 2^32-1.");
143
144 // Stage the CSC arrays into the on-disk uint32 buffers.
145 std::vector<std::uint32_t> indptr(n_cols + 1);
146 S::to_u32_outer(src, indptr.data());
147 std::vector<std::uint32_t> indices(nnz);
148 if (nnz > 0) S::to_u32_inner(src, indices.data());
149
150 std::array<std::uint64_t, 2> shape = {
151 static_cast<std::uint64_t>(n_rows),
152 static_cast<std::uint64_t>(n_cols)
153 };
154
155 h5::gr_t gr = h5::gcreate(parent, path);
156 const ::hid_t gid = static_cast<::hid_t>(gr);
157
158 // Empty matrices still need a `data` dataset so the group is a
159 // complete CSC record; HDF5 accepts a zero-length simple dataspace.
160 impl::sparse::write_1d<value_type>(gid, h5::meta::sparse::k_data,
161 S::values(src), static_cast<hsize_t>(nnz));
162 impl::sparse::write_1d<std::uint32_t>(gid, h5::meta::sparse::k_indices,
163 indices.data(), static_cast<hsize_t>(nnz));
164 impl::sparse::write_1d<std::uint32_t>(gid, h5::meta::sparse::k_indptr,
165 indptr.data(), static_cast<hsize_t>(n_cols + 1));
166 impl::sparse::write_1d<std::uint64_t>(gid, h5::meta::sparse::k_shape,
167 shape.data(), 2u);
168
169 // Self-describing attributes so scipy / h5sparse / 10x readers can
170 // dispatch without out-of-band knowledge.
171 h5::awrite(gr, h5::meta::sparse::k_format, std::string(h5::meta::sparse::v_csc));
172 h5::awrite(gr, h5::meta::sparse::k_axis, std::string(h5::meta::sparse::v_column));
173 return gr;
174 } catch (const std::exception& err) {
175 throw h5::error::io::dataset::write(err.what());
176 }

178 // -------- read ------------------------------------------------------
179 // SFINAE: activates only for sparse T; the dense h5::read<T>(loc, path)
180 // overload excludes is_sparse_v<T>.
181 template <class T, class LOC,
182 class = std::enable_if_t<h5::impl::is_valid_group_parent<LOC>::value
183 && h5::meta::is_sparse_v<T>>>
184 inline T read(const LOC& parent, const std::string& path) try {
185 using S = h5::meta::sparse_traits<T>;
186 using value_type = typename S::value_type;
187
188 h5::gr_t gr = h5::gopen(parent, path);
189 const ::hid_t gid = static_cast<::hid_t>(gr);
190
191 // Validate format; future RowMajor / BCRS / etc. support will branch here.
192 const std::string fmt = impl::sparse::read_string_attr(gid, h5::meta::sparse::k_format);
193 if (fmt != h5::meta::sparse::v_csc)
194 throw h5::error::io::dataset::read(
195 "h5::read<sparse>: expected @format='csc', got '" + fmt + "'");
196
197 auto shape = impl::sparse::read_1d<std::uint64_t>(gid, h5::meta::sparse::k_shape);
198 if (shape.size() != 2)
199 throw h5::error::io::dataset::read(
200 "h5::read<sparse>: @shape must have exactly 2 elements [rows, cols]");
201 const std::size_t n_rows = static_cast<std::size_t>(shape[0]);
202 const std::size_t n_cols = static_cast<std::size_t>(shape[1]);
203
204 auto indptr = impl::sparse::read_1d<std::uint32_t>(gid, h5::meta::sparse::k_indptr);
205 auto indices = impl::sparse::read_1d<std::uint32_t>(gid, h5::meta::sparse::k_indices);
206 auto data = impl::sparse::read_1d<value_type> (gid, h5::meta::sparse::k_data);
207
208 if (indptr.size() != n_cols + 1)
209 throw h5::error::io::dataset::read(
210 "h5::read<sparse>: indptr length doesn't match cols+1");
211 if (indices.size() != data.size())
212 throw h5::error::io::dataset::read(
213 "h5::read<sparse>: indices/data length mismatch");
214
215 return S::construct(n_rows, n_cols, indices.size(),
216 indptr.data(), indices.data(), data.data());
217 } catch (const std::exception& err) {
218 throw h5::error::io::dataset::read(err.what());
219 }
