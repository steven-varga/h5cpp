7// Phase II executor — one thread per async fd. Every HDF5 C-API call on an
8// h5::async::* descriptor is serialized through this thread via
9// submit_and_wait, restoring thread-safety for HDF5 builds compiled without
10// --enable-threadsafe. Compression continues to parallelize through the
11// Phase I worker_pool_t held by reference.
12//
13// Construction model mirrors Phase I's worker_pool_t (H5Pthreads.hpp): single
14// mutex-guarded queue + doorbell + stoppable_thread_t, std::packaged_task in
15// shared_ptr wrapped in std::function<void()> for type erasure. The only
16// substantive difference is the single worker thread and the sync wait
17// facade.

35struct executor_t {
36 // pool is optional today; Phase II operation overloads (PR-B) will
37 // require it. Default-construct keeps the scaffold trivially usable
38 // in lifecycle tests that don't exercise compression.
39 explicit executor_t(std::shared_ptr<worker_pool_t> pool = nullptr)
40 : pool_(std::move(pool)),
41 worker_([this](h5::detail::stop_token_t st) { worker_loop(st); })
42 {}

44 ~executor_t() {
45 wait_idle();
46 stopping_.store(true, std::memory_order_release);
47 bell_.ring();
48 // stoppable_thread_t dtor requests stop + joins
49 }

56 // Submit a callable and block until it completes on the executor thread.
57 // Exception in the callable propagates back to the submitting thread via
58 // future::get.
59 //
60 // Reentrant safety: if the caller is already running on the executor
61 // thread (e.g. one h5cpp operation calling another inside the executor),
62 // run the callable inline rather than queueing — queueing would
63 // deadlock the executor on itself.
64 template <typename Fn>
65 auto submit_and_wait(Fn&& fn) -> std::invoke_result_t<Fn> {
66 using R = std::invoke_result_t<Fn>;
67
68 if (std::this_thread::get_id() ==
69 worker_thread_id_.load(std::memory_order_acquire)) {
70 // Already on the executor thread — run inline, no queue trip.
71 // Exception thrown in fn propagates back to caller naturally
72 // (no packaged_task wrap needed in this branch).
73 return std::forward<Fn>(fn)();
74 }
75
76 auto task = std::make_shared<std::packaged_task<R()>>(std::forward<Fn>(fn));
77 auto fut = task->get_future();
78 in_flight_.fetch_add(1, std::memory_order_release);
79 {
80 std::lock_guard<std::mutex> lk(m_);
81 tasks_.emplace([task] { (*task)(); });
82 }
83 bell_.ring();
84 // Decrement counter on the submitter's path, AFTER fut.get()
85 // unblocks (causal: the worker has completed task() because the
86 // future is ready). RAII guard ensures the decrement also fires
87 // when fut.get() rethrows. This ties in_flight_ to "submit_and_wait
88 // is in flight on this thread" rather than "task is still queued"
89 // — tests that check in_flight() right after the call now see 0
90 // deterministically, no race with the worker thread.
91 struct decrement_on_exit {
92 std::atomic<int>& counter;
93 ~decrement_on_exit() { counter.fetch_sub(1, std::memory_order_release); }
94 } guard{in_flight_};
95 return fut.get(); // blocks on executor; rethrows exception
96 }
97
98 // Block until the queue drains. Called by ~executor_t before stop.
99 void wait_idle() {
100 while (in_flight_.load(std::memory_order_acquire) > 0)
101 std::this_thread::yield();
102 }
103
104 [[nodiscard]] std::shared_ptr<worker_pool_t> pool() const { return pool_; }
105 [[nodiscard]] int in_flight() const noexcept { return in_flight_.load(std::memory_order_acquire); }
106 [[nodiscard]] std::thread::id worker_thread_id() const noexcept {
107 return worker_thread_id_.load(std::memory_order_acquire);
108 }

110private:
111 void worker_loop(h5::detail::stop_token_t st) {
112 worker_thread_id_.store(std::this_thread::get_id(),
113 std::memory_order_release);
114 while (!st.stop_requested()) {
115 std::function<void()> task;
116 std::uint32_t last_seq = 0;
117 bool got_task = false;
118 {
119 std::lock_guard<std::mutex> lk(m_);
120 if (!tasks_.empty()) {
121 task = std::move(tasks_.front());
122 tasks_.pop();
123 got_task = true;
124 } else {
125 last_seq = bell_.load();
126 }
127 }
128
129 if (got_task) {
130 try { task(); } catch (...) { /* packaged_task captures it */ }
131 // in_flight_ is decremented by submit_and_wait on the
132 // submitter's path (see RAII guard there), not here.
133 // Worker just executes the task; the counter belongs to
134 // the submission's lifetime.
135 continue;
136 }
137
138 if (st.stop_requested() || stopping_.load(std::memory_order_acquire)) return;
139 bell_.wait(last_seq);
140 }
141 }
142
143 std::shared_ptr<worker_pool_t> pool_;
144 std::mutex m_;
145 std::queue<std::function<void()>> tasks_;
146 h5::detail::doorbell_t bell_;
147 std::atomic<int> in_flight_{0};
148 std::atomic<bool> stopping_{false};
149 std::atomic<std::thread::id> worker_thread_id_{};
150 h5::detail::stoppable_thread_t worker_; // declared LAST so all
151 // members are live when
152 // worker_loop starts
153};
