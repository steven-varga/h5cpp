37 // Phase II — async descriptors carry a shared_ptr<executor_t> field
38 // directly on the wrapper. Why direct storage and not the FAPL slot
39 // pattern from Phase I: HDF5 1.10.9's H5Fget_access_plist returns a
40 // synthetic FAPL reconstructed from standard properties only; user
41 // properties installed via H5Pinsert2 are dropped. Storing the
42 // executor inside the wrapper class lets operation overloads in
43 // Phase II PR-B reach it as `fd.exec` without round-tripping through
44 // HDF5's property machinery. std::shared_ptr's type-erased deleter
45 // makes the forward declaration sufficient — the complete type is
46 // only needed at h5::async::create / open (defined in H5async.hpp).
47 struct executor_t;
48}

126 // Phase II async-mode specialization — operator ::hid_t() is = delete'd so
127 // user code that accidentally hands an async descriptor to a raw HDF5 C
128 // API fails to compile with a clear "use of deleted function" diagnostic.
129 // h5cpp internal code reaches the raw handle via the public `handle`
130 // field (see workplan §4.4); user code routes through h5::write / h5::read
131 // / etc. which detect the type via is_async_v<> and dispatch through the
132 // FAPL-resolved executor.
133 template<class T, capi_close_t capi_close>
134 struct hid_t<T,capi_close, false,false,hdf5::any> {
135 using hidtype = T;
136
137 // from CAPI — mirrors the true,true ctor; explicit so an accidental
138 // implicit promotion from ::hid_t doesn't slip an async wrapper in.
139 H5CPP__EXPLICIT hid_t( ::hid_t handle_ ) : handle( handle_ ){
140 if( H5Iis_valid( handle_ ) )
141 H5Iinc_ref( handle_ );
142 }
143
144 // Factory ctor — h5::async::create / open construct the executor
145 // during file creation and inject it here so operation overloads
146 // (Phase II PR-B) can reach it via `fd.exec`. Used by mode-
147 // transitive factories too (ds_t inherits parent fd's executor).
148 hid_t( ::hid_t handle_, std::shared_ptr<h5::impl::executor_t> e ) noexcept
149 : handle( handle_ ), exec( std::move(e) ) {}
150
151 // TO CAPI — DELETED. Async descriptors must not be implicitly
152 // converted back to ::hid_t; doing so would let user code call
153 // HDF5 directly and bypass the executor thread. Internal code
154 // reads the raw value from `handle` directly.
155 operator ::hid_t() const = delete;
156
157 // direct-initialization ctor; matches the classic shape — does not
158 // increment the refcount (caller owns the handle).
159 hid_t( std::initializer_list<::hid_t> fd ) : handle( *fd.begin() ){}
160
161 hid_t() : handle(H5I_UNINIT) {}
162
163 hid_t( const hid_t& ref ){
164 handle = ref.handle;
165 if( H5Iis_valid( handle ) )
166 H5Iinc_ref( handle );
167 exec = ref.exec; // shared_ptr copy bumps refcount
168 }
169 hid_t& operator=( const hid_t& ref ){
170 if( this == &ref ) return *this;
171 if( H5Iis_valid( handle ) )
172 capi_close( handle );
173 handle = ref.handle;
174 if( H5Iis_valid( handle ) )
175 H5Iinc_ref( handle );
176 exec = ref.exec;
177 return *this;
178 }
179 hid_t( hid_t&& ref ) noexcept {
180 handle = ref.handle;
181 ref.handle = H5I_UNINIT;
182 exec = std::move(ref.exec);
183 }
184 hid_t& operator=( hid_t&& ref ) noexcept {
185 if( this == &ref ) return *this;
186 if( H5Iis_valid( handle ) )
187 capi_close( handle );
188 handle = ref.handle;
189 ref.handle = H5I_UNINIT;
190 exec = std::move(ref.exec);
191 return *this;
192 }
193 ~hid_t(){
194 if( H5Iis_valid( handle ) )
195 capi_close( handle );
196 }
197
198 // Public so internal h5cpp code (the executor, dispatch lambdas)
199 // can read the raw id without invoking the deleted conversion.
200 // User code is expected to use h5::write / h5::read / h5::async::*
201 // factories rather than touch this field directly.
202 ::hid_t handle;
203
204 // Phase II — shared_ptr to the executor that owns this descriptor's
205 // HDF5 lifetime. Populated by h5::async::create / open at the
206 // file-level, then propagated to derived descriptors (async ds,
207 // async at, etc.) by mode-transitive factories. May be null on
208 // default-constructed async wrappers (un-initialized state).
209 std::shared_ptr<h5::impl::executor_t> exec;
210 };
