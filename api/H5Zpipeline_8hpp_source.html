40 inline size_t round_up_to_alignment(size_t size, size_t alignment) {
41 return alignment ? ((size + alignment - 1) / alignment) * alignment : size;
42 }
43
44 inline aligned_ptr make_aligned(size_t alignment, size_t size) {
45 const size_t allocation_size = round_up_to_alignment(size, alignment);
46 void* ptr = nullptr;
47#ifdef _WIN32
48 ptr = _aligned_malloc(allocation_size, alignment);
49#else
50 if (posix_memalign(&ptr, alignment, allocation_size) != 0)
51 ptr = nullptr;
52#endif
53 if (!ptr)
54 throw std::bad_alloc();
55 return aligned_ptr(static_cast<char*>(ptr));
56 }

58 // ------------------------------------------------------------------
59 // Bump-pointer arena: eliminates per-allocation system calls.
60 // Default 256 MiB holds ~32K 8 KB chunks; tunable at construction.
61 // ------------------------------------------------------------------
62 struct chunk_arena_t {
63 static constexpr size_t alignment = H5CPP_MEM_ALIGNMENT;
64 static constexpr size_t default_capacity = 256 * 1024 * 1024;
65
66 struct arena_deleter {
67 void operator()(char* ptr) const {
68#ifdef _WIN32
69 _aligned_free(ptr);
70#else
71 std::free(ptr);
72#endif
73 }
74 };
75 std::unique_ptr<char, arena_deleter> base;
76 char* bump = nullptr;
77 char* end = nullptr;
78
79 explicit chunk_arena_t(size_t capacity = default_capacity) {
80 void* ptr = nullptr;
81#ifdef _WIN32
82 ptr = _aligned_malloc(capacity, alignment);
83#else
84 if (posix_memalign(&ptr, alignment, capacity) != 0)
85 ptr = nullptr;
86#endif
87 if (!ptr)
88 throw std::bad_alloc();
89 base.reset(static_cast<char*>(ptr));
90 bump = base.get();
91 end = bump + capacity;
92 }
93
94 chunk_arena_t(chunk_arena_t&&) = default;
95 chunk_arena_t& operator=(chunk_arena_t&&) = default;
96 chunk_arena_t(const chunk_arena_t&) = delete;
97 chunk_arena_t& operator=(const chunk_arena_t&) = delete;
98
99 [[nodiscard]] char* allocate(size_t size) {
100 size = round_up_to_alignment(size, alignment);
101 if (bump + size > end) [[unlikely]]
102 return allocate_fallback(size);
103 char* ptr = bump;
104 bump += size;
105 return ptr;
106 }
107
108 void reset() noexcept { bump = base.get(); }
109
110 bool owns(const void* ptr) const noexcept {
111 const char* p = static_cast<const char*>(ptr);
112 return p >= base.get() && p < end;
113 }
114
115 private:
116 [[nodiscard]] char* allocate_fallback(size_t size) {
117 void* ptr = nullptr;
118#ifdef _WIN32
119 ptr = _aligned_malloc(size, alignment);
120#else
121 if (posix_memalign(&ptr, alignment, size) != 0)
122 ptr = nullptr;
123#endif
124 if (!ptr)
125 throw std::bad_alloc();
126 return static_cast<char*>(ptr);
127 }
128 };

137 inline void simd_memset_zero(char* dst, size_t n) {
138#if defined(__AVX2__)
139 size_t m = n;
140 const __m256i zero = _mm256_setzero_si256();
141 for (; m >= 32; m -= 32, dst += 32)
142 _mm256_store_si256(reinterpret_cast<__m256i*>(dst), zero);
143 std::memset(dst, 0, m);
144#else
145 std::memset(dst, 0, n);
146#endif
147 }

149 inline void nontemporal_memcpy(char* __restrict dst, const char* __restrict src, size_t n) {
150#if defined(__AVX__)
151 size_t m = n;
152 for (; m >= 32; m -= 32, dst += 32, src += 32) {
153 _mm256_stream_si256(reinterpret_cast<__m256i*>(dst),
154 _mm256_loadu_si256(reinterpret_cast<const __m256i*>(src)));
155 }
156 _mm_sfence();
157 std::memcpy(dst, src, m);
158#else
159 std::memcpy(dst, src, n);
160#endif
161 }

288inline void h5::impl::pipeline_t<Derived>::set_cache( const h5::dcpl_t& dcpl, size_t element_size ) {
289 n = 1, tail = 0; this->element_size = element_size;
290
291 // grab chunk dimensions, which is a block we're breaking data into
292 h5::chunk_t block;
293 if( (rank = h5::get_chunk_dims( dcpl, block )) == 0 )
294 throw std::runtime_error("data-space is rank 0, is data space a scalar? ");
295
296 //fix B block/chunk size for the lifespan of pipeline
297 for(hsize_t i=0; i<rank; i++ )
298 n *= block[i], B[i] = block[rank-i-1];
299
300 block_size = n*element_size;
301 unsigned filter_config;
302 unsigned N = H5Pget_nfilters( dcpl );
303 for(unsigned i=0; i<N; i++){
304 cd_size[i] = H5CPP_MAX_FILTER_PARAM;
305 H5Z_filter_t id = H5Pget_filter2( dcpl, i, &flags[i], &cd_size[i], cd_values[i], 0, nullptr, &filter_config );
306 push( filter::get_callback( id ) );
307 filter_id[i] = id;
308 // Guarantee that params[1] always holds the uncompressed chunk byte count as a
309 // reliable decompression output-size hint. External HDF5 files written by
310 // community plugins (LZ4 ID 32004, Zstd ID 32015, …) may store only
311 // compression level in params[0] and leave params[1] absent; libdeflate also
312 // requires the exact output size. Overwriting here is safe: encoders only read
313 // params[0] (compression level) and ignore params[1].
314 if (cd_size[i] < 2) cd_size[i] = 2;
315 cd_values[i][1] = static_cast<unsigned>(block_size & 0xFFFFFFFFu);
316 }
317
318 const size_t scratch_size = filter::filter_scratch_bound(block_size);
319 chunk0 = arena.allocate(scratch_size);
320 chunk1 = arena.allocate(scratch_size);
321
322 if( chunk0 == nullptr || chunk1 == nullptr )
323 throw h5::error::io::dataset::open( H5CPP_ERROR_MSG("CTOR: couldn't allocate memory for caching chunks, invalid/check size?"));
324}

332 filter_direction_t direction, const hsize_t* chunk_offset, const hsize_t* dims, void* ptr_ ){
333 (void)direction;
334 char* ptr = static_cast<char*>( ptr_ );
335 // compute edges if any - for the data size
336 // computes R residuals, and sets N dimension in reverse order
337 // when actual data dimensions - current_dims are greater then a chunk, it must be broken into chunk size
338 // if there are residuals on the edges, then padding is needed.
339 unsigned i=0;
340 for(; i<rank; i++ )
341 N[i] = dims[rank-i-1], Ry[i] = N[i] % B[i], Rx[i] = B[i] - chunk_offset[i] % B[i];
342 // set the rest to default
343 for(; i<H5CPP_MAX_RANK; i++) N[i] = B[i] = 1, Rx[i] = Ry[i] = 0;

350 h5cpp_outer( 2 ){ h5cpp_outer( 1 ){ h5cpp_outer( 0 ){
351 char* p = chunk0;
352 // at this point we have a single 'chunk' in this->chunk0 buffer ready to go
353 // the coordinates are in j indices
354 D[0] = j0, D[1]=j1, D[2]=j2, D[3]=j3, D[4]=j4, D[5]=j5, D[6]=j6;
355 //coordinates are reversed in D, invert them:

358 read_chunk( this->C, this->block_size, this->chunk0 );

374 inline void h5::impl::pipeline_t<Derived>::split_to_chunk_write(
375 filter_direction_t direction, const hsize_t* chunk_offset, const hsize_t* dims, const void* ptr_ ){
376 (void)direction;
377 const char* ptr = static_cast<const char*>( ptr_ ); const hsize_t* O = chunk_offset;
378 // compute edges if any - for the data size
379 // computes R residuals, and sets N dimension in reverse order
380 // when actual data dimensions - current_dims are greater then a chunk, it must be broken into chunk size
381 // if there are residuals on the edges, then padding is needed.
382 unsigned i=0;
383 for(; i<rank; i++ )
384 N[i] = dims[rank-i-1], Rx[i] = B[i] - O[i] % B[i], Ry[i] = (N[i] - Rx[i]) % B[i];
385 for(; i<H5CPP_MAX_RANK; i++) N[i] = B[i] = 1, Rx[i] = Ry[i] = 0;
386
387 // b - block size, j - block index pos, n - actual dimension of data
388 // rx - remainder at the leading edges, ry - remainder at trailing edges
389 h5cpp_def(0) h5cpp_def(1) h5cpp_def(2) h5cpp_def(3) h5cpp_def(4) h5cpp_def(5) h5cpp_def(6)
390
391 // rank-1 fast path: single large memcpy per chunk, no nested loops
392 if (rank == 1 && (O[0] % B[0]) == 0) [[likely]] {
393 constexpr hsize_t prefetch_distance = 4;
394 for (hsize_t j = 0; j < N[0]; j += B[0]) {
395#ifndef _WIN32
396 if (j + (prefetch_distance + 1) * B[0] < N[0])
397 __builtin_prefetch(ptr + (j + prefetch_distance * B[0]) * element_size, 0, 3);
398#endif
399 hsize_t bytes_in_chunk = (j + B[0] <= N[0]) ? B[0] : (N[0] - j);
400 hsize_t bytes_to_copy = bytes_in_chunk * element_size;
401 if (bytes_to_copy < block_size) [[unlikely]]
402 simd_memset_zero(chunk0, block_size);
403 if (tail == 0)
404 std::memcpy(chunk0, ptr + j * element_size, bytes_to_copy);
405 else
406 memcpy(chunk0, ptr + j * element_size, bytes_to_copy);
407 C[0] = j + O[0];
408 write_chunk(C, block_size, chunk0);
409 }
410 return;
411 }
