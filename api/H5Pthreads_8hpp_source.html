7// FAPL-scoped worker pool — opt-in via h5::threads{N}.
8//
9// User-facing API:
10//
11// h5::fd_t fd = h5::create("data.h5", H5F_ACC_TRUNC, h5::threads{8});
12// h5::fd_t fd = h5::create("data.h5", H5F_ACC_TRUNC, h5::threads{}); // hw_concurrency
13//
14// Per-dataset opt-in is the orthogonal h5::high_throughput DAPL flag
15// (existing). This FAPL property controls "is there a pool, how many
16// workers"; whether any given dataset uses it is the DAPL's call.
17//
18// Storage mechanism mirrors h5::high_throughput (H5Pdapl.hpp):
19//
20// - H5Pinsert2 stores a pointer to a worker_pool_slot_t in the FAPL
21// skip list.
22// - The slot owns a std::shared_ptr<worker_pool_t>.
23// - HDF5 internally copies the FAPL during H5Fopen/H5Fcreate; the
24// copy callback allocates a fresh slot aliasing the same pool
25// (shared_ptr refcount++). Every FAPL copy shares the pool.
26// - The close callback drops the slot. Pool is destroyed when the
27// last live FAPL copy releases its slot, at which point worker
28// std::jthreads receive request_stop() and join cleanly.
29//
30// This is the shared-ownership variant of the H5Pinsert2 + slot pattern.
31// Compare with H5Pdapl.hpp's fresh-allocation-per-copy semantics used by
32// the high_throughput pipeline property: that pattern allocates a fresh
33// pipeline scratch buffer per copy because pipelines are per-write
34// scratch state; this pattern shares one live resource across all
35// copies because workers ARE the resource we want shared. See
36// tasks/h5cpp-fapl-multithreading-workplan.md §2-§3.
37//
38// PHASE 1.1 STATUS: lifecycle scaffolding only. worker_pool_t owns N
39// std::jthreads whose only job today is to honor std::stop_token on
40// shutdown. Phase 1.2 extends with bounded MPMC queues, compress_sync /
41// compress_async API, and integration with the filter pipeline. Phase
42// 1.3 wires pt_t / h5::write / h5::read consumer sites.
43

62// Default in-flight chunk cap when h5::threads{N} is set without an
63// accompanying h5::backpressure{N}. Resolves to 8 × worker_count, which
64// is loose enough to keep all workers fed during steady-state streaming
65// and tight enough to bound memory growth (chunk_size × cap bytes per pt_t).
66// Override at compile time via -DH5CPP_FAPL_BACKPRESSURE_DEFAULT_FACTOR=K.
67#ifndef H5CPP_FAPL_BACKPRESSURE_DEFAULT_FACTOR
68#define H5CPP_FAPL_BACKPRESSURE_DEFAULT_FACTOR 8u
69#endif

73// ─── Pool ────────────────────────────────────────────────────────────────────
74//
75// Generic compute pool: workers pull type-erased tasks off a single MPMC-ish
76// queue (std::mutex + std::queue + doorbell signal) and execute them. The
77// submit<F>(callable) API returns a std::future of the callable's result type,
78// using std::packaged_task wrapped in a std::function<void()> for storage.
79//
80// The pool is deliberately HDF5-agnostic at this layer. Consumer sites
81// (Phase 1.3 — pt_t, h5::write, h5::read) wrap their HDF5-specific compress
82// logic in a closure and submit() it. This keeps the pool reusable for
83// Phase II's executor and any future async work.
84struct worker_pool_t {
85 // Pool size is fixed at construction; cannot resize at runtime.
86 // n == 0 means "use std::thread::hardware_concurrency()".
87 explicit worker_pool_t(unsigned n) {
88 const unsigned count = n ? n : std::max(1u, std::thread::hardware_concurrency());
89 workers_.reserve(count);
90 for (unsigned i = 0; i < count; ++i)
91 workers_.emplace_back([this](h5::detail::stop_token_t st) { worker_loop(st); });
92 }
93
94 // Destruction sequence:
95 // 1. wait_idle() — let all submitted tasks complete (caller can also
96 // have called this earlier).
97 // 2. Set stopping_ flag and ring the doorbell so all waiters wake.
98 // 3. stoppable_thread_t destructors automatically request stop + join.
99 ~worker_pool_t() {
100 wait_idle();
101 stopping_.store(true, std::memory_order_release);
102 bell_.ring_all();
103 }
104
105 worker_pool_t(const worker_pool_t&) = delete;
106 worker_pool_t& operator=(const worker_pool_t&) = delete;
107 worker_pool_t(worker_pool_t&&) = delete;
108 worker_pool_t& operator=(worker_pool_t&&) = delete;
109
110 // Introspection for tests and scheduling decisions.
111 [[nodiscard]] unsigned worker_count() const noexcept {
112 return static_cast<unsigned>(workers_.size());
113 }
114
115 // Submit a callable for execution by any worker. Returns a future of
116 // the callable's result type. Callable must be invocable with no
117 // arguments; capture state via closure if needed.
118 //
119 // Internally uses std::packaged_task to bridge the move-only callable
120 // into a copyable std::function (wrapped in shared_ptr) so it can sit
121 // in the queue.
122 template <typename Fn>
123 auto submit(Fn&& fn) -> std::future<std::invoke_result_t<Fn>> {
124 using R = std::invoke_result_t<Fn>;
125 auto task = std::make_shared<std::packaged_task<R()>>(std::forward<Fn>(fn));
126 auto fut = task->get_future();
127 in_flight_.fetch_add(1, std::memory_order_release);
128 {
129 std::lock_guard<std::mutex> lk(m_);
130 tasks_.emplace([task] { (*task)(); });
131 }
132 bell_.ring();
133 return fut;
134 }
135
136 // Block until all submitted tasks have completed. Intended for clean
137 // shutdown or as a synchronization barrier between submission phases.
138 void wait_idle() {
139 while (in_flight_.load(std::memory_order_acquire) > 0)
140 std::this_thread::yield();
141 }
142
143 // Coarse approximation of pending+running work. Useful for tests and
144 // monitoring; not a strict ordering guarantee.
145 [[nodiscard]] int in_flight() const noexcept {
146 return in_flight_.load(std::memory_order_acquire);
147 }
148
149private:
150 void worker_loop(h5::detail::stop_token_t st) {
151 while (!st.stop_requested()) {
152 std::function<void()> task;
153 std::uint32_t last_seq = 0;
154 bool got_task = false;
155 {
156 std::lock_guard<std::mutex> lk(m_);
157 if (!tasks_.empty()) {
158 task = std::move(tasks_.front());
159 tasks_.pop();
160 got_task = true;
161 } else {
162 // Snapshot the doorbell sequence WHILE HOLDING the lock.
163 // submit() will ring the bell only AFTER it has released
164 // the same lock, so any subsequent ring advances past
165 // last_seq — bell_.wait(last_seq) below cannot miss it.
166 last_seq = bell_.load();
167 }
168 }
169
170 if (got_task) {
171 try { task(); } catch (...) { /* packaged_task captures it */ }
172 in_flight_.fetch_sub(1, std::memory_order_release);
173 continue;
174 }
175
176 // Queue was empty — wait for a ring or for shutdown.
177 if (st.stop_requested() ||
178 stopping_.load(std::memory_order_acquire)) return;
179 bell_.wait(last_seq);
180 }
181 }
182
183 std::mutex m_;
184 std::queue<std::function<void()>> tasks_;
185 h5::detail::doorbell_t bell_;
186 std::atomic<int> in_flight_{0};
187 std::atomic<bool> stopping_{false};
188 std::vector<h5::detail::stoppable_thread_t> workers_;
189};

191// ─── FAPL slot + lifecycle callbacks ─────────────────────────────────────────
192
193// The heap-allocated holder whose pointer lives in the FAPL skip-list
194// value slot. Indirection is necessary because H5Pinsert2 stores raw
195// bytes — it cannot run shared_ptr's constructor/destructor for us.
196struct worker_pool_slot_t {
197 std::shared_ptr<worker_pool_t> pool;
198};
199
200// Copy callback: HDF5 cloned the property bytes (the slot pointer was
201// memcpy'd into the destination's value slot). Allocate a NEW slot whose
202// shared_ptr aliases the same pool — refcount++ via shared_ptr copy.
203//
204// This is the contract that makes "every FAPL copy shares one pool" work.
205inline herr_t fapl_pool_copy_cb(const char* /*name*/, size_t /*size*/, void* value) {
206 auto** slot_loc = static_cast<worker_pool_slot_t**>(value);
207 *slot_loc = new worker_pool_slot_t{(*slot_loc)->pool};
208 return 0;
209}
210
211// Close callback: drop one slot. shared_ptr inside the slot releases its
212// reference; worker_pool_t destructor runs when the last slot is freed
213// (refcount reaches 0), which stops and joins the jthreads.
214inline herr_t fapl_pool_close_cb(const char* /*name*/, size_t /*size*/, void* ptr) {
215 delete *static_cast<worker_pool_slot_t**>(ptr);
216 return 0;
217}

219// Setter invoked when the user applies h5::threads{N} to an FAPL.
220// Idempotent — if a pool property is already installed, leaves it untouched.
221//
222// n == 0 maps to std::thread::hardware_concurrency() inside worker_pool_t.
223inline herr_t fapl_threads_set(::hid_t fapl, unsigned n) {
224 if (H5Pexist(fapl, H5CPP_FAPL_WORKER_POOL)) return 0;
225 auto* slot = new worker_pool_slot_t{
226 std::make_shared<worker_pool_t>(n)
227 };
228 return H5Pinsert2(fapl, H5CPP_FAPL_WORKER_POOL,
229 sizeof(worker_pool_slot_t*), &slot,
230 nullptr, // set
231 nullptr, // get
232 nullptr, // prp_del
233 fapl_pool_copy_cb,
234 nullptr, // compare
235 fapl_pool_close_cb);
236}

238// Consumer-site helper: given a FAPL id, retrieve the worker pool shared_ptr
239// if one is installed. Returns nullptr (= no pool, fall back to synchronous
240// pipeline) if the property is absent. Used by pt_t, h5::write, h5::read
241// in Phase 1.3.
242inline std::shared_ptr<worker_pool_t> resolve_worker_pool(::hid_t fapl_id) noexcept {
243 if (fapl_id < 0 || H5Iis_valid(fapl_id) <= 0) return nullptr;
244 if (!H5Pexist(fapl_id, H5CPP_FAPL_WORKER_POOL)) return nullptr;
245 worker_pool_slot_t* slot = nullptr;
246 H5Pget(fapl_id, H5CPP_FAPL_WORKER_POOL, &slot);
247 return slot ? slot->pool : nullptr;
248}

250// ─── Back-pressure cap (separate FAPL property) ──────────────────────────────
251
252// Setter invoked when the user applies h5::backpressure{N} to an FAPL.
253// Stores a plain unsigned by memcpy semantics — no lifecycle callbacks
254// needed since the value is trivially copyable and owns no heap.
255//
256// The cap is consumed by pt_t (and Phase 1.3's h5::write/read) when
257// queueing work to the pool: write_chunk blocks on drain_completed
258// once the in-flight deque reaches the cap.
259inline herr_t fapl_backpressure_set(::hid_t fapl, unsigned cap) {
260 if (H5Pexist(fapl, H5CPP_FAPL_BACKPRESSURE)) return 0;
261 return H5Pinsert2(fapl, H5CPP_FAPL_BACKPRESSURE,
262 sizeof(unsigned), &cap,
263 nullptr, nullptr, nullptr,
264 nullptr, // copy: memcpy is correct for POD
265 nullptr,
266 nullptr); // close: nothing to release
267}
268
269// Consumer-site helper: returns the user-set back-pressure cap, or computes
270// the default (H5CPP_FAPL_BACKPRESSURE_DEFAULT_FACTOR × worker_count) when
271// no h5::backpressure{N} was applied. Returns 0 only when no pool is
272// installed either — caller should already have bailed in that case.
273inline unsigned resolve_backpressure(::hid_t fapl_id,
274 unsigned worker_count) noexcept {
275 if (fapl_id < 0 || H5Iis_valid(fapl_id) <= 0) return 0;
276 if (H5Pexist(fapl_id, H5CPP_FAPL_BACKPRESSURE)) {
277 unsigned cap = 0;
278 H5Pget(fapl_id, H5CPP_FAPL_BACKPRESSURE, &cap);
279 if (cap > 0) return cap;
280 }
281 return H5CPP_FAPL_BACKPRESSURE_DEFAULT_FACTOR * worker_count;
282}
