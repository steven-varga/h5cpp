19 struct reference_t {
20#if H5_VERSION_GE(1,12,0)
21 H5R_ref_t value{}; //< HDF5 1.12+ generic reference storage; zero-init means "empty"
22
23 // RAII: every successful H5Rcreate_*/H5Rcopy must be matched with H5Rdestroy.
24 // Without these, std::vector<reference_t> and bare scalar references leak the
25 // internal handle each H5R_ref_t carries, and HDF5 hits its "infinite loop
26 // closing library" path at shutdown (the leaked handles keep parents alive).
27 // The opaque H5R_ref_t blob remains memcpy-safe for HDF5's on-disk serializer
28 // (H5Dread/H5Dwrite go through H5T_STD_REF), which is what `is_trivially_packable`
29 // in access_traits_t<reference_t> asserts. The rule-of-five below governs only
30 // *in-memory* C++ copy/move; the disk path is unaffected.
31 reference_t() noexcept = default;
32 ~reference_t() {
33 if (!is_empty()) H5Rdestroy(&value);
34 }
35 reference_t(const reference_t& other) {
36 if (!other.is_empty()) H5Rcopy(&other.value, &value);
37 }
38 reference_t& operator=(const reference_t& other) {
39 if (this != &other) {
40 if (!is_empty()) H5Rdestroy(&value);
41 std::memset(&value, 0, sizeof(value));
42 if (!other.is_empty()) H5Rcopy(&other.value, &value);
43 }
44 return *this;
45 }
46 reference_t(reference_t&& other) noexcept {
47 std::memcpy(&value, &other.value, sizeof(value));
48 std::memset(&other.value, 0, sizeof(other.value));
49 }
50 reference_t& operator=(reference_t&& other) noexcept {
51 if (this != &other) {
52 if (!is_empty()) H5Rdestroy(&value);
53 std::memcpy(&value, &other.value, sizeof(value));
54 std::memset(&other.value, 0, sizeof(other.value));
55 }
56 return *this;
57 }
58 private:
59 bool is_empty() const noexcept {
60 // A value-initialized H5R_ref_t is all-zero bytes; H5Rdestroy on it
61 // would log an HDF5 error. memcmp against a static zero blob is the
62 // cheapest portable check.
63 static constexpr H5R_ref_t zero{};
64 return std::memcmp(&value, &zero, sizeof(value)) == 0;
65 }
66 public:
67#else
68 hdset_reg_ref_t value{}; //< region or object ref storage (pre-1.12 POD, no handle)
69#endif
70 };
71#if H5_VERSION_GE(1,12,0)
72 static_assert(sizeof(reference_t) == sizeof(H5R_ref_t), "reference_t must match HDF5 reference storage");
73 static_assert(std::is_nothrow_move_constructible<reference_t>::value,
74 "reference_t move must be noexcept so std::vector reallocation uses move, not copy");
75#else
76 static_assert(sizeof(reference_t) == sizeof(hdset_reg_ref_t), "reference_t must match HDF5 dataset-region reference storage");
77#endif
78}

136/* template specialization is for the preceding class, and should be used only for HDF5 ELEMENT types
137 * which are in C/C++ the integral types of: char,short,int,long, ... and C POD types.
138 * anything else, the ones which are considered objects/classes are broken down into integral types + container
139 * then pointer read|write is obtained to the continuous slab and delegated to h5::read | h5::write.
140 * IF the data is not in a continuous memory region then it must be copied!
141 */

143#define H5CPP_REGISTER_TYPE_( C_TYPE, H5_TYPE ) \
144namespace h5::impl::detail { \
145 template <> struct hid_t<C_TYPE,H5Tclose,true,true,hdf5::type> : public dt_p<C_TYPE> {\
146 using parent = dt_p<C_TYPE>; \
147 using dt_p<C_TYPE>::hid_t; \
148 using hidtype = C_TYPE; \
149 hid_t() : parent( H5Tcopy( H5_TYPE ) ) { \
150 hid_t id = static_cast<hid_t>( *this ); \
151 if constexpr ( std::is_pointer_v<C_TYPE> ) \
152 H5Tset_size (id,H5T_VARIABLE), H5Tset_cset(id, H5T_CSET_UTF8); \
153 } \
154 }; \
155} \
156namespace h5 { \
157 template <> struct name<C_TYPE> { \
158 static constexpr char const * value = #C_TYPE; \
159 }; \
160} \
161namespace h5::meta { \
162 /* W4 (review item A7): every type registered through this macro — including all \
163 * H5CPP_REGISTER_STRUCT(Foo) expansions — marks itself as a known compound, which \
164 * lets the aggregate storage_representation_impl fallback accept it. Unregistered \
165 * aggregates remain storage=unsupported and are caught at compile time. */ \
166 template <> struct has_registered_compound<C_TYPE> : std::true_type {}; \
167} \

185 // dt_t<std::string> = H5T_C_S1 with H5T_VARIABLE size. Without this spec the
186 // dt_t primary template returns H5I_UNINIT for std::string fields, causing
187 // H5Tinsert("...", offset, UNINIT) to fail (and the dt_t destructor calls
188 // H5Tclose(UNINIT) which corrupts the allocator). Surfaces when a struct or
189 // std::tuple<...,std::string,...> field is built via the compound dispatch.
190 namespace h5::impl::detail {
191 // Note: the H5CPP_REGISTER_TYPE_ macro uses `using dt_p<C>::hid_t;` to
192 // inherit the parent's constructors. The h5cpp-compiler's clang AST
193 // scanner (Clang 10 backed) parses that inheriting-ctor declaration
194 // fine on non-templated specialisations but rejects it inside this
195 // templated partial specialisation with "member 'hid_t' has the same
196 // name as its class". Write the forwarding ctors explicitly to keep
197 // both the compiler-tool scan and the regular C++ build happy.
198 template <class CharT, class Tr, class A>
199 struct hid_t<std::basic_string<CharT, Tr, A>, H5Tclose, true, true, hdf5::type>
200 : public dt_p<std::basic_string<CharT, Tr, A>> {
201 using parent = dt_p<std::basic_string<CharT, Tr, A>>;
202 using hidtype = std::basic_string<CharT, Tr, A>;
203 hid_t( std::initializer_list<::hid_t> fd ) : parent( fd ) {}
204 hid_t() : parent( H5Tcopy(H5T_C_S1) ) {
205 ::hid_t id = static_cast<::hid_t>(*this);
206 H5Tset_size(id, H5T_VARIABLE);
207 H5Tset_cset(id, H5T_CSET_UTF8);
208 }
209 };
210 }

653// Dispatch wrapper: prefers new type engine when supported, falls back to dt_t<T> for
654// unregistered compound types registered via H5CPP_REGISTER_STRUCT.
655template <class T>
656class resolved_type_t {
657 ::hid_t id_;
658 bool owns_;
659 std::optional<h5::dt_t<T>> fallback_;
660public:
661 resolved_type_t() noexcept {
662 if constexpr (storage_traits_t<T>::supported) {
663 id_ = storage_traits_t<T>::create_type();
664 owns_ = storage_traits_t<T>::owns_handle;
665 } else {
666 fallback_.emplace();
667 id_ = static_cast<::hid_t>(*fallback_);
668 owns_ = false;
669 }
670 }
671 ~resolved_type_t() noexcept {
672 if (owns_ && H5Iis_valid(id_)) H5Tclose(id_);
673 }
674 operator ::hid_t() const noexcept { return id_; }
675 // Allows H5Dcreate's custom dt_t override path: type = static_cast<hid_t>(custom_dt)
676 resolved_type_t& operator=(::hid_t id) noexcept {
677 if (owns_ && H5Iis_valid(id_)) H5Tclose(id_);
678 fallback_.reset();
679 id_ = id;
680 owns_ = H5Iis_valid(id_);
681 return *this;
682 }
683 resolved_type_t(const resolved_type_t&) = delete;
684 resolved_type_t& operator=(const resolved_type_t&) = delete;
685};
