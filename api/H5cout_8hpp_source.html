15inline std::ostream& operator<< (std::ostream& os, const h5::dxpl_t& dxpl) {
16 os <<"handle: " << static_cast<hid_t>( dxpl );
17#ifdef H5_HAVE_PARALLEL
18 H5D_mpio_actual_io_mode_t io_mode;
19 H5Pget_mpio_actual_io_mode( static_cast<hid_t>(dxpl), &io_mode);
20
21 switch( io_mode ){
22 case H5D_MPIO_NO_COLLECTIVE:
23 os << "No collective I/O was performed. Collective I/O was not requested or collective I/O isn't possible on this dataset.";
24 break;
25 case H5D_MPIO_CHUNK_INDEPENDENT:
26 os << "HDF5 performed one the chunk collective optimization schemes and each chunk was accessed independently.";
27 break;
28 case H5D_MPIO_CHUNK_COLLECTIVE:
29 os << "HDF5 performed one the chunk collective optimization schemes and each chunk was accessed collectively";
30 break;
31 case H5D_MPIO_CHUNK_MIXED:
32 os <<"HDF5 performed one the chunk collective optimization schemes and some chunks were accessed independently, some collectively";
33 break;
34 case H5D_MPIO_CONTIGUOUS_COLLECTIVE:
35 os <<"Collective I/O was performed on a contiguous dataset.";
36 break;
37 }
38#endif
39 return os;
40}

43std::ostream& operator<<(std::ostream& os, const h5::impl::array<T>& arr){
44 os << "{";
45 if( arr.rank )
46 // rank > 0 such as: vector,matrix,cube,...

48 char sep = i != arr.rank - 1 ? ',' : '}';

60std::ostream& operator<<(std::ostream &os, const h5::sp_t& sp) {
61 //htri_t H5Sis_regular_hyperslab( hid_t space_id ) 1.10.0
62 //herr_t H5Sget_select_bounds(hid_t space_id, hsize_t *start, hsize_t *end )
63 // hssize_t H5Sget_select_npoints( hid_t space_id )
64 hid_t id = static_cast<hid_t>( sp );
65 #if H5_VERSION_GE(1,10,0)
66
67 #endif
68 h5::mute();
69
70 h5::offset_t start,end;
71 h5::current_dims_t current_dims;
72 h5::max_dims_t max_dims;
73 unsigned rank = h5::get_simple_extent_dims( sp, current_dims, max_dims);
74 hsize_t total_elements = H5Sget_simple_extent_npoints( id );
75 H5Sget_select_bounds(id, *start, *end);
76 start.rank = end.rank = rank;
77 hssize_t nblocks = H5Sget_select_hyper_nblocks( id );
78 hssize_t nelements = H5Sget_select_elem_npoints( id );
79
80 // if slection valid
81 std::string is_valid;
82 htri_t is_valid_ = H5Sselect_valid(id);
83 if(is_valid_ > 0 ) is_valid = "within extent";
84 if(is_valid_ == 0 ) is_valid = "not in extent";
85 if(is_valid_ < 0 ) is_valid = "error occured";
86
87 os << "[rank]\t" << rank << "\t[total elements]\t" << total_elements << std::endl;
88 os << "[dimensions]\tcurrent: " << current_dims << "\tmaximum: " << max_dims << std::endl;
89 os << "[selection]\tstart: " << start << "\tend:" << end << std::endl;
90 os << "[selection]\t" << is_valid <<std::endl;
91 // nblocks is -1 when no hyperslab is selected; guard to avoid calloc overflow
92 if( nblocks > 0 ){
93 hsize_t ncoordinates = static_cast<hsize_t>(2 * rank) * static_cast<hsize_t>(nblocks);
94 h5::impl::unique_ptr<hsize_t> buffer{
95 static_cast<hsize_t*>( std::calloc( ncoordinates, sizeof(hsize_t))) };
96 if( H5Sget_select_hyper_blocklist(id, 0, static_cast<hsize_t>(nblocks), buffer.get() ) >= 0 ){
97 os << "[selected element count]\t" << nelements << std::endl;
98 os << "[selected block count]\t" << nblocks <<std::endl;
99 os << "[selected blocks]\t";
100 for( hsize_t i=0; i<static_cast<hsize_t>(nblocks); i++){
101 os << "[{";
102 for(hsize_t j=0; j<rank; j++) os << *( buffer.get() + i*2*rank+j ) << (j < rank-1 ? "," : "}{");
103 for(hsize_t j=rank; j<2*rank; j++) os << *( buffer.get() + i*2*rank+j ) << ( j < 2*rank-1 ? "," : "}");
104 os << "] ";
105 }
106 }
107 }
108 h5::unmute();
109 return os;
110}

120// =============================================================================
121// Handle pretty-printers — generic + selective specializations
122// =============================================================================
123//
124// Every h5cpp handle (fd_t / ds_t / gr_t / at_t / sp_t / 14 property-list
125// variants / 5 async variants) is an alias of h5::impl::detail::hid_t<...>.
126// The generic operator<< below covers them all uniformly: handle id, RAII
127// validity, refcount. Higher-information specializations follow for the
128// six handles where deep introspection is most valuable (fd_t, ds_t, gr_t,
129// at_t, dcpl_t, fapl_t).
130//
131// All printers bracket their HDF5 introspection in h5::mute()/unmute() so
132// invalid or H5I_UNINIT handles don't dump error stacks into the output.
133//
134// Note: these printers call H5I* / H5P* / H5D* introspection APIs and are
135// not free — fine for debug logs, avoid inside hot loops.
136// =============================================================================

175 // Generic printer — works for both classic (true,true) and async
176 // (false,false) variants because both expose the raw `handle` field.
177 template <class T, capi_close_t F, bool From, bool To, int K>
178 inline std::ostream& operator<<(std::ostream& os, const hid_t<T,F,From,To,K>& h) {
179 h5::mute();
180 // Classic (true,true) variants have `handle` protected — use the
181 // explicit converting operator. Async (false,false) variants delete
182 // that operator but expose `handle` publicly.
183 ::hid_t id;
184 if constexpr (From && To) id = static_cast<::hid_t>(h);
185 else id = h.handle;
186 os << "h5::" << class_tag<T>() << "{ handle=" << id;
187 if (id != H5I_UNINIT && H5Iis_valid(id) > 0) {
188 os << " valid=yes refs=" << H5Iget_ref(id) << " }";
189 } else {
190 os << " valid=no }";
191 }
192 h5::unmute();
193 return os;
194 }
195}

203 // h5::fd_t — file path + open mode + libver bounds + file size.
204 inline std::ostream& operator<<(std::ostream& os, const h5::fd_t& fd) {
205 h5::mute();
206 const ::hid_t id = static_cast<::hid_t>(fd);
207 os << "h5::fd_t{ handle=" << id;
208 if (id == H5I_UNINIT || H5Iis_valid(id) <= 0) { os << " valid=no }"; h5::unmute(); return os; }
209
210 // Path on disk: H5Fget_name needs sizing first.
211 ssize_t name_len = H5Fget_name(id, nullptr, 0);
212 std::string path(name_len > 0 ? static_cast<size_t>(name_len) : 0, '\0');
213 if (name_len > 0) H5Fget_name(id, path.data(), name_len + 1);
214 os << " path='" << path << "'";
215
216 // Open intent (read-only vs read-write).
217 unsigned intent = 0;
218 if (H5Fget_intent(id, &intent) >= 0)
219 os << " mode=" << ((intent & H5F_ACC_RDWR) ? "RDWR" : "RDONLY");
220
221 // libver bounds from the file's FAPL.
222 ::hid_t fapl = H5Fget_access_plist(id);
223 if (fapl >= 0) {
224 H5F_libver_t low = H5F_LIBVER_EARLIEST, high = H5F_LIBVER_EARLIEST;
225 if (H5Pget_libver_bounds(fapl, &low, &high) >= 0)
226 os << " libver=[" << static_cast<int>(low) << "," << static_cast<int>(high) << "]";
227 H5Pclose(fapl);
228 }
229
230 // File size in bytes.
231 hsize_t fsize = 0;
232 if (H5Fget_filesize(id, &fsize) >= 0)
233 os << " size=" << fsize;
234
235 os << " }";
236 h5::unmute();
237 return os;
238 }

240 // h5::ds_t — dataset path + class + rank + dims + chunk dims if chunked + filter count.
241 inline std::ostream& operator<<(std::ostream& os, const h5::ds_t& ds) {
242 h5::mute();
243 const ::hid_t id = static_cast<::hid_t>(ds);
244 os << "h5::ds_t{ handle=" << id;
245 if (id == H5I_UNINIT || H5Iis_valid(id) <= 0) { os << " valid=no }"; h5::unmute(); return os; }
246
247 // Object path (may be empty if anonymous / not yet linked).
248 ssize_t name_len = H5Iget_name(id, nullptr, 0);
249 std::string path(name_len > 0 ? static_cast<size_t>(name_len) : 0, '\0');
250 if (name_len > 0) H5Iget_name(id, path.data(), name_len + 1);
251 os << " path='" << path << "'";
252
253 // Datatype class + size.
254 ::hid_t tid = H5Dget_type(id);
255 if (tid >= 0) {
256 static const char* tcls[] = {"INTEGER","FLOAT","TIME","STRING","BITFIELD","OPAQUE",
257 "COMPOUND","REFERENCE","ENUM","VLEN","ARRAY"};
258 H5T_class_t c = H5Tget_class(tid);
259 if (c >= 0 && c < 11) os << " dtype=" << tcls[c];
260 os << " elsize=" << H5Tget_size(tid);
261 H5Tclose(tid);
262 }
263
264 // Dataspace dims.
265 ::hid_t sid = H5Dget_space(id);
266 if (sid >= 0) {
267 int rank = H5Sget_simple_extent_ndims(sid);
268 os << " rank=" << rank;
269 if (rank > 0 && rank <= H5S_MAX_RANK) {
270 std::vector<hsize_t> dims(rank);
271 H5Sget_simple_extent_dims(sid, dims.data(), nullptr);
272 os << " dims={";
273 for (int i = 0; i < rank; ++i) os << dims[i] << (i+1<rank ? "," : "");
274 os << "}";
275 }
276 H5Sclose(sid);
277 }
278
279 // Chunk dims + filter count when chunked.
280 ::hid_t dcpl = H5Dget_create_plist(id);
281 if (dcpl >= 0) {
282 H5D_layout_t layout = H5Pget_layout(dcpl);
283 static const char* lyt[] = {"COMPACT","CONTIGUOUS","CHUNKED","VIRTUAL"};
284 if (layout >= 0 && layout < 4) os << " layout=" << lyt[layout];
285 if (layout == H5D_CHUNKED) {
286 int crank = H5Pget_chunk(dcpl, 0, nullptr);
287 if (crank > 0) {
288 std::vector<hsize_t> cdims(crank);
289 H5Pget_chunk(dcpl, crank, cdims.data());
290 os << " chunk={";
291 for (int i = 0; i < crank; ++i) os << cdims[i] << (i+1<crank ? "," : "");
292 os << "}";
293 }
294 }
295 int nfilters = H5Pget_nfilters(dcpl);
296 if (nfilters >= 0) os << " filters=" << nfilters;
297 H5Pclose(dcpl);
298 }
299
300 // Attribute count is intentionally omitted — H5Oget_info* has
301 // a moving API surface across HDF5 1.12.x patch releases (info2/info3
302 // gated by version-and-compat-macros mix), too brittle for a
303 // pretty-printer. Use h5ls if you need it.
304
305 os << " }";
306 h5::unmute();
307 return os;
308 }

382 }

384 H5D_alloc_time_t at = H5D_ALLOC_TIME_DEFAULT;
385 if (H5Pget_alloc_time(id, &at) >= 0) {
386 static const char* a[] = {"DEFAULT","EARLY","LATE","INCR"};

401 os << " filters=[";
