73 template <class T, class... args_t>
74 inline h5::ds_t write( const h5::ds_t& ds, const T* ptr, args_t&&... args ) try {
75 // element types: pod | [signed|unsigned](int8 | int16 | int32 | int64) | float | double
76 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
77 using toffset = typename arg::tpos<const h5::offset_t&, const args_t&...>;
78 using tstride = typename arg::tpos<const h5::stride_t&, const args_t&...>;
79 using tblock = typename arg::tpos<const h5::block_t&, const args_t&...>;
80 static_assert( tcount::present,"h5::count_t{ ... } must be provided to describe T* memory region" );
81 //static_assert( utils::is_supported<T>, "error: " H5CPP_supported_elementary_types );
82
83 auto tuple = std::forward_as_tuple(args...);
84 h5::count_t count = std::get<tcount::value>( tuple ); //we make a copy of it
85 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...); // gets reference to default value if property not present
86 h5::sp_t file_space{H5Dget_space( static_cast<::hid_t>(ds) )};
87
88 int rank = h5::get_simple_extent_ndims( file_space );
89 hsize_t n_elements = static_cast<hsize_t>(count);
90 // Resolve threads{N} from the dataset's REAL access plist, not ds.dapl — that
91 // member caches the open-time hid_t, which dangles once a temporary dapl
92 // (h5::open(fd, name, h5::threads{N})) is destroyed. See H5Dread.hpp. #287.
93 hid_t dapl = H5Dget_access_plist( static_cast<hid_t>(ds) );
94 const unsigned dapl_threads = h5::impl::resolve_dataset_threads(dapl);
95 const unsigned dapl_cap = dapl_threads
96 ? h5::impl::resolve_dataset_backpressure(dapl, dapl_threads) : 0u;
97 if (dapl >= 0) H5Pclose(dapl);
98 herr_t err = 0;
99
100 // ── Write-path dispatch (3 options) ──────────────────────────────────
101 // 1. CAPI hyperslab — h5::offset/stride/block present (COMPILE-TIME),
102 // or a non-chunked dataset: HDF5's own chunk processor + filters.
103 // 2. direct chunk (DEFAULT) — no hyperslab, chunked, no h5::threads{N}:
104 // basic_pipeline_t → H5Dwrite_chunk with h5cpp's filter chain.
105 // 3. parallel — no hyperslab, chunked, h5::threads{N} on the dataset's
106 // DAPL: pool_pipeline_t fans compression across the global pool.
107 // Options 2/3 need H5D_CHUNKED (H5Dwrite_chunk); contiguous/compact falls
108 // to option 1 (also dodges the #242 MSVC SegFault). Whether a hyperslab
109 // is selected is a COMPILE-TIME fact (tag presence), so a hyperslab write
110 // can never reach the pool — the exclusion is structural, no runtime guard.
111 constexpr bool hyperslab = toffset::present || tstride::present || tblock::present;
112 bool direct_chunk = false;
113 if constexpr (!hyperslab) {
114 hid_t dcpl_id = H5Dget_create_plist(static_cast<hid_t>(ds));
115 if (dcpl_id >= 0) {
116 if (H5Pget_layout(dcpl_id) == H5D_CHUNKED) {
117 // Options 2/3 apply the filter chain in-process. NBIT and
118 // SCALEOFFSET are HDF5-internal transforms h5cpp passes through
119 // (it relies on the C library to apply them); direct-chunk would
120 // write them UNFILTERED, so fall back to the CAPI path (option 1).
121 direct_chunk = true;
122 int nf = H5Pget_nfilters(dcpl_id);
123 for (int fi = 0; fi < nf; ++fi) {
124 size_t ne = 0; unsigned fl = 0, cfg = 0;
125 H5Z_filter_t id = H5Pget_filter2(dcpl_id, static_cast<unsigned>(fi),
126 &fl, &ne, nullptr, 0, nullptr, &cfg);
127 if (id == H5Z_FILTER_NBIT || id == H5Z_FILTER_SCALEOFFSET) {
128 direct_chunk = false; break;
129 }
130 }
131 }
132 H5Pclose(dcpl_id);
133 }
134 }
135 if( direct_chunk ){
136 const h5::block_t& block = arg::get( h5::default_block, args...);
137 const h5::offset_t& offset = arg::get( h5::default_offset, args...);
138 const h5::stride_t& stride = arg::get( h5::default_stride, args...);
139 // A 1-D container written to an N-D chunked dataset arrives with a
140 // rank-1 count, but the buffer is a flat image of the full dataset —
141 // tile by the dataset's actual N-D dimensions so the chunk decomposition
142 // is correct (the stock CAPI path reshaped via the dataspace; the
143 // direct-chunk pipeline must be told the real dims).
144 h5::count_t eff_count = count;
145 if (count.rank < static_cast<unsigned>(rank)) {
146 hsize_t fd_dims[H5CPP_MAX_RANK];
147 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), fd_dims, nullptr);
148 eff_count.rank = static_cast<unsigned>(rank);
149 for (int i = 0; i < rank; i++) eff_count[i] = fd_dims[i];
150 }
151 // set_cache populates the filter chain from the dataset's DCPL — both
152 // pipelines need it before write() (the old DAPL pipeline had it
153 // cached at dataset-open; an inline one does not).
154 h5::dcpl_t dcpl{H5Dget_create_plist(static_cast<hid_t>(ds))};
155 hid_t type_id = H5Dget_type(static_cast<hid_t>(ds));
156 size_t elem_sz = H5Tget_size(type_id);
157 H5Tclose(type_id);
158 // threads{N} lives on the dataset's DAPL — resolved up front from the real
159 // access plist (no #286 registry).
160 if (dapl_threads > 0) { // OPTION 3 — parallel
161 h5::impl::pool_pipeline_t pipe(h5::impl::global_pool_ptr(), dapl_cap);
162 pipe.set_cache(dcpl, elem_sz);
163 pipe.write(ds, offset, stride, block, eff_count, dxpl, ptr);
164 // pipe destructor drains in_flight before returning.
165 } else { // OPTION 2 — direct chunk (default)
166 h5::impl::pipeline_t<impl::basic_pipeline_t> pipe;
167 pipe.set_cache(dcpl, elem_sz);
168 pipe.write(ds, offset, stride, block, eff_count, dxpl, ptr);
169 }
170 } else {
171 // Scalar dataspaces don't support hyperslab selection; H5Sselect_all
172 // on both sides is the equivalent path for H5S_SCALAR file spaces.
173 H5S_class_t file_cls = H5Sget_simple_extent_type(static_cast<hid_t>(file_space));
174 h5::sp_t mem_space = (file_cls == H5S_SCALAR)
175 ? h5::sp_t{H5Screate(H5S_SCALAR)}
176 : h5::create_simple( n_elements );
177 h5::select_all( mem_space );
178 if (file_cls == H5S_SCALAR) {
179 err = H5Sselect_all(static_cast<hid_t>(file_space));
180 } else if constexpr (toffset::present || tstride::present || tblock::present){
181 // HYPERBLOCK selection: we either have the argument in `args...` or using default values
182 const h5::block_t& block = arg::get( h5::default_block, args...);
183 const h5::offset_t& offset = arg::get( h5::default_offset, args...);
184 const h5::stride_t& stride = arg::get( h5::default_stride, args...);
185 if constexpr( tblock::present ){ // we have to normalise `count` such that `size[i] = count[i] * block[i]` holds
186 for(int i=0; i < rank; i++) count[i] /= block[i];
187 err = H5Sselect_hyperslab(static_cast<hid_t>(file_space), H5S_SELECT_SET, *offset, *stride, *count, *block);
188 } else { // we have to convert h5::count_t{..} to h5::block{..} and initiate a single block transfer
189 h5::block_t block_ = static_cast<h5::block_t>(count);
190 block_.rank = rank; // memory space may have different rank, be sure to use the rank of file_space
191 err = H5Sselect_hyperslab(static_cast<hid_t>(file_space), H5S_SELECT_SET, *offset, *stride, *h5::default_count, *block_);
192 }
193 } else // SELECT_ALL this is the fastest approach, mem_space and file_space must match
194 err = H5Sselect_all(static_cast<hid_t>(file_space));
195 // throw an exception if eny error
196 H5CPP_CHECK_NZ(err, h5::error::io::dataset::write, h5::error::msg::select_hyperslab);
197 // MSVC partial-ordering bug: the unqualified `::h5::write(ds, mem, file, dxpl, ptr)`
198 // is ambiguous between the inner overload at line 21 and the variadic forwarder at
199 // line 181 (with T=sp_t). Inline the H5Dwrite call here — matches the H5Dread.hpp:73
200 // idiom — so there is no recursive call back into this overload set.
201 using element_t = typename impl::decay<T>::type;
202 h5::meta::resolved_type_t<element_t> type;
203 H5CPP_CHECK_NZ(
204 H5Dwrite( static_cast<hid_t>(ds), static_cast<hid_t>(type),
205 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
206 static_cast<hid_t>(dxpl), ptr),
207 h5::error::io::dataset::write, h5::error::msg::write_dataset);
208 }
209 return ds;
210 } catch ( const std::exception& err ){
211 throw h5::error::io::dataset::write( err.what() );
212 }

268 // The SFINAE excludes raw pointers (they go through the T* overload) but
269 // carves out C-arrays — `T[N]` decays to a pointer, but we still want the
270 // by-reference path to handle them (e.g., char[N] fixed-length strings).
271 template <class T, class... args_t,
272 class = std::enable_if_t<!std::is_pointer_v<std::decay_t<T>> || std::is_array_v<T>>>
273 inline h5::ds_t write(const h5::ds_t& ds, const T& ref, args_t&&... args) try {
274 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
275 using element_t = typename impl::decay<T>::type;
276 using traits = h5::meta::access_traits_t<T>; // we classify type
277 using sr_t = h5::meta::storage_representation_t;
278
279 constexpr auto kind = traits::kind;
280 constexpr auto storage = h5::meta::storage_representation_v<T>;
281
282 // Stopper: scatter types must use the fd-gateway overload so the compiler-generated
283 // h5::scatter<T> specialization can dispatch. Calling h5::write(ds, ref) on a scatter
284 // type would silently bypass scatter and fall through to the aggregate path with
285 // register_struct<T>() == H5I_UNINIT.
286 static_assert(!h5::has_scatter<std::decay_t<T>>::value,
287 "h5::write(ds, ref): scatter types must use h5::write(fd, path, ref) so the "
288 "compiler-generated h5::scatter<T> specialization can dispatch.");
289
290 // Stopper: unsupported storage usually means an unregistered POD aggregate, a
291 // deeply-nested container, or std::vector<bool>. Without this the dispatch can
292 // fall through to H5Dwrite with H5I_UNINIT and produce a broken file or crash.
293 // Guarded for the h5cpp-compiler bootstrap pass: when generated.h is the empty
294 // stub, H5CPP_REGISTER_STRUCT hasn't fired yet, so has_registered_compound<T>
295 // is false and the storage falls through to 'unsupported'. The compiler sets
296 // -DH5CPP_BUILDING_TYPE_INFO so this check is skipped during its AST scan.
297#ifndef H5CPP_BUILDING_TYPE_INFO
298 static_assert(storage != sr_t::unsupported,
299 "h5::write: storage_representation_v<T> resolved to 'unsupported'. "
300 "Check: unregistered POD aggregate (use H5CPP_REGISTER_STRUCT), "
301 "std::vector<bool>, or container nesting beyond vector<vector<T>>/vector<string>.");
302#endif
303
304 // Stopper (borrowed from #274): containers of containers must route via VLEN
305 // storage. Iterator-staging and pointers-gather paths can't faithfully serialize
306 // nested containers. element_t comes from impl::decay<T>, which is SFINAE-safe
307 // for non-container T (returns T itself, so is_stl_like<element_t>=false).
308 static_assert(
309 !h5::meta::is_stl_like<element_t>::value ||
310 storage == sr_t::ragged_vlen_dataset ||
311 storage == sr_t::vlen_text_dataset ||
312 storage == sr_t::array_dataset ||
313 storage == sr_t::array_element ||
314 storage == sr_t::fls_dataset ||
315 storage == sr_t::fixed_inner_extent_dataset,
316 "h5::write: containers of containers are only supported for vector<string> "
317 "(vlen_text_dataset) or vector<vector<T>> (ragged_vlen_dataset).");
318
319 if constexpr (storage == sr_t::array_element) {
320 // Top-level T[N] / std::array<T,N> (non-char) — scalar dataspace +
321 // H5T_ARRAY[N] dt_t<T> element. The traits::size dims drive the
322 // array extent; traits::data points at the contiguous buffer.
323 using element_t_loc = typename traits::element_t;
324 auto dims = traits::size(ref);
325 hsize_t array_dims[H5CPP_MAX_RANK];
326 for (std::size_t i = 0; i < dims.size(); ++i) array_dims[i] = dims[i];
327 h5::meta::resolved_type_t<element_t_loc> base_type;
328 hid_t array_type = H5Tarray_create(static_cast<hid_t>(base_type),
329 static_cast<unsigned>(dims.size()), array_dims);
330 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
331 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
332 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
333 H5Sselect_all(static_cast<hid_t>(file_space));
334 H5CPP_CHECK_NZ(
335 H5Dwrite(static_cast<hid_t>(ds), array_type,
336 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
337 static_cast<hid_t>(dxpl), traits::data(ref)),
338 h5::error::io::dataset::write, h5::error::msg::write_dataset);
339 H5Tclose(array_type);
340 } else if constexpr (storage == sr_t::array_dataset) {
341 // Outer<std::array<T,N>> (non-char T) — rank-1 dataspace of
342 // H5T_ARRAY[N] dt_t<T> elements. For vector (which exposes
343 // .data()) we hand HDF5 the buffer directly; for list/set/etc.
344 // we copy elements into a contiguous scratch vector first.
345 using inner_t = std::remove_cv_t<typename std::remove_reference_t<T>::value_type>;
346 using elem_scalar = typename inner_t::value_type;
347 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
348 hsize_t array_dims[1] = { static_cast<hsize_t>(N_inner) };
349 h5::meta::resolved_type_t<elem_scalar> base_type;
350 hid_t array_type = H5Tarray_create(static_cast<hid_t>(base_type), 1, array_dims);
351 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
352 std::size_t outer_count = std::distance(std::begin(ref), std::end(ref));
353 std::vector<inner_t> scratch;
354 const void* ptr = nullptr;
355 if constexpr (h5::meta::has_data_pointer<std::remove_cv_t<std::remove_reference_t<T>>>::value) {
356 ptr = ref.empty() ? nullptr : ref.front().data();
357 } else {
358 scratch.assign(std::begin(ref), std::end(ref));
359 ptr = scratch.empty() ? nullptr : scratch.front().data();
360 }
361 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(outer_count));
362 h5::select_all(mem_space);
363 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
364 H5Sselect_all(static_cast<hid_t>(file_space));
365 H5CPP_CHECK_NZ(
366 H5Dwrite(static_cast<hid_t>(ds), array_type,
367 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), ptr),
368 h5::error::io::dataset::write, h5::error::msg::write_dataset);
369 H5Tclose(array_type);
370 } else if constexpr (storage == sr_t::fls_dataset) {
371 // Outer<std::array<char,N>> — rank-1 dataspace of fixed-length-
372 // string elements. Iterator-stages for non-vector outers.
373 using inner_t = std::remove_cv_t<typename std::remove_reference_t<T>::value_type>;
374 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
375 hid_t str_type = H5Tcopy(H5T_C_S1);
376 H5Tset_size(str_type, N_inner);
377 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
378 std::size_t outer_count = std::distance(std::begin(ref), std::end(ref));
379 std::vector<inner_t> scratch;
380 const void* ptr = nullptr;
381 if constexpr (h5::meta::has_data_pointer<std::remove_cv_t<std::remove_reference_t<T>>>::value) {
382 ptr = ref.empty() ? nullptr : ref.front().data();
383 } else {
384 scratch.assign(std::begin(ref), std::end(ref));
385 ptr = scratch.empty() ? nullptr : scratch.front().data();
386 }
387 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(outer_count));
388 h5::select_all(mem_space);
389 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
390 H5Sselect_all(static_cast<hid_t>(file_space));
391 H5CPP_CHECK_NZ(
392 H5Dwrite(static_cast<hid_t>(ds), str_type,
393 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), ptr),
394 h5::error::io::dataset::write, h5::error::msg::write_dataset);
395 H5Tclose(str_type);
396 } else if constexpr (kind == h5::meta::access_t::composite) {
397 // scalar composite (std::tuple<Ts...>): pack into buffer matching the
398 // HDF5 compound type's field offsets (dt_t<tuple> + tuple_layout agree).
399 std::vector<char> buf(traits::bytes());
400 traits::pack(ref, buf.data());
401 h5::meta::resolved_type_t<T> mem_type;
402 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
403 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
404 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
405 H5Sselect_all(static_cast<hid_t>(file_space));
406 H5CPP_CHECK_NZ(
407 H5Dwrite(static_cast<hid_t>(ds), static_cast<hid_t>(mem_type),
408 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
409 static_cast<hid_t>(dxpl), buf.data()),
410 h5::error::io::dataset::write, h5::error::msg::write_dataset);
411 } else if constexpr (kind == h5::meta::access_t::text) {
412 // Scalar text — single string element on a scalar dataspace.
413 // Two storage flavours share this branch:
414 // - vlen_text_dataset (std::string, string_view): H5T_VARIABLE
415 // size, write the relay pointer (&relay) so HDF5 stores the
416 // pointed-to bytes.
417 // - fixed_length_string (char[N]): H5Tset_size(N), write the
418 // N raw bytes directly.
419 const char* relay = traits::data(ref);
420 hid_t str_type = H5Tcopy(H5T_C_S1);
421 if constexpr (storage == sr_t::fixed_length_string) {
422 H5Tset_size(str_type, traits::fixed_length);
423 } else {
424 H5Tset_size(str_type, H5T_VARIABLE);
425 }
426 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
427 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
428 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
429 H5Sselect_all(static_cast<hid_t>(file_space));
430 H5CPP_CHECK_NZ(
431 H5Dwrite(static_cast<hid_t>(ds), str_type,
432 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
433 static_cast<hid_t>(dxpl),
434 (storage == sr_t::fixed_length_string)
435 ? static_cast<const void*>(relay)
436 : static_cast<const void*>(&relay)),
437 h5::error::io::dataset::write, h5::error::msg::write_dataset);
438 H5Tclose(str_type);
439 } else if constexpr (kind == h5::meta::access_t::contiguous || kind == h5::meta::access_t::object) {
440 auto ptr = traits::data(ref);
441 if constexpr (!tcount::present) {
442 h5::count_t count = traits::size( ref );
443 ::h5::write(ds, ptr, count, args...);
444 } else ::h5::write(ds, ptr, args...);
445 } else if constexpr (kind == h5::meta::access_t::pointers) {
446 if constexpr (storage == sr_t::vlen_text_dataset) {
447 // vector<string> — relay array of char*; HDF5 owns no memory here
448 std::vector<const char*> relay;
449 relay.reserve(ref.size());
450 for (const auto& s : ref) relay.push_back(s.c_str());
451 hid_t vlen_str = H5Tcopy(H5T_C_S1);
452 H5Tset_size(vlen_str, H5T_VARIABLE);
453 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
454 h5::count_t count = traits::size(ref);
455 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(count[0]));
456 h5::select_all(mem_space);
457 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
458 H5Sselect_all(static_cast<hid_t>(file_space));
459 H5CPP_CHECK_NZ(
460 H5Dwrite(static_cast<hid_t>(ds), vlen_str,
461 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), relay.data()),
462 h5::error::io::dataset::write, h5::error::msg::write_dataset);
463 H5Tclose(vlen_str);
464 } else if constexpr (storage == sr_t::ragged_vlen_dataset) {
465 // vector<L> where L is an iterable container — hvl_t relay.
466 // For L=std::vector we can point hvl_t.p directly at the inner
467 // buffer; for L=list/set/deque/etc. we copy each inner into a
468 // flat scratch buffer first.
469 using inner_t = typename traits::element_t;
470 using elem_t = typename inner_t::value_type;
471 std::vector<hvl_t> relay(ref.size());
472 std::vector<std::vector<elem_t>> scratch;
473 constexpr bool inner_has_data = h5::meta::has_data_pointer<inner_t>::value;
474 if constexpr (!inner_has_data) scratch.resize(ref.size());
475 for (std::size_t i = 0; i < ref.size(); ++i) {
476 if constexpr (inner_has_data) {
477 relay[i].len = ref[i].size();
478 relay[i].p = const_cast<void*>(static_cast<const void*>(ref[i].data()));
479 } else {
480 scratch[i].assign(ref[i].begin(), ref[i].end());
481 relay[i].len = scratch[i].size();
482 relay[i].p = scratch[i].data();
483 }
484 }
485 h5::meta::resolved_type_t<elem_t> base_type;
486 hid_t vlen_type = H5Tvlen_create(static_cast<hid_t>(base_type));
487 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
488 h5::count_t count = traits::size(ref);
489 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(count[0]));
490 h5::select_all(mem_space);
491 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
492 H5Sselect_all(static_cast<hid_t>(file_space));
493 H5CPP_CHECK_NZ(

568 H5Sselect_all(static_cast<hid_t>(file_space));

579 // compound type. Composite element types route through the branch

588 std::vector<element_t> buffer;
589 buffer.reserve(n);
590 for (const auto& elem : ref)
591 buffer.push_back(elem);
592 ::h5::write(ds, buffer.data(), h5::count_t(count), args...);
593 }
594 } else static_assert(kind != h5::meta::access_t::unsupported, "unsupported type for h5::write");
