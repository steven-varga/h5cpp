74 template <class T, class hid_t>
75 H5CPP_ATTR_READ_RET(hid_t, T)
76 aread( const hid_t& ds, const std::string& name, const h5::acpl_t& acpl = h5::default_acpl ){
77 using traits = h5::meta::access_traits_t<T>;
78 using sr_t = h5::meta::storage_representation_t;
79 using element_t = typename impl::decay<T>::type;
80 constexpr auto kind = traits::kind;
81 constexpr auto storage = h5::meta::storage_representation_v<T>;
82
83 // Stopper: unsupported storage usually means an unregistered POD aggregate, a
84 // deeply-nested container, or std::vector<bool>. Bootstrap-aware (see H5Dwrite).
85#ifndef H5CPP_BUILDING_TYPE_INFO
86 static_assert(storage != sr_t::unsupported,
87 "h5::aread: storage_representation_v<T> resolved to 'unsupported'. "
88 "Check: unregistered POD aggregate (use H5CPP_REGISTER_STRUCT), "
89 "std::vector<bool>, or container nesting beyond vector<vector<T>>/vector<string>.");
90#endif
91
92 // Stopper: containers of containers must route via VLEN storage.
93 static_assert(
94 !h5::meta::is_stl_like<element_t>::value || storage == sr_t::ragged_vlen_dataset ||
95 storage == sr_t::vlen_text_dataset || storage == sr_t::array_dataset ||
96 storage == sr_t::array_element || storage == sr_t::fls_dataset || storage == sr_t::fixed_inner_extent_dataset,
97 "h5::aread: containers of containers are only supported for vector<string> "
98 "(vlen_text_dataset) or vector<vector<T>> (ragged_vlen_dataset).");
99
100 h5::at_t attr = h5::open(ds, name, h5::default_acpl);
101 ::hid_t sid;
102 H5CPP_CHECK_NZ( (sid = H5Aget_space( static_cast<::hid_t>(attr) )),
103 h5::error::io::attribute::read, "couldn't get space...");
104 h5::sp_t file_space{sid};
105 h5::current_dims_t current_dims;
106 get_simple_extent_dims(file_space, current_dims);
107
108 if constexpr (kind == h5::meta::access_t::composite) {
109 // scalar composite (std::tuple<Ts...>): H5Aread into pack buffer, unpack into ref.
110 std::vector<char> buf(traits::bytes());
111 h5::meta::resolved_type_t<T> mem_type;
112 H5CPP_CHECK_NZ( H5Aread( static_cast<::hid_t>(attr), static_cast<::hid_t>(mem_type), buf.data() ),
113 h5::error::io::attribute::read, "couldn't read composite attribute.");
114 T object{};
115 traits::unpack(object, buf.data());
116 return object;
117 } else if constexpr (storage == sr_t::array_element) {
118 // std::array<T,N> / T[N] — scalar dataspace, H5T_ARRAY element.
119 // Use the file's stored type via H5Aget_type for an exact match.
120 T object{};
121 ::hid_t mem_type = H5Aget_type(static_cast<::hid_t>(attr));
122 H5CPP_CHECK_NZ(H5Aread(static_cast<::hid_t>(attr), mem_type, traits::data(object)),
123 h5::error::io::attribute::read, "couldn't read array_element attribute.");
124 H5Tclose(mem_type);
125 return object;
126 } else if constexpr (storage == sr_t::array_dataset) {
127 using inner_t = std::remove_cv_t<typename T::value_type>;
128 using elem_scalar = typename inner_t::value_type;
129 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
130 hsize_t array_dims[1] = { static_cast<hsize_t>(N_inner) };
131 h5::meta::resolved_type_t<elem_scalar> base_type;
132 ::hid_t array_type = H5Tarray_create(static_cast<::hid_t>(base_type), 1, array_dims);
133 std::size_t outer = impl::nelements(current_dims);
134 T object; object.resize(outer);
135 H5CPP_CHECK_NZ(H5Aread(static_cast<::hid_t>(attr), array_type,
136 object.empty() ? nullptr : object.front().data()),
137 h5::error::io::attribute::read, "couldn't read array_dataset attribute.");
138 H5Tclose(array_type);
139 return object;
140 } else if constexpr (storage == sr_t::fls_dataset) {
141 using inner_t = std::remove_cv_t<typename T::value_type>;
142 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
143 ::hid_t fls_type = H5Tcopy(H5T_C_S1);
144 H5Tset_size(fls_type, N_inner);
145 std::size_t outer = impl::nelements(current_dims);
146 T object; object.resize(outer);
147 H5CPP_CHECK_NZ(H5Aread(static_cast<::hid_t>(attr), fls_type,
148 object.empty() ? nullptr : object.front().data()),
149 h5::error::io::attribute::read, "couldn't read fls_dataset attribute.");
150 H5Tclose(fls_type);
151 return object;
152 } else if constexpr (kind == h5::meta::access_t::text
153 && storage == sr_t::fixed_length_string) {
154 // char[N] / std::array<char,N> — H5T_C_S1 + H5Tset_size(N).
155 T object{};
156 ::hid_t fls_type = H5Tcopy(H5T_C_S1);
157 H5Tset_size(fls_type, traits::fixed_length);
158 H5CPP_CHECK_NZ(H5Aread(static_cast<::hid_t>(attr), fls_type, traits::data(object)),
159 h5::error::io::attribute::read, "couldn't read fixed-length string attribute.");
160 H5Tclose(fls_type);
161 return object;
162 } else if constexpr (kind == h5::meta::access_t::text || (kind == h5::meta::access_t::pointers && h5::meta::is_text_like<element_t>::value)) {
163 // Scalar std::string / std::vector<std::string>. The on-disk attribute might be either VLEN (h5::awrite(ds, "x", std::string{...}))
164 // or fixed-length (h5::awrite(ds, "x", "literal") — the literal has type `char[N]` and routes to H5Awrite.hpp's fixed_length
165 // branch). HDF5 has no fixed↔VLEN conversion path, so we must match the source type on read; pick the mem_type based on the
166 // attribute's stored datatype rather than always using VLEN.
167 ::hid_t atype = H5Aget_type(static_cast<::hid_t>(attr));
168 htri_t is_vlen = (H5Tget_class(atype) == H5T_STRING)
169 ? H5Tis_variable_str(atype) : htri_t{-1};
170 if (is_vlen == 0 && std::is_same_v<std::string, T>) {
171 // Fixed-length string on disk: read N raw bytes into a buffer
172 // and construct std::string trimming NULLTERM padding.
173 size_t asize = H5Tget_size(atype);
174 std::vector<char> buf(asize, '\0');
175 ::hid_t fls_type = H5Tcopy(H5T_C_S1);
176 H5Tset_size(fls_type, asize);
177 H5CPP_CHECK_NZ( H5Aread(static_cast<::hid_t>(attr), fls_type, buf.data()),
178 h5::error::io::attribute::read, "couldn't read fixed-length string attribute.");
179 H5Tclose(fls_type);
180 H5Tclose(atype);
181 size_t len = 0; while (len < asize && buf[len] != '\0') ++len;
182 if constexpr (std::is_same_v<std::string, T>)
183 return T(buf.data(), len);
184 else return T{}; // unreachable (guarded by is_same_v above)
185 }
186 H5Tclose(atype);
187 // VLEN path
188 h5::dt_t<char*> type;
189 T object = impl::get<T>::ctor(current_dims);
190 size_t nelem = impl::nelements(current_dims);
191 char** ptr = static_cast<char**>(malloc(nelem * sizeof(char*)));
192 H5CPP_CHECK_NZ( H5Aread( static_cast<::hid_t>(attr), static_cast<::hid_t>(type), ptr ),
193 h5::error::io::attribute::read, "couldn't read attribute...");
194 if constexpr (std::is_same_v<std::string, T>) {
195 object = std::string(*ptr);
196 } else {
197 for (size_t i = 0; i < nelem; i++)
198 if (ptr[i] != nullptr) object[i] = std::string(ptr[i]);
199 }
200 detail::attr_vlen_reclaim(static_cast<::hid_t>(type), static_cast<::hid_t>(file_space), ptr);
201 free(ptr);
202 return object;
203 } else if constexpr (kind == h5::meta::access_t::object) {
204 // Scalar path: arithmetic, POD aggregates, complex<T>, pair<K,V>
205 h5::dt_t<element_t> type;
206 T object{};
207 H5CPP_CHECK_NZ( H5Aread( static_cast<::hid_t>(attr), static_cast<::hid_t>(type), &object ),
208 h5::error::io::attribute::read, "couldn't read attribute...");
209 return object;
210 } else if constexpr (kind == h5::meta::access_t::pointers) {
211 // Nested: traits::element_t only referenced inside this block, where
212 // the pointers access_traits_t spec guarantees it exists.
213 if constexpr (storage == sr_t::ragged_vlen_dataset) {
214 // vector<vector<T>>: hvl_t relay + reclaim.
215 using inner_t = typename traits::element_t;
216 using elem_t = typename inner_t::value_type;
217 std::size_t n = impl::nelements(current_dims);
218 T object; object.resize(n);
219 std::vector<hvl_t> relay(n);
220 h5::meta::resolved_type_t<elem_t> base_type;
221 ::hid_t vlen_type = H5Tvlen_create(static_cast<::hid_t>(base_type));
222 H5CPP_CHECK_NZ( H5Aread(static_cast<::hid_t>(attr), vlen_type, relay.data()),
223 h5::error::io::attribute::read, "couldn't read ragged_vlen attribute.");
224 for (std::size_t i = 0; i < n; ++i) {
225 const elem_t* src = static_cast<const elem_t*>(relay[i].p);
226 object[i].assign(src, src + relay[i].len);
227 }
228 detail::attr_vlen_reclaim(vlen_type, static_cast<::hid_t>(file_space), relay.data());
229 H5Tclose(vlen_type);
230 return object;
231 } else if constexpr (h5::meta::access_kind_v<typename traits::element_t> == h5::meta::access_t::composite) {
232 // vector<tuple<Ts...>>: H5Aread into pack buffer, unpack each.
233 using elem_t = typename traits::element_t;
234 using elem_traits = h5::meta::access_traits_t<elem_t>;
235 std::size_t n = impl::nelements(current_dims);
236 T object; object.resize(n);
237 std::vector<char> buf(n * elem_traits::bytes());
238 h5::meta::resolved_type_t<elem_t> mem_type;
239 H5CPP_CHECK_NZ( H5Aread(static_cast<::hid_t>(attr), static_cast<::hid_t>(mem_type), buf.data()),
240 h5::error::io::attribute::read, "couldn't read composite-element attribute.");
241 for (std::size_t i = 0; i < n; ++i)
242 elem_traits::unpack(object[i], buf.data() + i * elem_traits::bytes());
243 return object;
244 } else {
245 static_assert(storage == sr_t::ragged_vlen_dataset, "h5::aread: unsupported pointers storage class.");
246 return T{};
247 }
248 } else if constexpr (kind == h5::meta::access_t::iterators) {
249 // Nested for the same reason as the pointers block above.
250 if constexpr (storage == sr_t::key_value_dataset) {
251 // map<K,V>: flat kv_t compound buffer, insert into map.
252 using map_elem_t = typename traits::element_t;
253 using key_t = std::remove_const_t<typename map_elem_t::first_type>;
254 using value_t = typename map_elem_t::second_type;
255 struct kv_t { key_t key; value_t value; };
256 h5::meta::resolved_type_t<key_t> kt;
257 h5::meta::resolved_type_t<value_t> vt;
258 ::hid_t compound = H5Tcreate(H5T_COMPOUND, sizeof(kv_t));
259 H5Tinsert(compound, "key", offsetof(kv_t, key), static_cast<::hid_t>(kt));
260 H5Tinsert(compound, "value", offsetof(kv_t, value), static_cast<::hid_t>(vt));
261 std::size_t n = impl::nelements(current_dims);
262 std::vector<kv_t> buffer(n);
263 H5CPP_CHECK_NZ( H5Aread(static_cast<::hid_t>(attr), compound, buffer.data()),
264 h5::error::io::attribute::read, "couldn't read key_value attribute.");
265 T object;
266 for (const auto& kv : buffer)
267 object.insert({kv.key, kv.value});
268 H5Tclose(compound);
269 return object;
270 } else if constexpr (h5::meta::access_kind_v<typename traits::element_t> == h5::meta::access_t::composite) {
271 // list<tuple>, set<tuple>, deque<tuple>: read pack buffer, unpack into container.
272 using elem_t = typename traits::element_t;
273 using elem_traits = h5::meta::access_traits_t<elem_t>;
274 std::size_t n = impl::nelements(current_dims);
275 std::vector<char> buf(n * elem_traits::bytes());
276 h5::meta::resolved_type_t<elem_t> mem_type;
277 H5CPP_CHECK_NZ( H5Aread(static_cast<::hid_t>(attr), static_cast<::hid_t>(mem_type), buf.data()),
278 h5::error::io::attribute::read, "couldn't read iter-composite attribute.");
279 std::vector<elem_t> staging(n);
280 for (std::size_t i = 0; i < n; ++i)
281 elem_traits::unpack(staging[i], buf.data() + i * elem_traits::bytes());
282 T object;
283 if constexpr (std::is_same_v<T, std::forward_list<elem_t>>) {
284 object.assign(staging.begin(), staging.end());
285 } else std::copy(staging.begin(), staging.end(), std::inserter(object, object.end()));
286 return object;
287 } else {
288 // list<T>, set<T>, deque<T> with non-composite, std-layout element_t.
289 using iter_elem_t = typename impl::decay<typename traits::element_t>::type;
290 static_assert(std::is_standard_layout_v<iter_elem_t>,
291 "h5::aread: iterator-staging path requires standard-layout element_t.");
292 std::size_t n = impl::nelements(current_dims);
293 std::vector<iter_elem_t> staging(n);
294 h5::dt_t<iter_elem_t> type;
295 H5CPP_CHECK_NZ( H5Aread(static_cast<::hid_t>(attr), static_cast<::hid_t>(type), staging.data()),
296 h5::error::io::attribute::read, "couldn't read iter attribute.");
297 T object;
298 if constexpr (std::is_same_v<T, std::forward_list<iter_elem_t>>) {
299 object.assign(staging.begin(), staging.end());
300 } else std::copy(staging.begin(), staging.end(), std::inserter(object, object.end()));
301 return object;
302 }
303 } else {
304 // Array/container path: std::array<T,N>, std::vector<T>, std::vector<std::complex<T>>,
305 // armadillo / Eigen / Blaze / Blitz / xtensor linalg mapper types.
306 //
307 // Use impl::data (not traits::data) to get a writable pointer — the linalg
308 // mapper access_traits_t specs only provide a const-qualified data() overload,
309 // and H5Aread needs void* (no implicit conversion from const T*).
310 //
311 // Direct-initialise via impl::get<T>::ctor for rank > 0 (mirrors H5Dread's
312 // array fallback at ~line 580). Default-construct + assign doesn't work for
313 // every container — blitz::Array, in particular, ends up unallocated after
314 // `T{}; obj = ctor(...)` because its operator= treats LHS shape as authoritative.
315 h5::dt_t<element_t> type;
316 if constexpr (impl::rank<T>::value > 0) {
317 T object = impl::get<T>::ctor(current_dims);
318 auto* ptr = impl::data(object);
319 H5CPP_CHECK_NZ( H5Aread( static_cast<::hid_t>(attr), static_cast<::hid_t>(type), ptr ),
320 h5::error::io::attribute::read, "couldn't read attribute...");
321 return object;
322 } else {
323 T object{};
324 auto* ptr = impl::data(object);
325 H5CPP_CHECK_NZ( H5Aread( static_cast<::hid_t>(attr), static_cast<::hid_t>(type), ptr ),
326 h5::error::io::attribute::read, "couldn't read attribute...");
327 return object;
328 }
329 }

333// Implicit read for `T v = parent["attr"]`. The at_t conversion operator
334// (inline in H5Iall.hpp) forwards here. Kept as a free function template — NOT
335// an out-of-line member-template conversion operator of the at_t specialization
336// — because MSVC (VS2022 14.4x) emits a C1001 internal compiler error on that
337// form when the examples' bracket-syntax reads instantiate it. Mirrors the
338// at_t::operator=(V) write path.
339//
340// Forwards the raw ::hid_t directly (is_valid_attr accepts it) rather than
341// wrapping in an h5::ob_t — the wrapper would close the underlying handle on
342// destruction and orphan the caller's parent. Throws if ds is invalid.
343template <class V>
344inline V h5::impl::detail::at_read(::hid_t ds, const std::string& name) {
345 if( !H5Iis_valid(ds) )
346 throw h5::error::io::attribute::read("h5::at_t implicit read: parent ds handle is invalid; use parent[\"name\"].");
347 return h5::aread<V>(ds, name);
348}
