54 template<class T, class... args_t>
55 inline std::enable_if_t<!std::is_same_v<T,char**>,
56 void> read( const h5::ds_t& ds, T* ptr, args_t&&... args ) try {
57 using tcount = typename arg::tpos<const h5::count_t&,const args_t&...>;
58 static_assert( tcount::present, "h5::count_t{ ... } must be specified" );
59 static_assert( utils::is_supported<T>, "error: " H5CPP_supported_elementary_types );
60
61 auto tuple = std::forward_as_tuple(args...);
62 const h5::count_t& count = std::get<tcount::value>( tuple );
63
64 h5::offset_t default_offset{0,0,0,0,0,0,0}; // must match H5CPP_MAX_RANK=7
65 const h5::offset_t& offset = arg::get( default_offset, args...);
66
67 h5::stride_t default_stride{1,1,1,1,1,1,1};
68 const h5::stride_t& stride = arg::get( default_stride, args...);
69
70 h5::block_t default_block{1,1,1,1,1,1,1};
71 const h5::block_t& block = arg::get( default_block, args...);
72
73 h5::count_t size; // compute actual memory space
74 for(int i=0;i<count.rank;i++) size[i] = count[i] * block[i];
75 size.rank = count.rank;
76
77 const h5::dxpl_t& dxpl = arg::get( h5::default_dxpl, args...);
78 H5CPP_CHECK_PROP( dxpl, h5::error::property_list::misc, "invalid data transfer property" );
79
80 h5::sp_t file_space = h5::get_space(ds);
81 int rank = h5::get_simple_extent_ndims( file_space );
82
83 if( rank != count.rank ) throw h5::error::io::dataset::read( H5CPP_ERROR_MSG( h5::error::msg::rank_mismatch ));
84 using element_t = typename impl::decay<T>::type;
85 h5::meta::resolved_type_t<element_t> mem_type;
86 // Resolve the parallel-read tag from the dataset's REAL access plist
87 // (H5Dget_access_plist), NOT ds.dapl: that member caches the open-time hid_t,
88 // which dangles once a temporary dapl — e.g. h5::open(fd, name, h5::threads{N})
89 // — is destroyed, so the tag would be silently lost and every read would fall
90 // back to the serial path. #287.
91 hid_t dapl = H5Dget_access_plist( static_cast<hid_t>(ds) );
92 // Read dispatch mirrors the write path: direct-chunk (H5Dread_chunk, which
93 // bypasses HDF5's raw-data chunk cache — and so does NOT leak it after a
94 // direct-chunk write on the same handle) by default for chunked datasets
95 // with h5cpp-applied filters; parallel decompression when h5::threads{N} is
96 // on the dataset's DAPL; stock H5Dread for a hyperslab selection (the
97 // direct-chunk path reads whole chunks, not a sub-region), contiguous, or
98 // NBIT / SCALEOFFSET.
99 using toffset = typename arg::tpos<const h5::offset_t&, const args_t&...>;
100 using tstride = typename arg::tpos<const h5::stride_t&, const args_t&...>;
101 using tblock = typename arg::tpos<const h5::block_t&, const args_t&...>;
102 constexpr bool hyperslab = toffset::present || tstride::present || tblock::present;
103 const unsigned read_threads = h5::impl::resolve_dataset_threads(dapl);
104 const unsigned read_cap = read_threads
105 ? h5::impl::resolve_dataset_backpressure(dapl, read_threads) : 0u;
106 if (dapl >= 0) H5Pclose(dapl); // owned copy from H5Dget_access_plist
107 const bool use_pipeline = [&]() {
108 if constexpr (hyperslab) return false; // hyperslab read → stock H5Dread
109 hid_t dcpl_id = H5Dget_create_plist(static_cast<hid_t>(ds));
110 if (dcpl_id < 0) return false;
111 bool ok = (H5Pget_layout(dcpl_id) == H5D_CHUNKED);
112 for (int fi = 0, nf = H5Pget_nfilters(dcpl_id); ok && fi < nf; ++fi) {
113 size_t ne = 0; unsigned fl = 0, cfg = 0;
114 H5Z_filter_t id = H5Pget_filter2(dcpl_id, static_cast<unsigned>(fi),
115 &fl, &ne, nullptr, 0, nullptr, &cfg);
116 if (id == H5Z_FILTER_NBIT || id == H5Z_FILTER_SCALEOFFSET) ok = false;
117 }
118 H5Pclose(dcpl_id);
119 return ok;
120 }();
121 if( use_pipeline ){
122 h5::dcpl_t dcpl{H5Dget_create_plist(static_cast<hid_t>(ds))};
123 hid_t type_id = H5Dget_type(static_cast<hid_t>(ds));
124 size_t elem_sz = H5Tget_size(type_id);
125 H5Tclose(type_id);
126 if (read_threads > 0) { // parallel decompression
127 h5::impl::pool_pipeline_t pipe(h5::impl::global_pool_ptr(), read_cap);
128 pipe.set_cache(dcpl, elem_sz);
129 pipe.read(ds, offset, stride, block, count, dxpl, ptr);
130 } else { // direct-chunk read (default)
131 h5::impl::pipeline_t<impl::basic_pipeline_t> pipe;
132 pipe.set_cache(dcpl, elem_sz);
133 pipe.read(ds, offset, stride, block, count, dxpl, ptr);
134 }
135 }else{
136 // Scalar dataspaces don't support hyperslab selection; H5Sselect_all
137 // on both sides is the equivalent path for rank=0 / H5S_SCALAR.
138 // Detect by file_space class rather than count.rank so we behave
139 // correctly when the caller passed an empty count.
140 H5S_class_t file_cls = H5Sget_simple_extent_type(static_cast<hid_t>(file_space));
141 h5::sp_t mem_space = (file_cls == H5S_SCALAR)
142 ? h5::sp_t{H5Screate(H5S_SCALAR)}
143 : h5::create_simple( size );
144 h5::select_all( mem_space );
145 if (file_cls == H5S_SCALAR) {
146 H5Sselect_all(static_cast<hid_t>(file_space));
147 } else {
148 h5::select_hyperslab( file_space, offset, stride, count, block);
149 }
150
151 H5CPP_CHECK_NZ( H5Dread(
152 static_cast<hid_t>( ds ), static_cast<hid_t>(mem_type), static_cast<hid_t>(mem_space),
153 static_cast<hid_t>(file_space), static_cast<hid_t>(dxpl), ptr ), h5::error::io::dataset::read, h5::error::msg::read_dataset);
154 }
155 } catch ( const std::runtime_error& err ){
156 throw h5::error::io::dataset::read( err.what() );
157 }
158
193 // SFINAE-gate the pointer overload on h5::count_t presence so a bare
194 // `read(fd, path, buf)` with no count is NOT matched here (where it would
195 // hit a hard static_assert) but instead routes to the by-reference path,
196 // letting char[N] reach the fixed-length-string dispatch.
197 template<class T, class... args_t,
198 class = std::enable_if_t<arg::tpos<const h5::count_t&, const args_t&...>::present>>
199 inline void read( const h5::fd_t& fd, const std::string& dataset_path, T* ptr, args_t&&... args ){
200 const h5::dapl_t& dapl = arg::get(h5::default_dapl, args...);
201 h5::ds_t ds = h5::open(fd, dataset_path, dapl ); // will throw its exception
202 ::h5::read<T>(ds, ptr, args...);
203 }

284 template<class T, class... args_t>
285 inline void read(const h5::ds_t& ds, T& ref, args_t&&... args) try {
286 using tcount = typename arg::tpos<const h5::count_t&, const args_t&...>;
287 using element_t = typename impl::decay<T>::type;
288 using traits = h5::meta::access_traits_t<T>;
289 using sr_t = h5::meta::storage_representation_t;
290 constexpr auto kind = traits::kind;
291 constexpr auto storage = h5::meta::storage_representation_v<T>;
292
293 static_assert(!tcount::present,
294 "h5::count_t{ ... } is already present when passing arg by reference, did you mean to pass by pointer?");
295
296 // Stopper: scatter types must use the fd-gateway overload so the compiler-generated
297 // h5::gather<T> specialization can dispatch. Calling h5::read(ds, ref) on a scatter
298 // type would silently bypass gather and fall through to the aggregate path with
299 // register_struct<T>() == H5I_UNINIT.
300 static_assert(!h5::has_scatter<std::decay_t<T>>::value,
301 "h5::read(ds, ref): scatter types must use h5::read(fd, path, ref) so the "
302 "compiler-generated h5::gather<T> specialization can dispatch.");
303
304 // Stopper: unsupported storage usually means an unregistered POD aggregate, a
305 // deeply-nested container, or std::vector<bool>. Bootstrap-aware: skipped when
306 // h5cpp-compiler defines H5CPP_BUILDING_TYPE_INFO (generated.h not yet emitted).
307#ifndef H5CPP_BUILDING_TYPE_INFO
308 static_assert(storage != sr_t::unsupported,
309 "h5::read: storage_representation_v<T> resolved to 'unsupported'. "
310 "Check: unregistered POD aggregate (use H5CPP_REGISTER_STRUCT), "
311 "std::vector<bool>, or container nesting beyond vector<vector<T>>/vector<string>.");
312#endif
313
314 // Stopper (borrowed from #274): containers of containers must route via VLEN
315 // storage. element_t comes from impl::decay<T>, SFINAE-safe for non-container T.
316 static_assert(
317 !h5::meta::is_stl_like<element_t>::value ||
318 storage == sr_t::ragged_vlen_dataset ||
319 storage == sr_t::vlen_text_dataset ||
320 storage == sr_t::array_dataset ||
321 storage == sr_t::array_element ||
322 storage == sr_t::fls_dataset ||
323 storage == sr_t::fixed_inner_extent_dataset,
324 "h5::read: containers of containers are only supported for vector<string> "
325 "(vlen_text_dataset) or vector<vector<T>> (ragged_vlen_dataset).");
326
327 if constexpr (storage == sr_t::array_element) {
328 // Scalar dataspace, single H5T_ARRAY element. Use the file's
329 // stored datatype directly via H5Dget_type so the in-memory and
330 // on-disk type IDs are guaranteed equal — and pass H5S_ALL on
331 // both sides to avoid scalar-space selection mismatches.
332 // const_cast strips any const that survives traits::data(ref)
333 // when ref is a nested std::array (the const overload picks up
334 // the implicit `this`-const propagation).
335 hid_t mem_type = H5Dget_type(static_cast<hid_t>(ds));
336 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
337 H5CPP_CHECK_NZ(
338 H5Dread(static_cast<hid_t>(ds), mem_type,
339 H5S_ALL, H5S_ALL,
340 static_cast<hid_t>(dxpl), const_cast<void*>(static_cast<const void*>(traits::data(ref)))),
341 h5::error::io::dataset::read, h5::error::msg::read_dataset);
342 H5Tclose(mem_type);
343 } else if constexpr (storage == sr_t::array_dataset) {
344 using inner_t = std::remove_cv_t<typename std::remove_reference_t<T>::value_type>;
345 using elem_scalar = typename inner_t::value_type;
346 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
347 hsize_t array_dims[1] = { static_cast<hsize_t>(N_inner) };
348 h5::meta::resolved_type_t<elem_scalar> base_type;
349 hid_t array_type = H5Tarray_create(static_cast<hid_t>(base_type), 1, array_dims);
350 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
351 hsize_t outer = 0;
352 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &outer, nullptr);
353 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
354 h5::sp_t mem_space = h5::create_simple(outer);
355 h5::select_all(mem_space);
356 H5Sselect_all(static_cast<hid_t>(file_space));
357 // Vector: direct buffer. List/set/etc.: read into scratch, then assign.
358 if constexpr (h5::meta::has_data_pointer<std::remove_cv_t<std::remove_reference_t<T>>>::value) {
359 ref.resize(static_cast<std::size_t>(outer));
360 H5CPP_CHECK_NZ(
361 H5Dread(static_cast<hid_t>(ds), array_type,
362 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
363 static_cast<hid_t>(dxpl), ref.empty() ? nullptr : ref.front().data()),
364 h5::error::io::dataset::read, h5::error::msg::read_dataset);
365 } else {
366 std::vector<inner_t> scratch(static_cast<std::size_t>(outer));
367 H5CPP_CHECK_NZ(
368 H5Dread(static_cast<hid_t>(ds), array_type,
369 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
370 static_cast<hid_t>(dxpl), scratch.empty() ? nullptr : scratch.front().data()),
371 h5::error::io::dataset::read, h5::error::msg::read_dataset);
372 if constexpr (h5::meta::is_set_like<std::remove_cv_t<std::remove_reference_t<T>>>::value) {
373 ref = std::remove_cv_t<std::remove_reference_t<T>>(scratch.begin(), scratch.end());
374 } else {
375 ref.assign(scratch.begin(), scratch.end());
376 }
377 }
378 H5Tclose(array_type);
379 } else if constexpr (storage == sr_t::fls_dataset) {
380 using inner_t = std::remove_cv_t<typename std::remove_reference_t<T>::value_type>;
381 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
382 hid_t str_type = H5Tcopy(H5T_C_S1);
383 H5Tset_size(str_type, N_inner);
384 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
385 hsize_t outer = 0;
386 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &outer, nullptr);
387 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
388 h5::sp_t mem_space = h5::create_simple(outer);
389 h5::select_all(mem_space);
390 H5Sselect_all(static_cast<hid_t>(file_space));
391 if constexpr (h5::meta::has_data_pointer<std::remove_cv_t<std::remove_reference_t<T>>>::value) {
392 ref.resize(static_cast<std::size_t>(outer));
393 H5CPP_CHECK_NZ(
394 H5Dread(static_cast<hid_t>(ds), str_type,
395 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
396 static_cast<hid_t>(dxpl), ref.empty() ? nullptr : ref.front().data()),
397 h5::error::io::dataset::read, h5::error::msg::read_dataset);
398 } else {
399 std::vector<inner_t> scratch(static_cast<std::size_t>(outer));
400 H5CPP_CHECK_NZ(
401 H5Dread(static_cast<hid_t>(ds), str_type,
402 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
403 static_cast<hid_t>(dxpl), scratch.empty() ? nullptr : scratch.front().data()),
404 h5::error::io::dataset::read, h5::error::msg::read_dataset);
405 if constexpr (h5::meta::is_set_like<std::remove_cv_t<std::remove_reference_t<T>>>::value) {
406 ref = std::remove_cv_t<std::remove_reference_t<T>>(scratch.begin(), scratch.end());
407 } else {
408 ref.assign(scratch.begin(), scratch.end());
409 }
410 }
411 H5Tclose(str_type);
412 } else if constexpr (kind == h5::meta::access_t::composite) {
413 // scalar composite (std::tuple<Ts...>): H5Dread into pack buffer, unpack into ref.
414 std::vector<char> buf(traits::bytes());
415 h5::meta::resolved_type_t<T> mem_type;
416 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
417 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
418 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
419 H5Sselect_all(static_cast<hid_t>(file_space));
420 H5CPP_CHECK_NZ(
421 H5Dread(static_cast<hid_t>(ds), static_cast<hid_t>(mem_type),
422 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
423 static_cast<hid_t>(dxpl), buf.data()),
424 h5::error::io::dataset::read, h5::error::msg::read_dataset);
425 traits::unpack(ref, buf.data());
426 } else if constexpr (kind == h5::meta::access_t::text) {
427 // Scalar text read — variable-length (std::string) reads via
428 // char** relay + reclaim; fixed-length (char[N]) reads N raw
429 // bytes straight into ref.
430 hid_t str_type = H5Tcopy(H5T_C_S1);
431 if constexpr (storage == sr_t::fixed_length_string) {
432 H5Tset_size(str_type, traits::fixed_length);
433 } else {
434 H5Tset_size(str_type, H5T_VARIABLE);
435 }
436 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
437 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
438 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
439 H5Sselect_all(static_cast<hid_t>(file_space));
440 if constexpr (storage == sr_t::fixed_length_string) {
441 H5CPP_CHECK_NZ(
442 H5Dread(static_cast<hid_t>(ds), str_type,
443 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
444 static_cast<hid_t>(dxpl), traits::data(ref)),
445 h5::error::io::dataset::read, h5::error::msg::read_dataset);
446 } else {
447 char* relay = nullptr;
448 H5CPP_CHECK_NZ(
449 H5Dread(static_cast<hid_t>(ds), str_type,
450 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
451 static_cast<hid_t>(dxpl), &relay),
452 h5::error::io::dataset::read, h5::error::msg::read_dataset);
453 ref = relay ? std::decay_t<T>(relay) : std::decay_t<T>{};
454 h5::impl::reference::reclaim(str_type, static_cast<hid_t>(mem_space), H5P_DEFAULT, &relay);
455 }
456 H5Tclose(str_type);
457 } else if constexpr (kind == h5::meta::access_t::contiguous || kind == h5::meta::access_t::object) {
458 auto ptr = traits::data(ref);
459 h5::count_t count = traits::size(ref);
460 ::h5::read<typename traits::element_t>(ds, ptr, count, args...);
461 } else if constexpr (kind == h5::meta::access_t::pointers) {
462 if constexpr (storage == sr_t::vlen_text_dataset) {
463 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
464 hsize_t n = 0;
465 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &n, nullptr);
466 ref.resize(static_cast<std::size_t>(n));
467 std::vector<char*> relay(static_cast<std::size_t>(n), nullptr);
468 hid_t vlen_str = H5Tcopy(H5T_C_S1);
469 H5Tset_size(vlen_str, H5T_VARIABLE);
470 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
471 h5::sp_t mem_space = h5::create_simple(n);
472 h5::select_all(mem_space);
473 H5Sselect_all(static_cast<hid_t>(file_space));
474 H5CPP_CHECK_NZ(
475 H5Dread(static_cast<hid_t>(ds), vlen_str,
476 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
477 static_cast<hid_t>(dxpl), relay.data()),
478 h5::error::io::dataset::read, h5::error::msg::read_dataset);
479 for (std::size_t i = 0; i < static_cast<std::size_t>(n); ++i)
480 ref[i] = relay[i] ? std::string(relay[i]) : std::string{};
481 h5::impl::reference::reclaim(vlen_str, static_cast<hid_t>(mem_space), H5P_DEFAULT, relay.data());
482 H5Tclose(vlen_str);
483 } else if constexpr (storage == sr_t::ragged_vlen_dataset) {
484 using inner_t = typename traits::element_t;
485 using elem_t = typename inner_t::value_type;
486 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
487 hsize_t n = 0;
488 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &n, nullptr);
489 ref.resize(static_cast<std::size_t>(n));
490 std::vector<hvl_t> relay(static_cast<std::size_t>(n));
491 h5::meta::resolved_type_t<elem_t> base_type;
492 hid_t vlen_type = H5Tvlen_create(static_cast<hid_t>(base_type));
493 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
494 h5::sp_t mem_space = h5::create_simple(n);
495 h5::select_all(mem_space);
496 H5Sselect_all(static_cast<hid_t>(file_space));
497 H5CPP_CHECK_NZ(
498 H5Dread(static_cast<hid_t>(ds), vlen_type,
499 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
500 static_cast<hid_t>(dxpl), relay.data()),
501 h5::error::io::dataset::read, h5::error::msg::read_dataset);
502 for (std::size_t i = 0; i < static_cast<std::size_t>(n); ++i) {
503 const elem_t* src = static_cast<const elem_t*>(relay[i].p);
504 // Sequence containers expose .assign(iter, iter); set-like
505 // containers don't, but accept range construction.
506 if constexpr (h5::meta::is_set_like<inner_t>::value ||
507 h5::meta::is_associative_like<inner_t>::value) {
508 ref[i] = inner_t(src, src + relay[i].len);
509 } else {
510 ref[i].assign(src, src + relay[i].len);
511 }
512 }
513 h5::impl::reference::reclaim(vlen_type, static_cast<hid_t>(mem_space), H5P_DEFAULT, relay.data());
514 H5Tclose(vlen_type);
515 } else if constexpr (h5::meta::access_kind_v<typename traits::element_t> == h5::meta::access_t::composite) {
516 // vector<tuple<Ts...>>: H5Dread into pack buffer, unpack each element.
517 using elem_traits = h5::meta::access_traits_t<typename traits::element_t>;
518 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
519 hsize_t n = 0;
520 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &n, nullptr);
521 ref.resize(static_cast<std::size_t>(n));
522 std::vector<char> buf(static_cast<std::size_t>(n) * elem_traits::bytes());
523 h5::meta::resolved_type_t<typename traits::element_t> mem_type;
524 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
525 h5::sp_t mem_space = h5::create_simple(n);
526 h5::select_all(mem_space);
527 H5Sselect_all(static_cast<hid_t>(file_space));
528 H5CPP_CHECK_NZ(
529 H5Dread(static_cast<hid_t>(ds), static_cast<hid_t>(mem_type),
530 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
531 static_cast<hid_t>(dxpl), buf.data()),
532 h5::error::io::dataset::read, h5::error::msg::read_dataset);
533 for (std::size_t i = 0; i < static_cast<std::size_t>(n); ++i)
534 elem_traits::unpack(ref[i], buf.data() + i * elem_traits::bytes());
535 }
536 } else if constexpr (kind == h5::meta::access_t::iterators) {
537 if constexpr (storage == sr_t::key_value_dataset) {
538 using element_t = typename traits::element_t;
539 using key_t = std::remove_const_t<typename element_t::first_type>;
540 using value_t = typename element_t::second_type;

542 h5::meta::resolved_type_t<key_t> kt;
543 h5::meta::resolved_type_t<value_t> vt;
544 hid_t compound = H5Tcreate(H5T_COMPOUND, sizeof(kv_t));
545 H5Tinsert(compound, "key", offsetof(kv_t, key), static_cast<hid_t>(kt));
546 H5Tinsert(compound, "value", offsetof(kv_t, value), static_cast<hid_t>(vt));
547 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
548 hsize_t n = 0;
549 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &n, nullptr);
550 std::vector<kv_t> buffer(static_cast<std::size_t>(n));
551 h5::sp_t mem_space = h5::create_simple(n);

553 H5Sselect_all(static_cast<hid_t>(file_space));
554 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
555 H5CPP_CHECK_NZ(
556 H5Dread(static_cast<hid_t>(ds), compound,
557 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
558 static_cast<hid_t>(dxpl), buffer.data()),
559 h5::error::io::dataset::read, h5::error::msg::read_dataset);
560 ref.clear();
561 for (const auto& kv : buffer)
562 ref.insert({kv.key, kv.value});
563 H5Tclose(compound);
564 } else if constexpr (h5::meta::access_kind_v<typename traits::element_t> == h5::meta::access_t::composite) {
565 // list<tuple>, set<tuple>, deque<tuple>: read pack buffer, unpack into container.
566 using elem_t = typename traits::element_t;
567 using elem_traits = h5::meta::access_traits_t<elem_t>;
568 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
569 hsize_t n = 0;
570 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &n, nullptr);
571 std::vector<char> buf(static_cast<std::size_t>(n) * elem_traits::bytes());
572 h5::meta::resolved_type_t<elem_t> mem_type;
573 h5::sp_t mem_space = h5::create_simple(n);
574 h5::select_all(mem_space);
575 H5Sselect_all(static_cast<hid_t>(file_space));
576 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
577 H5CPP_CHECK_NZ(
578 H5Dread(static_cast<hid_t>(ds), static_cast<hid_t>(mem_type),
579 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
580 static_cast<hid_t>(dxpl), buf.data()),
581 h5::error::io::dataset::read, h5::error::msg::read_dataset);
582 ref.clear();
583 std::vector<elem_t> staging(static_cast<std::size_t>(n));
584 for (std::size_t i = 0; i < static_cast<std::size_t>(n); ++i)
585 elem_traits::unpack(staging[i], buf.data() + i * elem_traits::bytes());
586 if constexpr (std::is_same_v<T, std::forward_list<elem_t>>) {
587 ref.assign(staging.begin(), staging.end());
588 } else {
589 std::copy(staging.begin(), staging.end(), std::inserter(ref, ref.end()));
590 }
591 } else {
592 using element_t = typename impl::decay<typename traits::element_t>::type;
593 // Guard (review item A2): non-std-layout element_t (e.g., escaped composite)
594 // would silently produce wrong on-disk layout. Composite element types route
595 // through the branch above; this catches any future escape.
596 static_assert(std::is_standard_layout_v<element_t>,
597 "h5::read: iterator-staging path requires standard-layout element_t. "
598 "Wrap non-std-layout types in std::tuple (composite kind) or convert "
599 "to a flat representation before reading.");
600 h5::sp_t file_space{H5Dget_space(static_cast<hid_t>(ds))};
601 hsize_t n = 0;
602 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &n, nullptr);
603 std::vector<element_t> buffer(static_cast<std::size_t>(n));
604 h5::count_t count{std::array<std::size_t,1>{static_cast<std::size_t>(n)}};
605 ::h5::read<element_t>(ds, buffer.data(), count, args...);
606 ref.clear();
607 if constexpr (std::is_same_v<T, std::forward_list<element_t>>) {
608 ref.assign(buffer.begin(), buffer.end());
609 } else {
610 std::copy(buffer.begin(), buffer.end(), std::inserter(ref, ref.end()));
611 }
612 }
613 } else {
614 static_assert(kind != h5::meta::access_t::unsupported, "unsupported type for h5::read");
615 }
616 } catch (const std::exception& err) {
617 throw h5::error::io::dataset::read(err.what());

729 template<class T, class... args_t>
730 inline T read(const h5::ds_t& ds, args_t&&... args) {
731 using tcount = typename arg::tpos<const h5::count_t&, const args_t&...>;
732 using element_t = typename impl::decay<T>::type;
733 using traits = h5::meta::access_traits_t<T>;
734 using sr_t = h5::meta::storage_representation_t;
735 constexpr auto kind = traits::kind;
736 constexpr auto storage = h5::meta::storage_representation_v<T>;
737
738 // Stopper: scatter types must use the fd-gateway overload.
739 static_assert(!h5::has_scatter<std::decay_t<T>>::value,
740 "h5::read<T>(ds): scatter types must use h5::read<T>(fd, path) so the "
741 "compiler-generated h5::gather<T> specialization can dispatch.");
742 // Stopper: unsupported storage. Bootstrap-aware (see ds-overload above).
743#ifndef H5CPP_BUILDING_TYPE_INFO
744 static_assert(storage != sr_t::unsupported,
745 "h5::read<T>: storage_representation_v<T> resolved to 'unsupported'. "
746 "Check: unregistered POD aggregate (use H5CPP_REGISTER_STRUCT), "
747 "std::vector<bool>, or container nesting beyond vector<vector<T>>/vector<string>.");
748#endif
749 // Stopper: containers of containers must route via VLEN storage.
750 static_assert(
751 !h5::meta::is_stl_like<element_t>::value ||
752 storage == sr_t::ragged_vlen_dataset ||
753 storage == sr_t::vlen_text_dataset ||
754 storage == sr_t::array_dataset ||
755 storage == sr_t::array_element ||
756 storage == sr_t::fls_dataset ||
757 storage == sr_t::fixed_inner_extent_dataset,
758 "h5::read<T>: containers of containers are only supported for vector<string> "
759 "(vlen_text_dataset) or vector<vector<T>> (ragged_vlen_dataset).");
760
761 h5::count_t size;
762 const h5::count_t& count = arg::get(size, args...);
763 h5::block_t default_block{1,1,1,1,1,1,1};
764 const h5::block_t& block = arg::get(default_block, args...);
765
766 if constexpr (!tcount::present) {
767 h5::sp_t file_space = h5::get_space(ds);
768 h5::get_simple_extent_dims(file_space, size);
769 } else {
770 for (int i = 0; i < count.rank; ++i) size[i] = count[i] * block[i];
771 size.rank = count.rank;
772 }
773
774 if constexpr (storage == sr_t::array_element) {
775 // Return-style read of std::array<T,N> / T[N] — construct T (which
776 // holds N consecutive elements) then H5Dread into traits::data(ref).
777 // Use the file's stored datatype directly via H5Dget_type so the
778 // in-memory and on-disk types are guaranteed to match (avoids
779 // HDF5's "no conversion path" error when two H5Tarray_create calls
780 // produce structurally identical but distinct type IDs).
781 T ref{};
782 hid_t mem_type = H5Dget_type(static_cast<hid_t>(ds));
783 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
784 // Use H5S_ALL for both spaces — equivalent to the dataset's own
785 // scalar dataspace with all elements selected. Avoids the
786 // "different number of elements selected" mismatch the explicit
787 // H5Sselect_all path produces on scalar dataspaces.
788 H5CPP_CHECK_NZ(
789 H5Dread(static_cast<hid_t>(ds), mem_type,
790 H5S_ALL, H5S_ALL,
791 static_cast<hid_t>(dxpl), traits::data(ref)),
792 h5::error::io::dataset::read, h5::error::msg::read_dataset);
793 H5Tclose(mem_type);
794 return ref;
795 } else if constexpr (storage == sr_t::array_dataset) {
796 using inner_t = std::remove_cv_t<typename T::value_type>;
797 using elem_scalar = typename inner_t::value_type;
798 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
799 hsize_t array_dims[1] = { static_cast<hsize_t>(N_inner) };
800 h5::meta::resolved_type_t<elem_scalar> base_type;
801 hid_t array_type = H5Tarray_create(static_cast<hid_t>(base_type), 1, array_dims);
802 h5::sp_t file_space = h5::get_space(ds);
803 hsize_t outer = 0;
804 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &outer, nullptr);
805 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
806 h5::sp_t mem_space = h5::create_simple(outer);
807 h5::select_all(mem_space);
808 H5Sselect_all(static_cast<hid_t>(file_space));
809 // Read into scratch (always — return-style can't assume direct
810 // .data() access since list / set / forward_list / etc. lack it).
811 std::vector<inner_t> scratch(static_cast<std::size_t>(outer));
812 H5CPP_CHECK_NZ(
813 H5Dread(static_cast<hid_t>(ds), array_type,
814 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
815 static_cast<hid_t>(dxpl), scratch.empty() ? nullptr : scratch.front().data()),
816 h5::error::io::dataset::read, h5::error::msg::read_dataset);
817 H5Tclose(array_type);
818 if constexpr (h5::meta::is_set_like<T>::value) {
819 return T(scratch.begin(), scratch.end());
820 } else if constexpr (std::is_same_v<T, std::vector<inner_t>>) {
821 return T(scratch.begin(), scratch.end());
822 } else {
823 T ref;
824 if constexpr (std::is_same_v<T, std::forward_list<inner_t>>) {
825 ref.assign(scratch.begin(), scratch.end());
826 } else {
827 ref.assign(scratch.begin(), scratch.end());
828 }
829 return ref;
830 }
831 } else if constexpr (storage == sr_t::fls_dataset) {
832 using inner_t = std::remove_cv_t<typename T::value_type>;
833 constexpr std::size_t N_inner = std::tuple_size<inner_t>::value;
834 hid_t str_type = H5Tcopy(H5T_C_S1);
835 H5Tset_size(str_type, N_inner);
836 h5::sp_t file_space = h5::get_space(ds);
837 hsize_t outer = 0;
838 H5Sget_simple_extent_dims(static_cast<hid_t>(file_space), &outer, nullptr);
839 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
840 h5::sp_t mem_space = h5::create_simple(outer);
841 h5::select_all(mem_space);
842 H5Sselect_all(static_cast<hid_t>(file_space));
843 std::vector<inner_t> scratch(static_cast<std::size_t>(outer));
844 H5CPP_CHECK_NZ(
845 H5Dread(static_cast<hid_t>(ds), str_type,
846 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
847 static_cast<hid_t>(dxpl), scratch.empty() ? nullptr : scratch.front().data()),
848 h5::error::io::dataset::read, h5::error::msg::read_dataset);
849 H5Tclose(str_type);
850 if constexpr (h5::meta::is_set_like<T>::value) {
851 return T(scratch.begin(), scratch.end());
852 } else {
853 return T(scratch.begin(), scratch.end());
854 }
855 } else if constexpr (kind == h5::meta::access_t::composite) {
856 // scalar composite (std::tuple<Ts...>): H5Dread into pack buffer, unpack into ref.
857 std::vector<char> buf(traits::bytes());
858 h5::meta::resolved_type_t<T> mem_type;
859 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
860 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
861 h5::sp_t file_space = h5::get_space(ds);
862 H5CPP_CHECK_NZ(
863 H5Dread(static_cast<hid_t>(ds), static_cast<hid_t>(mem_type),
864 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
865 static_cast<hid_t>(dxpl), buf.data()),
866 h5::error::io::dataset::read, h5::error::msg::read_dataset);
867 T ref{};
868 traits::unpack(ref, buf.data());
869 return ref;
870 } else if constexpr (h5::meta::access_kind_v<element_t> == h5::meta::access_t::composite) {
871 // container-of-composite (vector<tuple>, list<tuple>, set<tuple>, deque<tuple>):
872 // read pack buffer, unpack each element, range-construct T (or assign for forward_list).
873 using elem_traits = h5::meta::access_traits_t<element_t>;
874 std::size_t n = 1;
875 for (int i = 0; i < size.rank; ++i) n *= size[i];
876 std::vector<char> buf(n * elem_traits::bytes());
877 h5::meta::resolved_type_t<element_t> mem_type;
878 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
879 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(n));
880 h5::select_all(mem_space);
881 h5::sp_t file_space = h5::get_space(ds);
882 H5Sselect_all(static_cast<hid_t>(file_space));
883 H5CPP_CHECK_NZ(
884 H5Dread(static_cast<hid_t>(ds), static_cast<hid_t>(mem_type),
885 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
886 static_cast<hid_t>(dxpl), buf.data()),
887 h5::error::io::dataset::read, h5::error::msg::read_dataset);
888 std::vector<element_t> flat;
889 flat.reserve(n);
890 for (std::size_t i = 0; i < n; ++i) {
891 element_t elem{};
892 elem_traits::unpack(elem, buf.data() + i * elem_traits::bytes());
893 flat.push_back(std::move(elem));
894 }
895 if constexpr (std::is_same_v<T, std::forward_list<element_t>>) {
896 T result; result.assign(flat.begin(), flat.end()); return result;
897 } else { return T(flat.begin(), flat.end()); }
898 } else if constexpr (kind == h5::meta::access_t::text) {
899 // Scalar text return-style read — variable-length lands a
900 // freshly-owned T (std::string) via char** relay; fixed-length
901 // (char[N]) isn't returnable by value (C-arrays can't be), so
902 // this branch only handles vlen_text. char[N] return-style is
903 // rejected with a static_assert at the call site.
904 static_assert(storage != sr_t::fixed_length_string,
905 "h5::read<char[N]>(fd, path) is not supported — C-arrays "
906 "cannot be returned by value. Use the by-reference form: "
907 "`char buf[N]; h5::read(fd, path, buf);`");
908 hid_t vlen_str = H5Tcopy(H5T_C_S1);
909 H5Tset_size(vlen_str, H5T_VARIABLE);
910 char* relay = nullptr;
911 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
912 h5::sp_t mem_space{H5Screate(H5S_SCALAR)};
913 h5::sp_t file_space = h5::get_space(ds);
914 H5Sselect_all(static_cast<hid_t>(file_space));
915 H5CPP_CHECK_NZ(
916 H5Dread(static_cast<hid_t>(ds), vlen_str,
917 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
918 static_cast<hid_t>(dxpl), &relay),
919 h5::error::io::dataset::read, h5::error::msg::read_dataset);
920 T ref = relay ? T(relay) : T{};
921 h5::impl::reference::reclaim(vlen_str, static_cast<hid_t>(mem_space), H5P_DEFAULT, &relay);
922 H5Tclose(vlen_str);
923 return ref;
924 } else if constexpr (storage == sr_t::vlen_text_dataset) {
925 // T = vector<string>: char** relay + reclaim.
926 // Partial IO: when tcount::present the user passed h5::count{};
927 // `size` already encodes count*block (see the cascade prologue),
928 // so size[0] is the number of strings to read. We size the relay
929 // to that, then narrow the file_space selection to the matching
930 // hyperslab. Without tcount::present the prior `size` was set
931 // from the file extent — select_all on file_space matches.
932 // Mirrors the numeric pointer-read path at line ~118 and the
933 // 2018 std::vector<std::string> specialization that the
934 // access_traits_t × storage_representation_v rewire (commits
935 // 30ca72af / 536408fa) consolidated but lost the hyperslab call.
936 std::size_t n = static_cast<std::size_t>(size[0]);
937 T ref; ref.resize(n);
938 std::vector<char*> relay(n, nullptr);
939 hid_t vlen_str = H5Tcopy(H5T_C_S1);
940 H5Tset_size(vlen_str, H5T_VARIABLE);
941 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
942 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(n));
943 h5::select_all(mem_space);
944 h5::sp_t file_space = h5::get_space(ds);
945 if constexpr (tcount::present) {
946 h5::offset_t default_offset{0,0,0,0,0,0,0};
947 h5::stride_t default_stride{1,1,1,1,1,1,1};
948 h5::block_t default_block_loc{1,1,1,1,1,1,1};
949 const h5::offset_t& offset = arg::get(default_offset, args...);
950 const h5::stride_t& stride = arg::get(default_stride, args...);
951 const h5::block_t& hyper_block = arg::get(default_block_loc, args...);
952 h5::select_hyperslab(file_space, offset, stride, count, hyper_block);
953 } else {
954 H5Sselect_all(static_cast<hid_t>(file_space));
955 }
956 H5CPP_CHECK_NZ(
957 H5Dread(static_cast<hid_t>(ds), vlen_str,
958 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
959 static_cast<hid_t>(dxpl), relay.data()),
960 h5::error::io::dataset::read, h5::error::msg::read_dataset);
961 for (std::size_t i = 0; i < n; ++i)
962 ref[i] = relay[i] ? std::string(relay[i]) : std::string{};
963 h5::impl::reference::reclaim(vlen_str, static_cast<hid_t>(mem_space), H5P_DEFAULT, relay.data());
964 H5Tclose(vlen_str);
965 return ref;
966 } else if constexpr (storage == sr_t::ragged_vlen_dataset) {
967 // T = vector<vector<E>>: hvl_t relay + reclaim
968 using inner_t = typename traits::element_t;
969 using elem_t = typename inner_t::value_type;
970 std::size_t n = static_cast<std::size_t>(size[0]);
971 T ref; ref.resize(n);
972 std::vector<hvl_t> relay(n);
973 h5::meta::resolved_type_t<elem_t> base_type;
974 hid_t vlen_type = H5Tvlen_create(static_cast<hid_t>(base_type));
975 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
976 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(n));
977 h5::select_all(mem_space);
978 h5::sp_t file_space = h5::get_space(ds);
979 H5Sselect_all(static_cast<hid_t>(file_space));
980 H5CPP_CHECK_NZ(
981 H5Dread(static_cast<hid_t>(ds), vlen_type,
982 static_cast<hid_t>(mem_space), static_cast<hid_t>(file_space),
983 static_cast<hid_t>(dxpl), relay.data()),
984 h5::error::io::dataset::read, h5::error::msg::read_dataset);
985 for (std::size_t i = 0; i < n; ++i) {
986 const elem_t* src = static_cast<const elem_t*>(relay[i].p);
987 if constexpr (h5::meta::is_set_like<inner_t>::value ||
988 h5::meta::is_associative_like<inner_t>::value) {
989 ref[i] = inner_t(src, src + relay[i].len);
990 } else {
991 ref[i].assign(src, src + relay[i].len);
992 }
993 }
994 h5::impl::reference::reclaim(vlen_type, static_cast<hid_t>(mem_space), H5P_DEFAULT, relay.data());
995 H5Tclose(vlen_type);
996 return ref;
997 } else if constexpr (storage == sr_t::key_value_dataset) {
998 // T = map<K,V>: flat kv_t compound buffer, insert into map
999 using element_t = typename traits::element_t;
1000 using key_t = std::remove_const_t<typename element_t::first_type>;
1001 using value_t = typename element_t::second_type;
1002 struct kv_t { key_t key; value_t value; };
1003 h5::meta::resolved_type_t<key_t> kt;
1004 h5::meta::resolved_type_t<value_t> vt;
1005 hid_t compound = H5Tcreate(H5T_COMPOUND, sizeof(kv_t));
1006 H5Tinsert(compound, "key", offsetof(kv_t, key), static_cast<hid_t>(kt));
1007 H5Tinsert(compound, "value", offsetof(kv_t, value), static_cast<hid_t>(vt));
1008 std::size_t n = static_cast<std::size_t>(size[0]);
1009 std::vector<kv_t> buffer(n);
1010 h5::sp_t mem_space = h5::create_simple(static_cast<hsize_t>(n));
1011 h5::select_all(mem_space);
1012 h5::sp_t file_space = h5::get_space(ds);
1013 H5Sselect_all(static_cast<hid_t>(file_space));
1014 const h5::dxpl_t& dxpl = arg::get(h5::default_dxpl, args...);
1015 H5CPP_CHECK_NZ(
