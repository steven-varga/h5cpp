7// Worker pool for the parallel filter stage + the per-dataset DAPL parallelism
8// tags (h5::threads{N} / h5::backpressure{M}).
9//
10// h5::write(fd, "ds", data, h5::chunk{C} | h5::gzip{6}, h5::threads{8});
11// h5::create<float>(fd, "ds", h5::chunk{C}, h5::threads{}); // hw_concurrency
12//
13// Parallelism is a per-DATASET concern, so it lives on the DAPL — which, unlike a
14// FAPL property, survives the H5Dget_access_plist round-trip and is read back
15// directly at the write/read site (no #286 fileno registry). One process-global
16// worker_pool_t (global_pool(), below) backs every dataset's filter fan-out:
17// HDF5 (threadsafety-OFF / under the global recursive mutex) funnels all library
18// calls through one thread, so datasets are written sequentially and never
19// contend for the pool — a per-file pool would only oversubscribe.

39// Default in-flight chunk cap when h5::threads{N} is set without an
40// accompanying h5::backpressure{N}. Resolves to 8 × worker_count, which
41// is loose enough to keep all workers fed during steady-state streaming
42// and tight enough to bound memory growth (chunk_size × cap bytes per pt_t).

50// ─── Pool ────────────────────────────────────────────────────────────────────
51//
52// Generic compute pool: workers pull type-erased tasks off a single MPMC-ish
53// queue (std::mutex + std::queue + doorbell signal) and execute them. The
54// submit<F>(callable) API returns a std::future of the callable's result type,
55// using std::packaged_task wrapped in a std::function<void()> for storage.
56//
57// The pool is deliberately HDF5-agnostic at this layer. Consumer sites
58// (Phase 1.3 — pt_t, h5::write, h5::read) wrap their HDF5-specific compress
59// logic in a closure and submit() it. This keeps the pool reusable for any
60// future parallel-compute work.
61struct worker_pool_t {
62 // Pool size is fixed at construction; cannot resize at runtime.
63 // n == 0 means "use std::thread::hardware_concurrency()".
64 explicit worker_pool_t(unsigned n) {
65 // Resolve vendored compressors' lazy CPU-feature dispatch single-threaded
66 // before any worker can run; otherwise the first parallel compress/decompress
67 // self-patches a process-global function pointer from several threads at once
68 // (TSan: data race on libdeflate's 'adler32_impl'). See filter::warm_dispatch.
69 h5::impl::filter::warm_dispatch();
70 const unsigned count = n ? n : std::max(1u, std::thread::hardware_concurrency());
71 workers_.reserve(count);
72 for (unsigned i = 0; i < count; ++i)
73 workers_.emplace_back([this](h5::detail::stop_token_t st) { worker_loop(st); });
74 }
75
76 // Destruction sequence:
77 // 1. wait_idle() — let all submitted tasks complete (caller can also
78 // have called this earlier).
79 // 2. Set stopping_ flag and ring the doorbell so all waiters wake.
80 // 3. stoppable_thread_t destructors automatically request stop + join.
81 ~worker_pool_t() {
82 wait_idle();
83 stopping_.store(true, std::memory_order_release);
84 bell_.ring_all();
85 }
86
87 worker_pool_t(const worker_pool_t&) = delete;
88 worker_pool_t& operator=(const worker_pool_t&) = delete;
89 worker_pool_t(worker_pool_t&&) = delete;
90 worker_pool_t& operator=(worker_pool_t&&) = delete;
91
92 // Introspection for tests and scheduling decisions.
93 [[nodiscard]] unsigned worker_count() const noexcept {
94 return static_cast<unsigned>(workers_.size());
95 }
96
97 // Submit a callable for execution by any worker. Returns a future of
98 // the callable's result type. Callable must be invocable with no
99 // arguments; capture state via closure if needed.
100 //
101 // Internally uses std::packaged_task to bridge the move-only callable
102 // into a copyable std::function (wrapped in shared_ptr) so it can sit
103 // in the queue.
104 template <typename Fn>
105 auto submit(Fn&& fn) -> std::future<std::invoke_result_t<Fn>> {
106 using R = std::invoke_result_t<Fn>;
107 auto task = std::make_shared<std::packaged_task<R()>>(std::forward<Fn>(fn));
108 auto fut = task->get_future();
109 in_flight_.fetch_add(1, std::memory_order_release);
110 {
111 std::lock_guard<std::mutex> lk(m_);
112 tasks_.emplace([task] { (*task)(); });
113 }
114 bell_.ring();
115 return fut;
116 }
117
118 // Block until all submitted tasks have completed. Intended for clean
119 // shutdown or as a synchronization barrier between submission phases.
120 void wait_idle() {
121 while (in_flight_.load(std::memory_order_acquire) > 0)
122 std::this_thread::yield();
123 }
124
125 // Coarse approximation of pending+running work. Useful for tests and
126 // monitoring; not a strict ordering guarantee.
127 [[nodiscard]] int in_flight() const noexcept {
128 return in_flight_.load(std::memory_order_acquire);
129 }
130
131private:
132 void worker_loop(h5::detail::stop_token_t st) {
133 while (!st.stop_requested()) {
134 std::function<void()> task;
135 std::uint32_t last_seq = 0;
136 bool got_task = false;
137 {
138 std::lock_guard<std::mutex> lk(m_);
139 if (!tasks_.empty()) {
140 task = std::move(tasks_.front());
141 tasks_.pop();
142 got_task = true;
143 } else {
144 // Snapshot the doorbell sequence WHILE HOLDING the lock.
145 // submit() will ring the bell only AFTER it has released
146 // the same lock, so any subsequent ring advances past
147 // last_seq — bell_.wait(last_seq) below cannot miss it.
148 last_seq = bell_.load();
149 }
150 }
151
152 if (got_task) {
153 try { task(); } catch (...) { /* packaged_task captures it */ }
154 in_flight_.fetch_sub(1, std::memory_order_release);
155 continue;
156 }
157
158 // Queue was empty — wait for a ring or for shutdown.
159 if (st.stop_requested() ||
160 stopping_.load(std::memory_order_acquire)) return;
161 bell_.wait(last_seq);
162 }
163 }
164
165 std::mutex m_;
166 std::queue<std::function<void()>> tasks_;
167 h5::detail::doorbell_t bell_;
168 std::atomic<int> in_flight_{0};
169 std::atomic<bool> stopping_{false};
170 std::vector<h5::detail::stoppable_thread_t> workers_;
171};

173// ─── Global worker pool ──────────────────────────────────────────────────────
174//
175// One process-wide pool backs every dataset's parallel filter stage. Justified
176// by the single-producer model: HDF5 (threadsafety-OFF, or the global recursive
177// mutex) funnels all library calls through one thread, so datasets are written
178// sequentially and never contend for the pool — a per-file pool would only
179// oversubscribe. Lazily constructed, hardware-sized; size is overridable BEFORE
180// first use via set_pool_size() or the H5CPP_THREADS env var. Lives until program
181// exit (the function-local static's jthreads join at static teardown; the pool
182// owns no HDF5 handles, so teardown order is irrelevant).
183inline unsigned& global_pool_size_override() noexcept { static unsigned n = 0; return n; }
184
185// Set the global pool size; no-op once the pool has been constructed.
186inline void set_pool_size(unsigned n) noexcept { global_pool_size_override() = n; }
187
188inline worker_pool_t& global_pool() {
189 static worker_pool_t pool([]() -> unsigned {
190 unsigned n = global_pool_size_override();
191 if (!n) if (const char* e = std::getenv("H5CPP_THREADS"))
192 n = static_cast<unsigned>(std::strtoul(e, nullptr, 10));
193 return n; // 0 → worker_pool_t maps to hardware_concurrency
194 }());
195 return pool;
196}
197
198// Non-owning shared_ptr alias to the global pool, for consumers (pool_pipeline_t,
199// pt_t) whose interface takes a shared_ptr. The global pool outlives every
200// pipeline, so the deleter is a no-op — nothing here owns the pool.
201inline std::shared_ptr<worker_pool_t> global_pool_ptr() {
202 return std::shared_ptr<worker_pool_t>(&global_pool(), [](worker_pool_t*){});
203}

205// ─── DAPL parallelism properties (the cleanup target) ────────────────────────
206//
207// threads{N} / backpressure{M} become per-DATASET DAPL properties: parallelism
208// is a dataset-IO concern, and — unlike a FAPL property — a DAPL user property
209// SURVIVES the H5Dget_access_plist round-trip, so it is read back directly at
210// the write site with no #286 registry. Both are plain trivially-copyable
211// unsigned values (HDF5's default memcpy copy is correct; no callbacks).
212#define H5CPP_DAPL_THREADS "h5cpp_dapl_threads"
213#define H5CPP_DAPL_BACKPRESSURE "h5cpp_dapl_backpressure"
214
215inline herr_t dapl_threads_set(::hid_t dapl, unsigned n) {
216 if (H5Pexist(dapl, H5CPP_DAPL_THREADS)) return 0;
217 // threads{} (n==0) means "fan out across the whole pool" — store the resolved
218 // worker count so a stored value of 0 unambiguously means "property absent /
219 // no pool" at the read site (resolve_dataset_threads).
220 if (n == 0) n = std::max(1u, std::thread::hardware_concurrency());
221 return H5Pinsert2(dapl, H5CPP_DAPL_THREADS, sizeof(unsigned), &n,
222 nullptr, nullptr, nullptr, nullptr, nullptr, nullptr);
223}
224inline herr_t dapl_backpressure_set(::hid_t dapl, unsigned cap) {
225 if (H5Pexist(dapl, H5CPP_DAPL_BACKPRESSURE)) return 0;
226 return H5Pinsert2(dapl, H5CPP_DAPL_BACKPRESSURE, sizeof(unsigned), &cap,
227 nullptr, nullptr, nullptr, nullptr, nullptr, nullptr);
228}

230// N for this dataset (0 = not requested → no pool / synchronous filters).
231inline unsigned resolve_dataset_threads(::hid_t dapl_id) noexcept {
232 if (dapl_id < 0 || H5Iis_valid(dapl_id) <= 0) return 0;
233 if (!H5Pexist(dapl_id, H5CPP_DAPL_THREADS)) return 0;
234 unsigned n = 0;
235 H5Pget(dapl_id, H5CPP_DAPL_THREADS, &n);
236 return n;
237}

239// In-flight chunk cap for this dataset: explicit backpressure{M}, else the
240// default factor × the dataset's requested concurrency N.
241inline unsigned resolve_dataset_backpressure(::hid_t dapl_id, unsigned n) noexcept {
242 if (dapl_id >= 0 && H5Iis_valid(dapl_id) > 0
243 && H5Pexist(dapl_id, H5CPP_DAPL_BACKPRESSURE)) {
244 unsigned cap = 0;
245 H5Pget(dapl_id, H5CPP_DAPL_BACKPRESSURE, &cap);
246 if (cap > 0) return cap;
247 }
248 return H5CPP_FAPL_BACKPRESSURE_DEFAULT_FACTOR * (n ? n : global_pool().worker_count());
249}

254// User-facing tags. Parallelism is a per-DATASET concern, so these are DAPL
255// properties applied alongside the dataset's other access/create properties:
256//
257// h5::write (fd, "ds", data, h5::chunk{C} | h5::gzip{6}, h5::threads{8});
258// h5::create<float>(fd, "ds", h5::chunk{C}, h5::threads{8} | h5::backpressure{32});
259// h5::pt_t pt = h5::create<float>(fd, "ds", h5::chunk{C}, h5::threads{8});
260//
261// h5::threads{N} marks this dataset to fan its filter stage out across N workers
262// of the process-global pool; h5::threads{} uses hardware_concurrency.
263// h5::backpressure{M} bounds in-flight chunks; without h5::threads{N} it is
264// silently a no-op (no pool → no queue to bound). Unlike the old FAPL pool, a
265// DAPL property survives the H5Dget_access_plist round-trip, so it is read back
266// directly at the write site — no #286 registry.
267using threads = impl::dapl_call<impl::dapl_args<hid_t, unsigned>,
268 impl::dapl_threads_set>;
269using backpressure = impl::dapl_call<impl::dapl_args<hid_t, unsigned>,
270 impl::dapl_backpressure_set>;
271}
