168 enum class storage_representation_t {
169 unsupported, scalar, c_array, linear_value_dataset, key_value_dataset, ragged_vlen_dataset, fixed_inner_extent_dataset, vlen_text_dataset, fixed_length_string,
170 // Canonical fixed-extent mapping (Winston model):

172 // array_dataset — vector/list/set/etc.<std::array<T,N>> (non-char) → rank-1 dataspace of H5T_ARRAY[N] elements

615 // Flat-buffer layout for std::tuple<Ts...>.
616 // std::tuple is not standard-layout, so staging buffers use this custom
617 // C-style layout (each field at its natural alignment, in declaration order).
618 template <class Tuple, std::size_t I, class = void> struct tuple_field_end;
619 template <class Tuple>
620 struct tuple_field_end<Tuple, 0, void> {
621 static constexpr std::size_t value = sizeof(std::tuple_element_t<0, Tuple>);
622 };
623 template <class Tuple, std::size_t I>
624 struct tuple_field_end<Tuple, I, std::enable_if_t<(I > 0)>> {
625 using elem_t = std::tuple_element_t<I, Tuple>;
626 static constexpr std::size_t prev = tuple_field_end<Tuple, I-1>::value;
627 static constexpr std::size_t value = ((prev + alignof(elem_t) - 1) & ~(alignof(elem_t) - 1)) + sizeof(elem_t);
628 };

630 template <class Tuple, std::size_t I, class = void> struct tuple_field_offset;
631 template <class Tuple>
632 struct tuple_field_offset<Tuple, 0, void> {
633 static constexpr std::size_t value = 0;
634 };
635 template <class Tuple, std::size_t I>
636 struct tuple_field_offset<Tuple, I, std::enable_if_t<(I > 0)>> {
637 using elem_t = std::tuple_element_t<I, Tuple>;
638 static constexpr std::size_t value =
639 (tuple_field_end<Tuple, I-1>::value + alignof(elem_t) - 1) & ~(alignof(elem_t) - 1);
640 };

642 template <class... Ts>
643 struct tuple_layout {
644 using tuple_t = std::tuple<Ts...>;
645 static constexpr std::size_t count = sizeof...(Ts);
646
647 template <std::size_t I>
648 static constexpr std::size_t offset() noexcept {
649 return tuple_field_offset<tuple_t, I>::value;
650 }
651 template <std::size_t... Is>
652 static constexpr std::size_t max_align_impl(std::index_sequence<Is...>) noexcept {
653 std::size_t r = 1;
654 ((r = r < alignof(std::tuple_element_t<Is, tuple_t>) ? alignof(std::tuple_element_t<Is, tuple_t>) : r), ...);
655 return r;
656 }
657 static constexpr std::size_t max_alignment() noexcept {
658 return max_align_impl(std::make_index_sequence<count>{});
659 }
660 static constexpr std::size_t total_size() noexcept {
661 constexpr std::size_t end = tuple_field_end<tuple_t, count - 1>::value;
662 constexpr std::size_t ma = max_align_impl(std::make_index_sequence<count>{});
663 return (end + ma - 1) & ~(ma - 1);
664 }
665 };
